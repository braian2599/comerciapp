    97→const emptyForm = {
    98→  id: "",
    99→  name: "",
   100→  description: "",
   101→  barcode: "",
   102→  sku: "",
   103→  categoryId: "",
   104→  costPrice: 0,
   105→  salePrice: 0,
   106→  stock: 0,
   107→  minStock: 5,
   108→  unit: "UNIDAD",
   109→  active: true,

   117→export function ProductsView() {
   118→  const { store } = useAppStore();
   119→  const [products, setProducts] = useState<Product[]>([]);
   120→  const [categories, setCategories] = useState<Category[]>([]);
   121→  const [loading, setLoading] = useState(true);
   122→  const [search, setSearch] = useState("");
   123→  const [filterCat, setFilterCat] = useState("all");
   124→  const [filterStock, setFilterStock] = useState("all");
   125→
   126→  const [formOpen, setFormOpen] = useState(false);
   127→  const [form, setForm] = useState<any>(emptyForm);
   128→  const [saving, setSaving] = useState(false);
   129→  const [deleteId, setDeleteId] = useState<string | null>(null);
   130→  const [catDialogOpen, setCatDialogOpen] = useState(false);
   131→  const [newCatName, setNewCatName] = useState("");
   132→  const [importOpen, setImportOpen] = useState(false);
   133→
   134→  // Lookup por código de barras
   135→  const [lookupLoading, setLookupLoading] = useState(false);
   136→  const [lookupResult, setLookupResult] = useState<
   137→    | {
   138→        status: "idle" | "loading" | "found" | "notfound" | "error";
   139→        source?: string;
   140→        imageUrl?: string;
   141→        labels?: string[];
   142→        allergens?: string[];
   143→      }
   144→    | null
   145→  >(null);
   146→  // Sección colapsable de datos adicionales
   147→  const [showExtraFields, setShowExtraFields] = useState(false);
   148→  // Etiquetas y alérgenos editados como chips
   149→  const [labelChips, setLabelChips] = useState<string[]>([]);
   150→  const [allergenChips, setAllergenChips] = useState<string[]>([]);
   151→  const [labelInput, setLabelInput] = useState("");
   152→  const [allergenInput, setAllergenInput] = useState("");
   153→
   154→  async function handleBarcodeLookup() {
   155→    const code = (form.barcode || "").trim();
   156→    if (!code) {
   157→      toast.error("Ingresá un código de barras primero");
   158→      return;
   159→    }
   160→    setLookupLoading(true);
   161→    setLookupResult({ status: "loading" });
   162→    try {
   163→      const { ok, data, error } = await safeFetchJSON<any>(
   164→        `/api/products/lookup?barcode=${encodeURIComponent(code)}`
   165→      );
   166→      if (!ok || !data) {
   167→        throw new Error(error || "No se pudo consultar la base de productos");
   168→      }
   169→      if (data.found) {
   170→        // Solo autocompletamos campos que el usuario aún no llenó.
   171→        setForm((prev: any) => ({
   172→          ...prev,
   173→          name: prev.name || data.name || "",
   174→          description: prev.description || data.description || data.brand || "",
   175→          barcode: prev.barcode || code,
   176→          brand: prev.brand || data.brand || "",
   177→          ingredients: prev.ingredients || data.ingredients || "",
   178→          imageUrl: prev.imageUrl || data.imageUrl || "",
   179→        }));
   180→        // Merge de etiquetas y alérgenos (sin duplicar)
   181→        if (Array.isArray(data.labels) && data.labels.length > 0) {
   182→          setLabelChips((prev) =>
   183→            Array.from(new Set([...prev, ...data.labels])).slice(0, 15)
   184→          );
   185→          setShowExtraFields(true);
   186→        }
   187→        if (Array.isArray(data.allergens) && data.allergens.length > 0) {
   188→          setAllergenChips((prev) =>
   189→            Array.from(new Set([...prev, ...data.allergens])).slice(0, 15)
   190→          );
   191→          setShowExtraFields(true);
   192→        }
   193→        setLookupResult({
   194→          status: "found",
   195→          source: data.source,
   196→          imageUrl: data.imageUrl,
   197→          labels: data.labels,
   198→          allergens: data.allergens,
   199→        });
   200→        const sourceLabel =
   201→          data.source === "openfoodfacts" ? "Open Food Facts" : "UPC Item DB";
   202→        toast.success(`Producto encontrado en ${sourceLabel}`, {
   203→          description: data.name,
   204→        });
   205→      } else {
   206→        setLookupResult({ status: "notfound" });
   207→        toast.info("No se encontró el código en las bases de datos públicas", {
   208→          description: "Completá los datos manualmente",
   209→        });
   210→      }
   211→    } catch (e: any) {
   212→      setLookupResult({ status: "error" });
   213→      toast.error("No se pudo consultar la base de productos");
   214→    } finally {
   215→      setLookupLoading(false);
   216→    }
   217→  }
   218→
   219→  function resetLookup() {
   220→    setLookupResult(null);
   221→  }
   222→
   223→  const symbol = store?.currencySymbol || "$";
   224→
   225→  async function load() {
   226→    setLoading(true);
   227→    try {
   228→      const [p, c] = await Promise.all([
   229→        safeFetchArray<Product>("/api/products"),
   230→        safeFetchArray<Category>("/api/categories"),
   231→      ]);
   232→      setProducts(p);
   233→      setCategories(c);
   234→    } catch {
   235→      toast.error("No se pudieron cargar los productos");
   236→      setProducts([]);
   237→      setCategories([]);
   238→    } finally {
   239→      setLoading(false);
   240→    }
   241→  }
   242→
   243→  useEffect(() => {
   244→    load();
   245→  }, []);
   246→
   247→  const filtered = useMemo(() => {
   248→    return products.filter((p) => {
   249→      const matchSearch =
   250→        !search ||
   251→        p.name.toLowerCase().includes(search.toLowerCase()) ||
   252→        p.barcode?.includes(search) ||
   253→        p.sku?.toLowerCase().includes(search.toLowerCase());
   254→      const matchCat =
   255→        filterCat === "all" || p.categoryId === filterCat;
   256→      const matchStock =
   257→        filterStock === "all" ||
   258→        (filterStock === "low" && p.stock <= p.minStock) ||
   259→        (filterStock === "out" && p.stock <= 0) ||
   260→        (filterStock === "ok" && p.stock > p.minStock);
   261→      return matchSearch && matchCat && matchStock;
   262→    });
   263→  }, [products, search, filterCat, filterStock]);
   264→
   265→  function openNew() {
   266→    setForm({ ...emptyForm });
   267→    resetLookup();
   268→    setLabelChips([]);
   269→    setAllergenChips([]);
   270→    setShowExtraFields(false);
   271→    setFormOpen(true);
   272→  }
   273→
   274→  function openEdit(p: Product) {
   275→    setForm({
   276→      ...p,
   277→      costPrice: p.costPrice,
   278→      salePrice: p.salePrice,
   279→      stock: p.stock,
   280→      minStock: p.minStock,
   281→      adjustReason: "",
   282→    });
   283→    resetLookup();
   284→    setLabelChips(p.labels ? p.labels.split(",").map((s) => s.trim()).filter(Boolean) : []);
   285→    setAllergenChips(p.allergens ? p.allergens.split(",").map((s) => s.trim()).filter(Boolean) : []);
   286→    setShowExtraFields(
   287→      !!(p.brand || p.ingredients || p.labels || p.allergens || p.imageUrl)
   288→    );
   289→    setFormOpen(true);
   290→  }
   291→
   292→  async function handleSave() {
   293→    if (!form.name || !form.salePrice) {
   294→      toast.error("Nombre y precio de venta son obligatorios");
   295→      return;
   296→    }
   297→    setSaving(true);
   298→    try {
   299→      const method = form.id ? "PUT" : "POST";
   300→      const payload = {
   301→        ...form,
   302→        labels: labelChips.join(","),
   303→        allergens: allergenChips.join(","),
   304→      };
   305→      const { ok, error } = await safeFetchJSON("/api/products", {
   306→        method,
   307→        body: JSON.stringify(payload),
   308→      });
   309→      if (!ok) throw new Error(error);
   310→      toast.success(form.id ? "Producto actualizado" : "Producto creado");
   311→      setFormOpen(false);
   312→      load();
   313→    } catch (e: any) {
   314→      const msg = e?.message || "No se pudo guardar el producto";
   315→      toast.error("Error al guardar el producto", { description: msg });
   316→    } finally {
   317→      setSaving(false);
   318→    }
   319→  }
   320→
   321→  async function handleDelete() {
   322→    if (!deleteId) return;
   323→    try {
   324→      const { ok, error } = await safeFetchJSON(
   325→        `/api/products?id=${deleteId}`,
   326→        { method: "DELETE" }
   327→      );
   328→      if (!ok) throw new Error(error);
   329→      toast.success("Producto desactivado");
   330→      setDeleteId(null);
   331→      load();
   332→    } catch (e: any) {
   333→      toast.error("Error al eliminar", { description: e.message });
   334→    }
   335→  }
   336→
   337→  async function handleAddCategory() {
   338→    if (!newCatName.trim()) return;
   339→    try {
   340→      const { ok, error } = await safeFetchJSON("/api/categories", {
   341→        method: "POST",
   342→        body: JSON.stringify({ name: newCatName }),
   343→      });
   344→      if (!ok) throw new Error(error);
   345→      toast.success("Categoría creada");
   346→      setNewCatName("");
   347→      setCatDialogOpen(false);
   348→      load();
   349→    } catch (e: any) {
   350→      toast.error("Error al crear categoría", { description: e.message });
   351→    }
   352→  }
   353→
   354→  async function handleDeleteCategory(id: string, name: string) {
   355→    if (!confirm(`¿Eliminar la categoría "${name}"? Los productos quedarán sin categoría.`)) return;
   356→    try {
   357→      const { ok, error } = await safeFetchJSON(`/api/categories?id=${id}`, {
   358→        method: "DELETE",
   359→      });
   360→      if (!ok) throw new Error(error);
   361→      toast.success("Categoría eliminada");
   362→      load();
   363→    } catch (e: any) {
   364→      toast.error("Error al eliminar categoría", { description: e.message });
   365→    }
   366→  }
   367→
   368→  return (
   369→    <div className="space-y-4">
   370→      <ImportDialog
   371→        open={importOpen}
   372→        onOpenChange={setImportOpen}
   373→        endpoint="/api/products/import"
   374→        fields={PRODUCT_IMPORT_FIELDS}
   375→        templateHeaders={[
   376→          "name",
   377→          "barcode",
   378→          "sku",
   379→          "category",
   380→          "costPrice",
   381→          "salePrice",
   382→          "stock",
   383→          "minStock",
   384→          "unit",
   385→          "active",
   386→          "brand",
   387→          "description",
   388→          "labels",
   389→          "ingredients",
   390→          "allergens",
   391→          "imageUrl",
   392→        ]}
   393→        templateName="plantilla_productos.csv"
   394→        entityLabel="producto"
   395→        entityLabelPlural="productos"
   396→        onImported={load}
   397→      />
   398→      <div className="flex items-center justify-between flex-wrap gap-3">
   399→        <div>
   400→          <h1 className="text-2xl font-bold tracking-tight">Productos</h1>
   401→          <p className="text-sm text-muted-foreground">
   402→            {products.length} productos · {categories.length} categorías
   403→          </p>
   404→        </div>
   405→        <div className="flex gap-2">
   406→          <Dialog open={catDialogOpen} onOpenChange={setCatDialogOpen}>
   407→            <DialogTrigger asChild>
   408→              <Button variant="outline">
   409→                <Tag className="w-4 h-4 mr-2" />
   410→                Categorías
   411→              </Button>
   412→            </DialogTrigger>
   413→            <DialogContent>
   414→              <DialogHeader>
   415→                <DialogTitle>Gestionar categorías</DialogTitle>
   416→                <DialogDescription>
   417→                  Agregá o eliminá categorías para organizar tus productos
   418→                </DialogDescription>
   419→              </DialogHeader>
   420→              <div className="flex gap-2">
   421→                <Input
   422→                  placeholder="Nueva categoría..."
   423→                  value={newCatName}
   424→                  onChange={(e) => setNewCatName(e.target.value)}
   425→                  onKeyDown={(e) => e.key === "Enter" && handleAddCategory()}
   426→                />
   427→                <Button onClick={handleAddCategory} className="bg-indigo-600 hover:bg-indigo-700">
   428→                  <Plus className="w-4 h-4" />
   429→                </Button>
   430→              </div>
   431→              <div className="max-h-60 overflow-y-auto space-y-1 mt-2">
   432→                {categories.length === 0 ? (
   433→                  <p className="text-sm text-muted-foreground text-center py-4">
   434→                    Sin categorías
   435→                  </p>
   436→                ) : (
   437→                  categories.map((c) => (
   438→                    <div
   439→                      key={c.id}
   440→                      className="flex items-center justify-between p-2 rounded-md bg-muted/50"
   441→                    >
   442→                      <div>
   443→                        <p className="text-sm font-medium">{c.name}</p>
   444→                        <p className="text-xs text-muted-foreground">
   445→                          {c._count?.products || 0} productos
   446→                        </p>
   447→                      </div>
   448→                      <Button
   449→                        variant="ghost"
   450→                        size="icon"
   451→                        className="h-7 w-7 text-red-600"
   452→                        onClick={() => handleDeleteCategory(c.id, c.name)}
   453→                      >
   454→                        <Trash2 className="w-3.5 h-3.5" />
   455→                      </Button>
   456→                    </div>
   457→                  ))
   458→                )}
   459→              </div>
   460→            </DialogContent>
   461→          </Dialog>
   462→
   463→          <Button
   464→            variant="outline"
   465→            onClick={() => setImportOpen(true)}
   466→          >
   467→            <Upload className="w-4 h-4 mr-2" />
   468→            Importar
   469→          </Button>
   470→          <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   471→            <Plus className="w-4 h-4 mr-2" />
   472→            Nuevo producto
   473→          </Button>
   474→        </div>
   475→      </div>
   476→
   477→      {/* Filtros */}
   478→      <Card>
   479→        <CardContent className="p-3">
   480→          <div className="grid gap-2 grid-cols-1 sm:grid-cols-3">
   481→            <div className="relative">
   482→              <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   483→              <Input
   484→                placeholder="Buscar por nombre, código o SKU..."
   485→                value={search}
   486→                onChange={(e) => setSearch(e.target.value)}
   487→                className="pl-9"
   488→              />
   489→            </div>
   490→            <Select value={filterCat} onValueChange={setFilterCat}>
   491→              <SelectTrigger>
   492→                <SelectValue placeholder="Todas las categorías" />
   493→              </SelectTrigger>
   494→              <SelectContent>
   495→                <SelectItem value="all">Todas las categorías</SelectItem>
   496→                {categories.map((c) => (
   497→                  <SelectItem key={c.id} value={c.id}>
   498→                    {c.name}
   499→                  </SelectItem>
   500→                ))}
   501→              </SelectContent>
   502→            </Select>
   503→            <Select value={filterStock} onValueChange={setFilterStock}>
   504→              <SelectTrigger>
   505→                <SelectValue />
   506→              </SelectTrigger>
   507→              <SelectContent>
   508→                <SelectItem value="all">Todo el stock</SelectItem>
   509→                <SelectItem value="ok">Stock OK</SelectItem>
   510→                <SelectItem value="low">Stock bajo</SelectItem>
   511→                <SelectItem value="out">Sin stock</SelectItem>
   512→              </SelectContent>
   513→            </Select>
   514→          </div>
   515→        </CardContent>
   516→      </Card>
   517→
   518→      {/* Tabla */}
   519→      <Card>
   520→        <CardContent className="p-0">
   521→          {loading ? (
   522→            <div className="p-8 text-center text-muted-foreground">
   523→              <Loader2 className="w-5 h-5 animate-spin inline mr-2" />
   524→              Cargando productos...
   525→            </div>
   526→          ) : filtered.length === 0 ? (
   527→            <div className="p-12 text-center">
   528→              <Package className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   529→              <p className="text-sm text-muted-foreground">
   530→                {search || filterCat !== "all" || filterStock !== "all"
   531→                  ? "No se encontraron productos con esos filtros"
   532→                  : "Aún no tenés productos. Creá el primero!"}
   533→              </p>
   534→            </div>
   535→          ) : (
   536→            <div className="overflow-x-auto">
   537→              <Table>
   538→                <TableHeader>
   539→                  <TableRow>
   540→                    <TableHead>Nombre</TableHead>
   541→                    <TableHead className="hidden md:table-cell">Categoría</TableHead>
   542→                    <TableHead className="text-right">Costo</TableHead>
   543→                    <TableHead className="text-right">Precio</TableHead>
   544→                    <TableHead className="text-right">Stock</TableHead>
   545→                    <TableHead className="text-center">Estado</TableHead>
   546→                    <TableHead className="text-right">Acciones</TableHead>
   547→                  </TableRow>
   548→                </TableHeader>
   549→                <TableBody>
   550→                  {filtered.map((p) => (
   551→                    <TableRow key={p.id} className={!p.active ? "opacity-50" : ""}>
   552→                      <TableCell>
   553→                        <div>
   554→                          <p className="font-medium">{p.name}</p>
   555→                          {(p.barcode || p.sku) && (
   556→                            <p className="text-xs text-muted-foreground">
   557→                              {p.barcode && `Cód: ${p.barcode}`}
   558→                              {p.barcode && p.sku && " · "}
   559→                              {p.sku && `SKU: ${p.sku}`}
   560→                            </p>
   561→                          )}
   562→                        </div>
   563→                      </TableCell>
   564→                      <TableCell className="hidden md:table-cell">
   565→                        {p.category ? (
   566→                          <Badge variant="outline">{p.category.name}</Badge>
   567→                        ) : (
   568→                          <span className="text-xs text-muted-foreground">—</span>
   569→                        )}
   570→                      </TableCell>
   571→                      <TableCell className="text-right text-muted-foreground">
   572→                        {formatCurrency(p.costPrice, symbol)}
   573→                      </TableCell>
   574→                      <TableCell className="text-right font-semibold">
   575→                        {formatCurrency(p.salePrice, symbol)}
   576→                      </TableCell>
   577→                      <TableCell className="text-right">
   578→                        <span
   579→                          className={
   580→                            p.stock <= 0
   581→                              ? "text-red-600 font-medium"
   582→                              : p.stock <= p.minStock
   583→                              ? "text-amber-600 font-medium"
   584→                              : "text-indigo-700"
   585→                          }
   586→                        >
   587→                          {p.stock} {p.unit === "KG" ? "kg" : "u"}
   588→                        </span>
   589→                      </TableCell>
   590→                      <TableCell className="text-center">
   591→                        {p.active ? (
   592→                          <Badge className="bg-indigo-100 text-indigo-700 hover:bg-indigo-100">
   593→                            Activo
   594→                          </Badge>
   595→                        ) : (
   596→                          <Badge variant="secondary">Inactivo</Badge>
   597→                        )}
   598→                      </TableCell>
   599→                      <TableCell className="text-right">
   600→                        <Button
   601→                          variant="ghost"
   602→                          size="icon"
   603→                          className="h-8 w-8"
