   146→// ===== VENTA =====
   147→model Sale {
   148→  id              String   @id @default(cuid())
   149→  storeId         String
   150→  userId          String
   151→  customerId      String?
   152→  cashRegisterId  String?
   153→  branchId        String?  // sucursal donde se realizó (null = principal)
   154→  subtotal        Float
   155→  discount        Float    @default(0)
   156→  discountReason  String?  // motivo del descuento (PROMOCION, MANUAL, PUNTOS, etc.)
   157→  tax             Float    @default(0)
   158→  surcharge       Float    @default(0) // recargo por método de pago
   159→  total           Float
   160→  paymentMethod   String   @default("EFECTIVO") // nombre del método (snapshot)
   161→  paymentMethodId String?
   162→  // Promociones aplicadas (resumen)
   163→  promotionId     String?  // promoción principal aplicada
   164→  promotionDiscount Float  @default(0) // monto descontado por promociones automáticas
   165→  // Fidelización
   166→  loyaltyPointsEarned Float @default(0) // puntos ganados en esta venta
   167→  loyaltyPointsUsed   Float @default(0) // puntos canjeados (descuento aplicado)
   168→  // Cuenta corriente
   169→  onCredit        Boolean  @default(false) // true = venta fiada
   170→  amountPaid      Float    @default(0) // monto cobrado (total si es contado, 0 si es fiado completo)
   171→  status          String   @default("COMPLETADA") // COMPLETADA, ANULADA
   172→  notes           String?
   173→  createdAt       DateTime @default(now())
   174→  updatedAt       DateTime @updatedAt
   175→
   176→  store           Store           @relation(fields: [storeId], references: [id], onDelete: Cascade)
   177→  user            User            @relation(fields: [userId], references: [id])
   178→  customer        Customer?       @relation(fields: [customerId], references: [id], onDelete: SetNull)
   179→  paymentMethodRef PaymentMethod? @relation(fields: [paymentMethodId], references: [id], onDelete: SetNull)
   180→  cashRegister    CashRegister?   @relation(fields: [cashRegisterId], references: [id], onDelete: SetNull)
   181→  branch          Branch?         @relation(fields: [branchId], references: [id], onDelete: SetNull)
   182→  promotion       Promotion?      @relation(fields: [promotionId], references: [id], onDelete: SetNull)
   183→  items           SaleItem[]
   184→  // Facturación y Mercado Pago
   185→  invoice         Invoice?
   186→  mercadoPagoPayment MercadoPagoPayment?
   187→  refund          Refund?
   188→
   189→  @@index([storeId, createdAt])
   190→  @@index([storeId, status])
   191→  @@index([customerId, onCredit])
   192→  @@index([branchId, createdAt])
   193→}

   436→// ===== FACTURA ELECTRÓNICA =====
   437→// Representa una factura electrónica emitida ante AFIP/ARCA
   438→model Invoice {
   439→  id              String   @id @default(cuid())
   440→  storeId         String
   441→  userId          String   // usuario que emitió
   442→  saleId          String?  @unique // venta asociada (opcional: una factura puede existir sin venta)
   443→  customerId      String?  // cliente (null = Consumidor Final)
   444→  taxConfigId     String
   445→  // Datos de la factura
   446→  tipo            String   // A, B, C, M, E
   447→  puntoVenta      Int
   448→  numero          Int      // número de factura dentro del punto de venta
   449→  numeroCompleto  String   // "0001-00000001" (legible)
   450→  fechaEmision    DateTime @default(now())
   451→  concepto        String   @default("PRODUCTOS") // PRODUCTOS, SERVICIOS, PRODUCTOS_Y_SERVICIOS
   452→  // Totales
   453→  netoGravado     Float    @default(0) // base imponible sin IVA
   454→  ivaRate         Float    @default(0)
   455→  ivaAmount       Float    @default(0) // monto de IVA
   456→  noGravado       Float    @default(0) // importes no gravados
   457→  exento          Float    @default(0) // importes exentos
   458→  total           Float
   459→  // Datos del cliente (snapshot)
   460→  customerName    String   @default("Consumidor Final")
   461→  customerCuit    String?  // CUIT del cliente (null si es CF)
   462→  customerTaxType String   @default("CONSUMIDOR_FINAL")
   463→  // Respuesta AFIP
   464→  cae             String?  // Código de Autorización Electrónico
   465→  caeVencimiento  DateTime? // fecha de vencimiento del CAE
   466→  qrData          String?  // datos codificados para el QR AFIP (RG 4291)
   467→  // Estado
   468→  status          String   @default("EMITIDA") // EMITIDA, ANULADA, RECHAZADA, PENDIENTE
   469→  observation     String?  // observaciones AFIP
   470→  resultado       String?  // A (aprobado), R (rechazado), O (observado)
   471→  // Anulación
   472→  anulatedAt      DateTime?
   473→  anulatedBy      String?
   474→  createdAt       DateTime @default(now())
   475→  updatedAt       DateTime @updatedAt
   476→
   477→  store           Store    @relation(fields: [storeId], references: [id], onDelete: Cascade)
   478→  user            User     @relation(fields: [userId], references: [id])
   479→  sale            Sale?    @relation(fields: [saleId], references: [id], onDelete: SetNull)
   480→  customer        Customer? @relation(fields: [customerId], references: [id], onDelete: SetNull)
   481→  taxConfig       TaxConfig @relation(fields: [taxConfigId], references: [id])
   482→
   483→  @@index([storeId, fechaEmision])
   484→  @@index([storeId, tipo, numero])
   485→  @@index([saleId])
   486→  @@index([cae])
   487→}
