   166→// ===== VENTA =====
   167→model Sale {
   168→  id              String   @id @default(cuid())
   169→  storeId         String
   170→  userId          String
   171→  customerId      String?
   172→  cashRegisterId  String?
   173→  branchId        String?  // sucursal donde se realizó (null = principal)
   174→  subtotal        Float
   175→  discount        Float    @default(0)
   176→  discountReason  String?  // motivo del descuento (PROMOCION, MANUAL, PUNTOS, etc.)
   177→  tax             Float    @default(0)
   178→  surcharge       Float    @default(0) // recargo por método de pago
   179→  total           Float
   180→  paymentMethod   String   @default("EFECTIVO") // nombre del método (snapshot)
   181→  paymentMethodId String?
   182→  // Promociones aplicadas (resumen)
   183→  promotionId     String?  // promoción principal aplicada
   184→  promotionDiscount Float  @default(0) // monto descontado por promociones automáticas
   185→  // Fidelización
   186→  loyaltyPointsEarned Float @default(0) // puntos ganados en esta venta
   187→  loyaltyPointsUsed   Float @default(0) // puntos canjeados (descuento aplicado)
   188→  // Cuenta corriente
   189→  onCredit        Boolean  @default(false) // true = venta fiada
   190→  amountPaid      Float    @default(0) // monto cobrado (total si es contado, 0 si es fiado completo)
   191→  status          String   @default("COMPLETADA") // COMPLETADA, ANULADA
   192→  notes           String?
   193→  createdAt       DateTime @default(now())
   194→  updatedAt       DateTime @updatedAt
   195→
   196→  store           Store           @relation(fields: [storeId], references: [id], onDelete: Cascade)
   197→  user            User            @relation(fields: [userId], references: [id])
   198→  customer        Customer?       @relation(fields: [customerId], references: [id], onDelete: SetNull)
   199→  paymentMethodRef PaymentMethod? @relation(fields: [paymentMethodId], references: [id], onDelete: SetNull)
   200→  cashRegister    CashRegister?   @relation(fields: [cashRegisterId], references: [id], onDelete: SetNull)
   201→  branch          Branch?         @relation(fields: [branchId], references: [id], onDelete: SetNull)
   202→  promotion       Promotion?      @relation(fields: [promotionId], references: [id], onDelete: SetNull)
   203→  items           SaleItem[]
   204→  // Facturación y Mercado Pago
   205→  invoice         Invoice?
   206→  mercadoPagoPayment MercadoPagoPayment?
   207→  refund          Refund?
   208→  // Fase 4: Comisión asociada (1:1)
   209→  commission      Commission?
   210→
   211→  @@index([storeId, createdAt])
   212→  @@index([storeId, status])
   213→  @@index([customerId, onCredit])
   214→  @@index([branchId, createdAt])
   215→}

   458→// ===== FACTURA ELECTRÓNICA =====
   459→// Representa una factura electrónica emitida ante AFIP/ARCA
   460→model Invoice {
   461→  id              String   @id @default(cuid())
   462→  storeId         String
   463→  userId          String   // usuario que emitió
   464→  saleId          String?  @unique // venta asociada (opcional: una factura puede existir sin venta)
   465→  customerId      String?  // cliente (null = Consumidor Final)
   466→  taxConfigId     String
   467→  // Datos de la factura
   468→  tipo            String   // A, B, C, M, E
   469→  puntoVenta      Int
   470→  numero          Int      // número de factura dentro del punto de venta
   471→  numeroCompleto  String   // "0001-00000001" (legible)
   472→  fechaEmision    DateTime @default(now())
   473→  concepto        String   @default("PRODUCTOS") // PRODUCTOS, SERVICIOS, PRODUCTOS_Y_SERVICIOS
   474→  // Totales
   475→  netoGravado     Float    @default(0) // base imponible sin IVA
   476→  ivaRate         Float    @default(0)
   477→  ivaAmount       Float    @default(0) // monto de IVA
   478→  noGravado       Float    @default(0) // importes no gravados
   479→  exento          Float    @default(0) // importes exentos
   480→  total           Float
   481→  // Datos del cliente (snapshot)
   482→  customerName    String   @default("Consumidor Final")
   483→  customerCuit    String?  // CUIT del cliente (null si es CF)
   484→  customerTaxType String   @default("CONSUMIDOR_FINAL")
   485→  // Respuesta AFIP
   486→  cae             String?  // Código de Autorización Electrónico
   487→  caeVencimiento  DateTime? // fecha de vencimiento del CAE
   488→  qrData          String?  // datos codificados para el QR AFIP (RG 4291)
   489→  // Estado
   490→  status          String   @default("EMITIDA") // EMITIDA, ANULADA, RECHAZADA, PENDIENTE
   491→  observation     String?  // observaciones AFIP
   492→  resultado       String?  // A (aprobado), R (rechazado), O (observado)
   493→  // Anulación
   494→  anulatedAt      DateTime?
   495→  anulatedBy      String?
   496→  createdAt       DateTime @default(now())
   497→  updatedAt       DateTime @updatedAt
   498→
   499→  store           Store    @relation(fields: [storeId], references: [id], onDelete: Cascade)
   500→  user            User     @relation(fields: [userId], references: [id])
   501→  sale            Sale?    @relation(fields: [saleId], references: [id], onDelete: SetNull)
   502→  customer        Customer? @relation(fields: [customerId], references: [id], onDelete: SetNull)
   503→  taxConfig       TaxConfig @relation(fields: [taxConfigId], references: [id])
   504→
   505→  @@index([storeId, fechaEmision])
   506→  @@index([storeId, tipo, numero])
   507→  @@index([saleId])
   508→  @@index([cae])
   509→}
