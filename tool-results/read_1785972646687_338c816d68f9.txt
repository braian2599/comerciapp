   154→   154→  async function handleBarcodeLookup() {
   155→   155→    const code = (form.barcode || "").trim();
   156→   156→    if (!code) {
   157→   157→      toast.error("Ingresá un código de barras primero");
   158→   158→      return;
   159→   159→    }
   160→   160→    setLookupLoading(true);
   161→   161→    setLookupResult({ status: "loading" });
   162→   162→    try {
   163→   163→      const { ok, data, error } = await safeFetchJSON<any>(
   164→   164→        `/api/products/lookup?barcode=${encodeURIComponent(code)}`
   165→   165→      );
   166→   166→      if (!ok || !data) {
   167→   167→        throw new Error(error || "No se pudo consultar la base de productos");
   168→   168→      }
   169→   169→      if (data.found) {
   170→   170→        // Solo autocompletamos campos que el usuario aún no llenó.
   171→   171→        setForm((prev: any) => ({
   172→   172→          ...prev,
   173→   173→          name: prev.name || data.name || "",
   174→   174→          description: prev.description || data.description || data.brand || "",
   175→   175→          barcode: prev.barcode || code,
   176→   176→          brand: prev.brand || data.brand || "",
   177→   177→          ingredients: prev.ingredients || data.ingredients || "",
   178→   178→          imageUrl: prev.imageUrl || data.imageUrl || "",
   179→   179→        }));
   180→   180→        // Merge de etiquetas y alérgenos (sin duplicar)
   181→   181→        if (Array.isArray(data.labels) && data.labels.length > 0) {
   182→   182→          setLabelChips((prev) =>
   183→   183→            Array.from(new Set([...prev, ...data.labels])).slice(0, 15)
   184→   184→          );
   185→   185→          setShowExtraFields(true);
   186→   186→        }
   187→   187→        if (Array.isArray(data.allergens) && data.allergens.length > 0) {
   188→   188→          setAllergenChips((prev) =>
   189→   189→            Array.from(new Set([...prev, ...data.allergens])).slice(0, 15)
   190→   190→          );
   191→   191→          setShowExtraFields(true);
   192→   192→        }
   193→   193→        setLookupResult({
   194→   194→          status: "found",
   195→   195→          source: data.source,
   196→   196→          imageUrl: data.imageUrl,
   197→   197→          labels: data.labels,
   198→   198→          allergens: data.allergens,
   199→   199→        });
   200→   200→        const sourceLabel =
   201→   201→          data.source === "openfoodfacts" ? "Open Food Facts" : "UPC Item DB";
   202→   202→        toast.success(`Producto encontrado en ${sourceLabel}`, {
   203→   203→          description: data.name,
   204→   204→        });
   205→   205→      } else {
   206→   206→        setLookupResult({ status: "notfound" });
   207→   207→        toast.info("No se encontró el código en las bases de datos públicas", {
   208→   208→          description: "Completá los datos manualmente",
   209→   209→        });
   210→   210→      }
   211→   211→    } catch (e: any) {
   212→   212→      setLookupResult({ status: "error" });
   213→   213→      toast.error("No se pudo consultar la base de productos");
   214→   214→    } finally {
   215→   215→      setLookupLoading(false);
   216→   216→    }
   217→   217→  }
