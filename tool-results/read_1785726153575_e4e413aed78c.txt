     1→"use client";
     2→
     3→import { useEffect, useState } from "react";
     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";
     8→import { Badge } from "@/components/ui/badge";
     9→import {
    10→  Table,
    11→  TableBody,
    12→  TableCell,
    13→  TableHead,
    14→  TableHeader,
    15→  TableRow,
    16→} from "@/components/ui/table";
    17→import {
    18→  Select,
    19→  SelectContent,
    20→  SelectItem,
    21→  SelectTrigger,
    22→  SelectValue,
    23→} from "@/components/ui/select";
    24→import {
    25→  BarChart,
    26→  Bar,
    27→  LineChart,
    28→  Line,
    29→  PieChart,
    30→  Pie,
    31→  Cell,
    32→  XAxis,
    33→  YAxis,
    34→  CartesianGrid,

    39→import { toast } from "sonner";
    40→import {
    41→  BarChart3,
    42→  TrendingUp,

    45→  Receipt,

    53→import { useAppStore } from "@/store/app-store";

    55→
    56→type TabKey = "sales" | "profits" | "taxes" | "products" | "customers" | "cashflow";
    57→
    58→const COLORS = ["#10b981", "#3b82f6", "#f59e0b", "#ef4444", "#8b5cf6", "#ec4899", "#06b6d4", "#84cc16"];
    59→
    60→export function ReportsView() {
    61→  const { store } = useAppStore();
    62→  const symbol = store?.currencySymbol || "$";
    63→  const [tab, setTab] = useState<TabKey>("sales");
    64→  const [from, setFrom] = useState("");
    65→  const [to, setTo] = useState("");
    66→  const [loading, setLoading] = useState(false);
    67→  const [data, setData] = useState<any>(null);
    68→
    69→  // Defaults: últimos 30 días
    70→  useEffect(() => {
    71→    if (!from && !to) {
    72→      const end = new Date();
    73→      const start = new Date();
    74→      start.setDate(start.getDate() - 30);
    75→      setFrom(start.toISOString().slice(0, 10));
    76→      setTo(end.toISOString().slice(0, 10));
    77→    }
    78→  }, []);
    79→
    80→  async function loadReport() {
    81→    if (!from || !to) return;
    82→    setLoading(true);
    83→    setData(null); // reset para que no se renderice data de otro tab
    84→    try {
    85→      const params = new URLSearchParams({ from, to });
    86→      const res = await fetch(`/api/reports/${tab}?${params}`);
    87→      const json = await res.json();
    88→      if (!res.ok) {
    89→        toast.error(json.error || "Error cargando reporte");
    90→        return;
    91→      }
    92→      setData(json);
    93→    } catch (err) {
    94→      toast.error("Error de conexión");
    95→    } finally {
    96→      setLoading(false);
    97→    }
    98→  }
    99→
   100→  useEffect(() => {
   101→    if (from && to) loadReport();
   102→  }, [tab, from, to]);
   103→
   104→  function setPeriod(days: number) {
   105→    const end = new Date();
   106→    const start = new Date();
   107→    start.setDate(start.getDate() - days);
   108→    setFrom(start.toISOString().slice(0, 10));
   109→    setTo(end.toISOString().slice(0, 10));
   110→  }
   111→
   112→  function exportarCSV() {
   113→    if (!data) return;
   114→    let rows: string[][] = [];
   115→    let filename: string = tab;
   116→
   117→    if (tab === "sales" && data.series) {
   118→      rows = [["Período", "Cantidad", "Total", "Descuentos", "Recargos"]];
   119→      data.series.forEach((s: any) => {
   120→        rows.push([s.period, s.count, s.total, s.discount, s.surcharge]);
   121→      });
   122→      filename = "ventas";
   123→    } else if (tab === "profits" && data.series) {
   124→      rows = [["Fecha", "Ventas", "Costo", "Gastos", "Ganancia Bruta", "Ganancia Neta"]];
   125→      data.series.forEach((s: any) => {
   126→        rows.push([s.date, s.ventas, s.costo, s.gastos, s.gananciaBruta, s.gananciaNeta]);
   127→      });
   128→      filename = "ganancias";
   129→    } else if (tab === "products" && data.ranking) {
   130→      rows = [["Producto", "Cantidad", "Ingresos", "Costo", "Ganancia", "Margen %", "Stock Actual"]];
   131→      data.ranking.forEach((p: any) => {
   132→        rows.push([p.productName, p.quantitySold, p.revenue, p.cost, p.profit, p.margin, p.currentStock]);
   133→      });
   134→      filename = "productos";
   135→    } else if (tab === "customers" && data.topClientes) {
   136→      rows = [["Cliente", "Compras", "Cantidad ventas", "Fiado", "Pagado"]];
   137→      data.topClientes.forEach((c: any) => {
   138→        rows.push([c.customerName, c.totalCompras, c.cantidadVentas, c.totalFiado, c.totalPagado]);
   139→      });
   140→      filename = "clientes";
   141→    } else if (tab === "taxes" && data.byTipo) {
   142→      rows = [["Tipo", "Cantidad", "Neto", "IVA", "Total"]];
   143→      data.byTipo.forEach((t: any) => {
   144→        rows.push([t.tipo, t.count, t.netoGravado, t.iva, t.total]);
   145→      });
   146→      filename = "fiscal";
   147→    } else if (tab === "cashflow" && data.series) {
   148→      rows = [["Fecha", "Ventas Efectivo", "Cobros Cuenta", "Ing. Manuales", "Gastos", "Egresos Manuales", "Flujo Neto"]];
   149→      data.series.forEach((s: any) => {
   150→        rows.push([s.date, s.ventasEfectivo, s.cobrosCuenta, s.ingresosManuales, s.gastosEfectivo, s.egresosManuales, s.flujoNeto]);
   151→      });
   152→      filename = "flujo-caja";
   153→    }
   154→
   155→    const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n");
   156→    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
   157→    const url = URL.createObjectURL(blob);
   158→    const a = document.createElement("a");
   159→    a.href = url;
   160→    a.download = `reporte-${filename}-${new Date().toISOString().slice(0, 10)}.csv`;
   161→    a.click();
   162→    URL.revokeObjectURL(url);
   163→  }
   164→
   165→  const TABS: { key: TabKey; label: string; icon: any }[] = [
   166→    { key: "sales", label: "Ventas", icon: Receipt },
   167→    { key: "profits", label: "Ganancias", icon: TrendingUp },
   168→    { key: "taxes", label: "Fiscal", icon: BarChart3 },
   169→    { key: "products", label: "Productos", icon: Package },
   170→    { key: "customers", label: "Clientes", icon: Users },
   171→    { key: "cashflow", label: "Flujo Caja", icon: Wallet },
   172→  ];
   173→
   174→  return (
   175→    <div className="p-4 space-y-4">
   176→      <div className="flex flex-wrap items-center justify-between gap-3">
   177→        <div>
   178→          <h1 className="text-2xl font-bold flex items-center gap-2">
   179→            <BarChart3 className="w-6 h-6 text-emerald-600" />
   180→            Reportes
   181→          </h1>
   182→          <p className="text-sm text-muted-foreground">
   183→            Análisis de ventas, ganancias, fiscal, productos y clientes
   184→          </p>
   185→        </div>
   186→        <Button variant="outline" onClick={exportarCSV} disabled={!data}>
   187→          <Download className="w-4 h-4 mr-2" />
   188→          Exportar CSV
   189→        </Button>
   190→      </div>
   191→
   192→      {/* Tabs */}
   193→      <div className="flex flex-wrap gap-2">
   194→        {TABS.map((t) => {
   195→          const Icon = t.icon;
   196→          return (
   197→            <Button
   198→              key={t.key}
   199→              variant={tab === t.key ? "default" : "outline"}
   200→              size="sm"
   201→              onClick={() => setTab(t.key)}
   202→              className={tab === t.key ? "bg-emerald-600 hover:bg-emerald-700" : ""}
   203→            >
   204→              <Icon className="w-4 h-4 mr-1" />
   205→              {t.label}
   206→            </Button>
   207→          );
   208→        })}
   209→      </div>
   210→
   211→      {/* Filtros */}
   212→      <Card>
   213→        <CardContent className="pt-4 flex flex-wrap items-end gap-3">
   214→          <div>
   215→            <Label className="text-xs">Desde</Label>
   216→            <Input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className="w-[160px]" />
   217→          </div>
   218→          <div>
   219→            <Label className="text-xs">Hasta</Label>
   220→            <Input type="date" value={to} onChange={(e) => setTo(e.target.value)} className="w-[160px]" />
   221→          </div>
   222→          <div className="flex gap-1">
   223→            <Button size="sm" variant="outline" onClick={() => setPeriod(7)}>7 días</Button>
   224→            <Button size="sm" variant="outline" onClick={() => setPeriod(30)}>30 días</Button>
   225→            <Button size="sm" variant="outline" onClick={() => setPeriod(90)}>90 días</Button>
   226→            <Button size="sm" variant="outline" onClick={() => setPeriod(365)}>1 año</Button>
   227→          </div>
   228→        </CardContent>
   229→      </Card>
   230→
   231→      {/* Contenido */}
   232→      {loading ? (
   233→        <div className="flex items-center justify-center py-12">
   234→          <Loader2 className="w-8 h-8 animate-spin text-emerald-600" />
   235→        </div>
   236→      ) : !data ? (
   237→        <Card>
   238→          <CardContent className="text-center py-12 text-muted-foreground">
   239→            <Calendar className="w-12 h-12 mx-auto mb-2 opacity-30" />
   240→            <p>Selecciona un rango de fechas</p>
   241→          </CardContent>
   242→        </Card>
   243→      ) : (
   244→        <>
   245→          {/* VENTAS */}
   246→          {tab === "sales" && <SalesReport data={data} symbol={symbol} />}
   247→          {/* GANANCIAS */}
   248→          {tab === "profits" && <ProfitsReport data={data} symbol={symbol} />}
   249→          {/* FISCAL */}
   250→          {tab === "taxes" && <TaxesReport data={data} symbol={symbol} />}
   251→          {/* PRODUCTOS */}
   252→          {tab === "products" && <ProductsReport data={data} symbol={symbol} />}
   253→          {/* CLIENTES */}
   254→          {tab === "customers" && <CustomersReport data={data} symbol={symbol} />}
   255→          {/* FLUJO CAJA */}
   256→          {tab === "cashflow" && <CashFlowReport data={data} symbol={symbol} />}
   257→        </>
   258→      )}
   259→    </div>
   260→  );
   261→}

   263→// ===== REPORTE DE VENTAS =====
   264→function SalesReport({ data, symbol }: { data: any; symbol: string }) {
   265→  return (
   266→    <div className="space-y-4">
   267→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

   269→        <StatCard label="Cantidad" value={data.summary.totalSales.toString()} icon={Receipt} color="blue" />
   270→        <StatCard label="Ticket promedio" value={formatCurrency(data.summary.averageTicket, symbol)} icon={TrendingUp} color="purple" />
   271→        <StatCard label="Recargos" value={formatCurrency(data.summary.totalSurcharge, symbol)} icon={TrendingUp} color="amber" />

   304→                  data={data.byPaymentMethod}

   321→                <TableRow>

   323→                  <TableHead className="text-right">Cantidad</TableHead>
   324→                  <TableHead className="text-right">Total</TableHead>
   325→                </TableRow>
   326→              </TableHeader>
   327→              <TableBody>
   328→                {data.byPaymentMethod.map((m: any) => (
   329→                  <TableRow key={m.method}>
   330→                    <TableCell className="font-medium">{m.method}</TableCell>
   331→                    <TableCell className="text-right">{m.count}</TableCell>

   335→              </TableBody>
   336→            </Table>

   338→        </Card>
   339→
   340→        <Card>
   341→          <CardHeader>
   342→            <CardTitle className="text-sm">Por vendedor</CardTitle>
   343→          </CardHeader>
   344→          <CardContent>
   345→            <ResponsiveContainer width="100%" height={250}>
   346→              <BarChart data={data.byUser}>
   347→                <CartesianGrid strokeDasharray="3 3" />
   348→                <XAxis dataKey="userName" fontSize={11} />
   349→                <YAxis fontSize={11} />
   350→                <Tooltip formatter={(v: any) => formatCurrency(v, symbol)} />

   353→            </ResponsiveContainer>
   354→            <Table>
   355→              <TableHeader>

   358→                  <TableHead className="text-right">Cantidad</TableHead>
   359→                  <TableHead className="text-right">Total</TableHead>
   360→                </TableRow>
   361→              </TableHeader>

   369→                ))}
   370→              </TableBody>
   371→            </Table>
   372→          </CardContent>
   373→        </Card>
   374→      </div>
   375→    </div>
   376→  );
   377→}

   439→
   440→      <Card>
   441→        <CardContent className="pt-4 grid grid-cols-2 md:grid-cols-4 gap-3 text-sm">
   442→          <div>
   443→            <p className="text-xs text-muted-foreground">Total ventas</p>
   444→            <p className="font-bold">{formatCurrency(data.summary.totalVentas, symbol)}</p>
   445→          </div>
   446→          <div>
   447→            <p className="text-xs text-muted-foreground">Total costo</p>
   448→            <p className="font-bold text-red-600">{formatCurrency(data.summary.totalCosto, symbol)}</p>
   449→          </div>
   450→          <div>
   451→            <p className="text-xs text-muted-foreground">Total descuentos</p>
   452→            <p className="font-bold text-amber-600">{formatCurrency(data.summary.totalDescuentos, symbol)}</p>
   453→          </div>
   454→          <div>
   455→            <p className="text-xs text-muted-foreground">Margen bruto</p>
   456→            <p className="font-bold text-emerald-600">{data.summary.margenBruto}%</p>
   457→          </div>
   458→        </CardContent>
   459→      </Card>
   460→    </div>
   461→  );
   462→}

   472→        <StatCard label="Ventas sin facturar" value={data.summary.ventasSinFactura.toString()} icon={TrendingDown} color="amber" />
   473→      </div>
   474→
   475→      <Card>
   476→        <CardHeader>
   477→          <CardTitle className="text-sm">Facturación por tipo de comprobante</CardTitle>
   478→        </CardHeader>
   479→        <CardContent>
   480→          <Table>
   481→            <TableHeader>

   488→                <TableHead className="text-right">Exento</TableHead>
   489→                <TableHead className="text-right">Total</TableHead>
   490→              </TableRow>
   491→            </TableHeader>
   492→            <TableBody>
   493→              {data.byTipo.map((t: any) => (
   494→                <TableRow key={t.tipo}>
   495→                  <TableCell>
   496→                    <Badge variant="outline">Factura {t.tipo}</Badge>
   497→                  </TableCell>
   498→                  <TableCell className="text-right">{t.count}</TableCell>
   499→                  <TableCell className="text-right">{formatCurrency(t.netoGravado, symbol)}</TableCell>
   500→                  <TableCell className="text-right">{formatCurrency(t.iva, symbol)}</TableCell>
