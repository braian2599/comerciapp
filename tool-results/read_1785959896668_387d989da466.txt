   359→   359→  async function printThermalSale() {
   360→   360→    if (!lastSale?.id) return;
   361→   361→    try {
   362→   362→      const printRes = await safeFetchJSON<any>("/api/print", {
   363→   363→        method: "POST",
   364→   364→        body: JSON.stringify({ type: "TICKET", saleId: lastSale.id }),
   365→   365→      });
   366→   366→      if (!printRes.ok) throw new Error(printRes.error);
   367→   367→
   368→   368→      // Descargar binario .bin que se puede enviar a impresora USB
   369→   369→      const blobRes = await fetch("/api/print", {
   370→   370→        method: "POST",
   371→   371→        headers: { "Content-Type": "application/json" },
   372→   372→        body: JSON.stringify({
   373→   373→          type: "TICKET",
   374→   374→          saleId: lastSale.id,
   375→   375→          returnFormat: "blob",
   376→   376→        }),
   377→   377→      });
   378→   378→      if (!blobRes.ok) {
   379→   379→        throw new Error("No se pudo generar el archivo del ticket");
   380→   380→      }
   381→   381→      const blob = await blobRes.blob();
   382→   382→      const url = URL.createObjectURL(blob);
   383→   383→      const a = document.createElement("a");
   384→   384→      a.href = url;
   385→   385→      a.download = `ticket-${lastSale.id.slice(-6)}.bin`;
   386→   386→      document.body.appendChild(a);
   387→   387→      a.click();
   388→   388→      document.body.removeChild(a);
   389→   389→      URL.revokeObjectURL(url);
   390→   390→
   391→   391→      toast.success("Ticket generado (archivo .bin)");
   392→   392→    } catch (e: any) {
   393→   393→      toast.error("Error al imprimir", { description: e.message });
   394→   394→    }
   395→   395→  }

   397→   397→  async function processSale() {
   398→   398→    if (cart.length === 0) {
   399→   399→      toast.error("El carrito está vacío");
   400→   400→      return;
   401→   401→    }
   402→   402→    if (!paymentMethodId) {
   403→   403→      toast.error("Seleccioná un método de pago");
   404→   404→      return;
   405→   405→    }
   406→   406→    if (selectedMethod?.type === "CUENTA" && !customerId) {
   407→   407→      toast.error("Para cuenta corriente tenés que seleccionar un cliente");
   408→   408→      return;
   409→   409→    }
   410→   410→    setProcessing(true);
   411→   411→    try {
   412→   412→      const saleRes = await safeFetchJSON<any>("/api/sales", {
   413→   413→        method: "POST",
   414→   414→        body: JSON.stringify({
   415→   415→          items: cart.map((i) => ({
   416→   416→            productId: i.product.id,
   417→   417→            quantity: i.qty,
   418→   418→          })),
   419→   419→          customerId: customerId || null,
   420→   420→          discount: manualDiscount,
   421→   421→          paymentMethodId,
   422→   422→          notes,
   423→   423→          taxRate: taxEnabled ? taxRate : 0,
   424→   424→          branchId: branchId || null,
   425→   425→          promotionId: appliedPromotion?.promotionId || null,
   426→   426→          promotionDiscount: appliedPromotion?.discountAmount || 0,
   427→   427→          loyaltyPointsUsed: effectivePointsToRedeem || 0,
   428→   428→        }),
   429→   429→      });
   430→   430→      if (!saleRes.ok || !saleRes.data) {
   431→   431→        throw new Error(saleRes.error || "No se pudo registrar la venta");
   432→   432→      }
   433→   433→      const data = saleRes.data;
   434→   434→      toast.success("Venta registrada!");
   435→   435→      setLastSale({
   436→   436→        ...data,
   437→   437→        items: cart,
   438→   438→        customer: customers.find((c) => c.id === customerId),
   439→   439→        paymentMethod: selectedMethod,
   440→   440→        discount: totalDiscount,
   441→   441→        tax: taxAmount,
   442→   442→        surcharge: surchargeAmount,
   443→   443→        total,
   444→   444→        subtotal,
   445→   445→        appliedPromotion,
   446→   446→        pointsUsed: effectivePointsToRedeem,
   447→   447→        pointsEarned: data.loyaltyPointsEarned || 0,
   448→   448→      });
   449→   449→      setCheckoutOpen(false);
   450→   450→      setReceiptOpen(true);
   451→   451→      clearCart();
   452→   452→      setAppliedPromotion(null);
   453→   453→      setPointsToRedeem(0);
   454→   454→      // Refrescar stock y clientes en paralelo, sin romper si fallan
   455→   455→      // (la venta ya se registró, no queremos que el cajero vea un error
   456→   456→      // acá y crea que la venta no se hizo).
   457→   457→      try {
   458→   458→        const [refreshedRes, refreshedCustRes] = await Promise.all([
   459→   459→          safeFetchJSON<Product[]>("/api/products"),
   460→   460→          safeFetchJSON<Customer[]>("/api/customers"),
   461→   461→        ]);
   462→   462→        const refreshed = Array.isArray(refreshedRes.data) ? refreshedRes.data : [];
   463→   463→        setProducts(refreshed.filter((x: Product) => x.active));
   464→   464→        if (Array.isArray(refreshedCustRes.data)) {
   465→   465→          setCustomers(refreshedCustRes.data);
   466→   466→        }
   467→   467→      } catch {
   468→   468→        // Silencioso: el refresh es best-effort.
   469→   469→      }
   470→   470→    } catch (e: any) {
   471→   471→      toast.error(e.message);
   472→   472→    } finally {
   473→   473→      setProcessing(false);
   474→   474→    }
   475→   475→  }
   476→   476→
