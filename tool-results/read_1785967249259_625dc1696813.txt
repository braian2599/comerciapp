     1→"use client";
     2→
     3→/**
     4→ * Dialog de importación masiva reutilizable con mapeo manual de columnas.
     5→ *
     6→ * Flujo (wizard de 4 pasos):
     7→ *   1. SUBIR: usuario elige un archivo CSV/XLSX/JSON.
     8→ *   2. MAPEAR: el sistema detecta automáticamente las columnas y sugiere
     9→ *      un mapeo. El usuario puede cambiar qué columna va a cada campo.
    10→ *      Una columna del archivo puede no estar mapeada; un campo puede
    11→ *      quedar sin mapear (salvo los obligatorios).
    12→ *   3. PREVIEW: el servidor procesa con el mapeo elegido y muestra la
    13→ *      lista de items a crear / actualizar / con error. El usuario
    14→ *      selecciona qué items confirmar.
    15→ *   4. RESULTADO: estadísticas finales (creados, actualizados, errores).
    16→ *
    17→ * Props:
    18→ *   - endpoint:       URL del API route ("/api/products/import" | "/api/customers/import")
    19→ *   - fields:         lista de ImportField (de @/lib/import-config) que definen
    20→ *                     qué campos puede mapear el usuario.
    21→ *   - templateHeaders: columnas sugeridas para la plantilla descargable
    22→ *   - templateName:    nombre del archivo de plantilla
    23→ *   - entityLabel:     "producto" | "cliente"
    24→ *   - entityLabelPlural: "productos" | "clientes"
    25→ *   - onImported:      callback al terminar (para que el view recargue la lista)
    26→ */
    27→import { useState, useRef, useCallback, useMemo } from "react";
    28→import {
    29→  Dialog,
    30→  DialogContent,
    31→  DialogDescription,
    32→  DialogFooter,
    33→  DialogHeader,
    34→  DialogTitle,
    35→} from "@/components/ui/dialog";
    36→import { Button } from "@/components/ui/button";

    39→import {
    40→  Table,
    41→  TableBody,
    42→  TableCell,
    43→  TableHead,
    44→  TableHeader,
    45→  TableRow,
    46→} from "@/components/ui/table";

    48→  Select,
    49→  SelectContent,
    50→  SelectItem,
    51→  SelectTrigger,
    52→  SelectValue,
    53→} from "@/components/ui/select";
    54→import { Checkbox } from "@/components/ui/checkbox";
    55→import { Alert, AlertDescription } from "@/components/ui/alert";
    56→import { ScrollArea } from "@/components/ui/scroll-area";

    71→  Eye,

    77→  type ParsedFile,

    80→  suggestColumnMapping,
    81→  normalizeHeader,

    83→} from "@/lib/import-config";
    84→
    85→export interface ImportPreviewItem {
    86→  action: "create" | "update" | "error";
    87→  rowIndex: number;
    88→  name: string;
    89→  existingId?: string;
    90→  existingName?: string;
    91→  matchBy?: string;
    92→  error?: string;
    93→  data?: any;
    94→}
    95→
    96→export interface ImportPreviewResponse {
    97→  items: ImportPreviewItem[];
    98→  summary: {
    99→    create: number;
   100→    update: number;
   101→    error: number;
   102→  };
   103→  columnMapping?: Record<string, number>;
   104→}
   105→
   106→export interface ImportCommitResponse {
   107→  created: number;
   108→  updated: number;
   109→  errors: { rowIndex: number; name: string; error: string }[];
   110→  total: number;
   111→}
   112→
   113→interface ImportDialogProps {
   114→  open: boolean;
   115→  onOpenChange: (open: boolean) => void;
   116→  endpoint: string;
   117→  fields: ImportField[];
   118→  templateHeaders: string[];
   119→  templateName: string;
   120→  entityLabel: string;
   121→  entityLabelPlural: string;
   122→  onImported?: () => void;
   123→}
   124→
   125→type Phase = "idle" | "parsing" | "mapping" | "previewing" | "committing" | "done";
   126→
   127→export function ImportDialog({
   128→  open,
   129→  onOpenChange,
   130→  endpoint,
   131→  fields,
   132→  templateHeaders,
   133→  templateName,
   134→  entityLabel,
   135→  entityLabelPlural,
   136→  onImported,
   137→}: ImportDialogProps) {
   138→  const [phase, setPhase] = useState<Phase>("idle");
   139→  const [fileName, setFileName] = useState("");
   140→  const [parsed, setParsed] = useState<ParsedFile | null>(null);
   141→  // columnMapping: { fieldKey: columnIndex }
   142→  // columnIndex = -1 significa "no mapeado" (lo usamos internamente para
   143→  // distinguir de un campo que todavía no fue tocado por el usuario).
   144→  const [columnMapping, setColumnMapping] = useState<Record<string, number>>({});
   145→  const [preview, setPreview] = useState<ImportPreviewResponse | null>(null);
   146→  const [selectedRowIndices, setSelectedRowIndices] = useState<Set<number>>(
   147→    new Set()
   148→  );
   149→  const [commitResult, setCommitResult] = useState<ImportCommitResponse | null>(
   150→    null
   151→  );
   152→  const [errorMsg, setErrorMsg] = useState("");
   153→  const fileInputRef = useRef<HTMLInputElement>(null);
   154→
   155→  const reset = useCallback(() => {
   156→    setPhase("idle");
   157→    setFileName("");
   158→    setParsed(null);
   159→    setColumnMapping({});
   160→    setPreview(null);
   161→    setSelectedRowIndices(new Set());
   162→    setCommitResult(null);
   163→    setErrorMsg("");
   164→    if (fileInputRef.current) fileInputRef.current.value = "";
   165→  }, []);
   166→
   167→  function handleOpenChange(next: boolean) {
   168→    if (!next) reset();
   169→    onOpenChange(next);
   170→  }
   171→
   172→  function handleTemplate() {
   173→    const csv = generateCSV(templateHeaders);
   174→    downloadTextFile(templateName, csv);
   175→    toast.success("Plantilla descargada", {
   176→      description: `Abrí ${templateName} en Excel o Google Sheets`,
   177→    });
   178→  }
   179→
   180→  // ─── Paso 1 → 2: parsear archivo y saltar a mapeo ───────────────────────────
   181→  async function handleFile(file: File) {
   182→    setPhase("parsing");
   183→    setErrorMsg("");
   184→    setFileName(file.name);
   185→
   186→    let result: ParsedFile;
   187→    try {
   188→      result = await parseFile(file);
   189→    } catch (e: any) {
   190→      setErrorMsg(e?.message || "No se pudo leer el archivo");
   191→      setPhase("idle");
   192→      return;
   193→    }
   194→
   195→    if (!result.headers.length || !result.rows.length) {
   196→      setErrorMsg("El archivo está vacío o no tiene filas de datos");
   197→      setPhase("idle");
   198→      return;
   199→    }
   200→
   201→    setParsed(result);
   202→    // Sugerir mapeo automático
   203→    const suggested = suggestColumnMapping(result.headers, fields);
   204→    setColumnMapping(suggested);
   205→    setPhase("mapping");
   206→  }
   207→
   208→  // ─── Helpers de mapeo ────────────────────────────────────────────────────────
   209→  function setFieldMapping(fieldKey: string, columnIndex: number) {
   210→    setColumnMapping((prev) => {
   211→      const next = { ...prev };
   212→      if (columnIndex < 0) {
   213→        delete next[fieldKey];
   214→      } else {
   215→        next[fieldKey] = columnIndex;
   216→      }
   217→      return next;
   218→    });
   219→  }
   220→
   221→  function autoDetectAll() {
   222→    if (!parsed) return;
   223→    const suggested = suggestColumnMapping(parsed.headers, fields);
   224→    setColumnMapping(suggested);
   225→    toast.success("Mapeo automático aplicado", {
   226→      description: "Revisá que las columnas estén bien asignadas",
   227→    });
   228→  }
   229→
   230→  function clearAllMappings() {
   231→    setColumnMapping({});
   232→  }
   233→
   234→  // Stats del mapeo para mostrar en la UI
   235→  const mappingStats = useMemo(() => {
   236→    const mapped = Object.keys(columnMapping).length;
   237→    const required = fields.filter((f) => f.required);
   238→    const requiredMapped = required.filter((r) => r.key in columnMapping).length;
   239→    const totalRequired = required.length;
   240→    return { mapped, requiredMapped, totalRequired };
   241→  }, [columnMapping, fields]);
   242→
   243→  const requiredSatisfied = mappingStats.requiredMapped === mappingStats.totalRequired;
   244→
   245→  // ─── Paso 2 → 3: enviar mapeo al servidor para preview ──────────────────────
   246→  async function handlePreview() {
   247→    if (!parsed) return;
   248→    if (!requiredSatisfied) {
   249→      toast.error("Faltan campos obligatorios por mapear");
   250→      return;
   251→    }
   252→
   253→    setPhase("previewing");
   254→    setErrorMsg("");
   255→    try {
   256→      const res = await fetch(endpoint, {
   257→        method: "POST",
   258→        headers: { "Content-Type": "application/json" },
   259→        body: JSON.stringify({
   260→          mode: "preview",
   261→          headers: parsed.headers,
   262→          rows: parsed.rows,
   263→          columnMapping,
   264→        }),
   265→      });
   266→      const data = await res.json();
   267→      if (!res.ok) throw new Error(data.error || "Error en el servidor");
   268→
   269→      const previewData = data as ImportPreviewResponse;
   270→      setPreview(previewData);
   271→
   272→      // Por defecto: seleccionar todos los create y update, excluir errors
   273→      const selected = new Set<number>();
   274→      previewData.items.forEach((item) => {
   275→        if (item.action === "create" || item.action === "update") {
   276→          selected.add(item.rowIndex);
   277→        }
   278→      });
   279→      setSelectedRowIndices(selected);
   280→      setPhase("preview");
   281→    } catch (e: any) {
   282→      setErrorMsg(e?.message || "Error al analizar el archivo");
   283→      setPhase("mapping");
   284→    }
   285→  }
   286→
   287→  // ─── Selección de items en el preview ───────────────────────────────────────
   288→  function toggleRow(rowIndex: number) {
   289→    setSelectedRowIndices((prev) => {
   290→      const next = new Set(prev);
   291→      if (next.has(rowIndex)) next.delete(rowIndex);
   292→      else next.add(rowIndex);
   293→      return next;
   294→    });
   295→  }
   296→  function selectAll() {
   297→    if (!preview) return;
   298→    const next = new Set<number>();
   299→    preview.items.forEach((item) => {
   300→      if (item.action === "create" || item.action === "update") {
   301→        next.add(item.rowIndex);
   302→      }
   303→    });
   304→    setSelectedRowIndices(next);
   305→  }
   306→  function selectNone() {
   307→    setSelectedRowIndices(new Set());
   308→  }
   309→  function selectOnlyCreates() {
   310→    if (!preview) return;
   311→    const next = new Set<number>();
   312→    preview.items.forEach((item) => {
   313→      if (item.action === "create") next.add(item.rowIndex);
   314→    });
   315→    setSelectedRowIndices(next);
   316→  }
   317→
   318→  // ─── Paso 3 → 4: confirmar y commitear ──────────────────────────────────────
   319→  async function handleCommit() {
   320→    if (!preview) return;
   321→    const itemsToCommit = preview.items.filter((i) =>
   322→      selectedRowIndices.has(i.rowIndex)
   323→    );
   324→    if (itemsToCommit.length === 0) {
   325→      toast.error("No seleccionaste ningún item para importar");
   326→      return;
   327→    }
   328→
   329→    setPhase("committing");
   330→    try {
   331→      const res = await fetch(endpoint, {
   332→        method: "POST",
   333→        headers: { "Content-Type": "application/json" },
   334→        body: JSON.stringify({
   335→          mode: "commit",
   336→          items: itemsToCommit,
   337→        }),
   338→      });
   339→      const data = await res.json();
   340→      if (!res.ok) throw new Error(data.error || "Error en el servidor");
   341→
   342→      const result = data as ImportCommitResponse;
   343→      setCommitResult(result);
   344→      setPhase("done");
   345→      onImported?.();
   346→
   347→      const parts: string[] = [];
   348→      if (result.created) parts.push(`${result.created} creados`);
   349→      if (result.updated) parts.push(`${result.updated} actualizados`);
   350→      if (result.errors.length) parts.push(`${result.errors.length} con error`);
   351→      toast.success(`Importación completa`, {
   352→        description: parts.join(" · ") || "Sin cambios",
   353→      });
   354→    } catch (e: any) {
   355→      setErrorMsg(e?.message || "Error al importar");
   356→      setPhase("preview");
   357→    }
   358→  }
   359→
   360→  // ─── Helpers de render del mapeo ─────────────────────────────────────────────
   361→  function columnLabel(idx: number): string {
   362→    if (!parsed) return "";
   363→    return parsed.headers[idx] || `(columna ${idx + 1})`;
   364→  }
   365→  function sampleValues(idx: number): string[] {
   366→    if (!parsed) return [];
   367→    return parsed.rows.slice(0, 3).map((r) => {
   368→      const v = r[idx];
   369→      if (v === null || v === undefined || v === "") return "—";
   370→      return String(v).slice(0, 20);
   371→    });
   372→  }
   373→
   374→  // ─── Render ──────────────────────────────────────────────────────────────────
   375→  return (
   376→    <Dialog open={open} onOpenChange={handleOpenChange}>
   377→      <DialogContent className="max-w-5xl w-[95vw] max-h-[92vh] overflow-hidden flex flex-col">
   378→        <DialogHeader>
   379→          <DialogTitle className="flex items-center gap-2">
   380→            <Upload className="w-5 h-5 text-indigo-600" />
   381→            Importar {entityLabelPlural}
   382→          </DialogTitle>
   383→          <DialogDescription>
   384→            {phase === "idle" && "Subí un archivo CSV, Excel o JSON para comenzar."}
   385→            {phase === "mapping" &&
   386→              "Asigná cada columna del archivo a un campo del sistema. Podés dejar columnas sin mapear."}
   387→            {phase === "preview" &&
   388→              "Revisá el resultado del mapeo y elegí qué items importar."}
   389→            {phase === "done" && "Resultado de la importación."}
   390→          </DialogDescription>
   391→        </DialogHeader>
   392→
   393→        {/* Indicador de pasos */}
   394→        {(phase === "mapping" || phase === "preview" || phase === "previewing") && (
   395→          <div className="flex items-center gap-1 text-xs shrink-0">
   396→            <StepBadge n={1} active={phase === "mapping"} done={phase !== "mapping"}>
   397→              Mapear
   398→            </StepBadge>
   399→            <ArrowRight className="w-3 h-3 text-muted-foreground" />
   400→            <StepBadge n={2} active={phase === "preview"} done={false}>
   401→              Revisar
   402→            </StepBadge>
   403→            <ArrowRight className="w-3 h-3 text-muted-foreground" />
   404→            <StepBadge n={3} active={false} done={false}>
   405→              Confirmar
   406→            </StepBadge>
   407→          </div>
   408→        )}
   409→
   410→        {/* ════════════════════════════════════════════════════════════════════════
   411→                PASO 1: SUBIR ARCHIVO                                                   */}
   412→        {phase === "idle" && (
   413→          <div className="space-y-4 py-2">
   414→            <div className="flex flex-wrap gap-2">
   415→              <Button
   416→                variant="outline"
   417→                size="sm"
   418→                onClick={handleTemplate}
   419→                className="border-dashed"
   420→              >
   421→                <Download className="w-4 h-4 mr-2" />
   422→                Descargar plantilla CSV
   423→              </Button>
   424→            </div>
   425→
   426→            <Alert>
   427→              <FileSpreadsheet className="h-4 w-4" />
   428→              <AlertDescription>
   429→                <div className="space-y-1">
   430→                  <p className="font-medium">Cómo funciona</p>
   431→                  <p className="text-xs text-muted-foreground">
   432→                    Subí tu archivo. En el próximo paso vas a poder mapear las
   433→                    columnas de tu planilla a los campos del sistema, sin
   434→                    importar cómo se llamen. Aceptamos <code>.csv</code>,{" "}
   435→                    <code>.xlsx</code> y <code>.json</code>.
   436→                  </p>
   437→                </div>
   438→              </AlertDescription>
   439→            </Alert>
   440→
   441→            <div
   442→              className="border-2 border-dashed rounded-lg p-8 text-center cursor-pointer hover:bg-muted/50 transition-colors"
   443→              onClick={() => fileInputRef.current?.click()}
   444→              onDragOver={(e) => e.preventDefault()}
   445→              onDrop={(e) => {
   446→                e.preventDefault();
   447→                const file = e.dataTransfer.files?.[0];
   448→                if (file) handleFile(file);
   449→              }}
   450→            >
   451→              <Upload className="w-10 h-10 mx-auto text-muted-foreground mb-2" />
   452→              <p className="text-sm font-medium">
   453→                Hacé click o arrastrá un archivo aquí
   454→              </p>
   455→              <p className="text-xs text-muted-foreground mt-1">
   456→                CSV, XLSX o JSON · máx ~5MB recomendado
   457→              </p>
   458→              <Input
   459→                ref={fileInputRef}
   460→                type="file"
   461→                accept=".csv,.xlsx,.xls,.json"
   462→                className="hidden"
   463→                onChange={(e) => {
   464→                  const file = e.target.files?.[0];
   465→                  if (file) handleFile(file);
   466→                }}
   467→              />
   468→            </div>
   469→
   470→            {errorMsg && (
   471→              <Alert variant="destructive">
   472→                <XCircle className="h-4 w-4" />
   473→                <AlertDescription>{errorMsg}</AlertDescription>
   474→              </Alert>
   475→            )}
   476→          </div>
   477→        )}
   478→
   479→        {/* ════════════════════════════════════════════════════════════════════════
   480→                PASO 2: MAPEAR COLUMNAS                                                 */}
   481→        {(phase === "mapping" || phase === "previewing") && parsed && (
   482→          <div className="space-y-3 flex-1 overflow-hidden flex flex-col">
   483→            <div className="flex items-center justify-between gap-2 flex-wrap shrink-0">
   484→              <div className="flex items-center gap-2 text-xs">
   485→                <Badge variant="outline" className="bg-muted/50">
   486→                  <Columns3 className="w-3 h-3 mr-1" />
   487→                  {parsed.headers.length} columnas en el archivo
   488→                </Badge>
   489→                <Badge variant="outline" className="bg-indigo-50 text-indigo-700 border-indigo-200">
   490→                  {mappingStats.mapped} mapeadas
   491→                </Badge>
   492→                <Badge
   493→                  variant="outline"
   494→                  className={
   495→                    requiredSatisfied
   496→                      ? "bg-emerald-50 text-emerald-700 border-emerald-200"
   497→                      : "bg-amber-50 text-amber-700 border-amber-200"
   498→                  }
   499→                >
   500→                  {mappingStats.requiredMapped}/{mappingStats.totalRequired} obligatorias
   501→                </Badge>
   502→              </div>
   503→              <div className="flex items-center gap-1">
   504→                <Button
   505→                  variant="outline"
   506→                  size="sm"
   507→                  onClick={autoDetectAll}
   508→                  className="h-7 text-xs"
   509→                >
   510→                  <Sparkles className="w-3 h-3 mr-1" />
   511→                  Auto-detectar
   512→                </Button>
   513→                <Button
   514→                  variant="ghost"
   515→                  size="sm"
   516→                  onClick={clearAllMappings}
   517→                  className="h-7 text-xs"
   518→                >
   519→                  Limpiar
   520→                </Button>
   521→              </div>
   522→            </div>
   523→
   524→            {/* Alert si faltan obligatorios */}
   525→            {!requiredSatisfied && (
   526→              <Alert variant="destructive" className="shrink-0">
   527→                <AlertTriangle className="h-4 w-4" />
   528→                <AlertDescription className="text-xs">
   529→                  Faltan mapear campos obligatorios. No podés continuar hasta
   530→                  asignarlos.
   531→                </AlertDescription>
   532→              </Alert>
   533→            )}
   534→
   535→            {/* Lista de campos a mapear */}
   536→            <ScrollArea className="flex-1 border rounded-md">
   537→              <div className="p-3 space-y-2">
   538→                {fields.map((field) => {
   539→                  const idx = columnMapping[field.key];
   540→                  const isMapped = idx !== undefined && idx >= 0;
   541→                  const samples = isMapped ? sampleValues(idx) : [];
   542→                  const usedColumnIndices = Object.values(columnMapping).filter(
   543→                    (i) => i !== idx
   544→                  );
   545→
   546→                  return (
   547→                    <div
   548→                      key={field.key}
   549→                      className={`rounded-md border p-3 transition-colors ${
   550→                        isMapped
   551→                          ? "border-indigo-200 bg-indigo-50/30"
   552→                          : field.required
   553→                          ? "border-amber-200 bg-amber-50/30"
   554→                          : "border-border bg-background"
   555→                      }`}
   556→                    >
   557→                      {/* Fila 1: label + tipo + estado (sin truncate, permite wrap) */}
   558→                      <div className="flex items-center gap-2 flex-wrap mb-2">
   559→                        <div className="flex items-center gap-1.5 flex-1 min-w-0">
   560→                          {field.required && (
   561→                            <span className="text-amber-600 text-sm font-bold">*</span>
   562→                          )}
   563→                          <Label className="text-sm font-medium">
   564→                            {field.label}
   565→                          </Label>
   566→                          {field.type !== "text" && (
   567→                            <Badge variant="outline" className="text-[9px] py-0 px-1 h-4">
   568→                              {field.type}
   569→                            </Badge>
   570→                          )}
   571→                        </div>
   572→                        {isMapped ? (
   573→                          <Badge
   574→                            variant="outline"
   575→                            className="bg-emerald-50 text-emerald-700 border-emerald-200 text-xs shrink-0"
   576→                          >
   577→                            <CheckCircle2 className="w-3 h-3 mr-0.5" />
   578→                            {columnLabel(idx)}
   579→                          </Badge>
   580→                        ) : (
   581→                          <span className="text-[11px] text-muted-foreground italic shrink-0">
   582→                            sin mapear
   583→                          </span>
   584→                        )}
   585→                      </div>
   586→                      {/* Hint (siempre visible, no truncado) */}
   587→                      {field.hint && (

   590→                        </p>

   592→                      {/* Fila 2: Select ancho */}
   593→                      <Select
   594→                        value={isMapped ? String(idx) : "none"}
   595→                        onValueChange={(v) =>
   596→                          setFieldMapping(
   597→                            field.key,

   600→                        }
   601→                      >
   602→                        <SelectTrigger className="h-9 w-full text-sm">
   603→                          <SelectValue placeholder="Elegir columna del archivo…" />

   605→                        <SelectContent>
   606→                          <SelectItem value="none">
   607→                            <span className="text-muted-foreground italic">
   608→                              — No mapear —
   609→                            </span>
   610→                          </SelectItem>
   611→                          {parsed.headers.map((h, i) => (
   612→                            <SelectItem
   613→                              key={i}
   614→                              value={String(i)}
   615→                              disabled={usedColumnIndices.includes(i)}
   616→                            >
   617→                              <span className={usedColumnIndices.includes(i) ? "opacity-50" : ""}>
   618→                                {h || `(columna ${i + 1})`}

   620→                            </SelectItem>
   621→                          ))}
   622→                        </SelectContent>
   623→                      </Select>
   624→                      {/* Sample values preview */}
   625→                      {isMapped && samples.length > 0 && (
   626→                        <div className="mt-2 flex items-center gap-2 text-[10px] text-muted-foreground flex-wrap">
   627→                          <span className="shrink-0">Ejemplos:</span>

   629→                            <code
   630→                              key={i}
   631→                              className="px-1.5 py-0.5 bg-muted rounded truncate max-w-[200px]"
   632→                            >
   633→                              {s}
   634→                            </code>
   635→                          ))}
   636→                        </div>
   637→                      )}
   638→                    </div>
   639→                  );
   640→                })}
   641→              </div>
   642→            </ScrollArea>
   643→
   644→            {/* Preview de primeras filas con el mapeo actual */}
   645→            <div className="shrink-0 border rounded-md overflow-hidden">
   646→              <div className="bg-muted/50 px-2 py-1 text-[11px] font-medium text-muted-foreground flex items-center gap-1">
   647→                <Eye className="w-3 h-3" />
   648→                Vista previa (primeras 3 filas)
   649→              </div>
   650→              <ScrollArea className="h-24">
   651→                <Table>
   652→                  <TableHeader>
   653→                    <TableRow>
   654→                      <TableHead className="h-7 text-[10px]">#</TableHead>
   655→                      {fields
   656→                        .filter((f) => columnMapping[f.key] !== undefined)
   657→                        .map((f) => (
   658→                          <TableHead key={f.key} className="h-7 text-[10px]">
   659→                            {f.label}
   660→                          </TableHead>
   661→                        ))}
   662→                    </TableRow>
   663→                  </TableHeader>
   664→                  <TableBody>
   665→                    {parsed.rows.slice(0, 3).map((row, rowIdx) => (

   667→                        <TableCell className="text-[10px] py-1 text-muted-foreground">

   669→                        </TableCell>

   671→                          .filter((f) => columnMapping[f.key] !== undefined)

   682→                      </TableRow>
   683→                    ))}
   684→                  </TableBody>
   685→                </Table>
   686→              </ScrollArea>
   687→            </div>
   688→
   689→            {errorMsg && (
   690→              <Alert variant="destructive" className="shrink-0">
   691→                <XCircle className="h-4 w-4" />
   692→                <AlertDescription>{errorMsg}</AlertDescription>
   693→              </Alert>
   694→            )}
   695→          </div>
   696→        )}
   697→
   698→        {/* ════════════════════════════════════════════════════════════════════════
   699→                PASO 2.5: loading preview                                               */}
   700→        {phase === "previewing" && (
   701→          <div className="py-12 text-center">
   702→            <Loader2 className="w-8 h-8 animate-spin mx-auto text-indigo-600 mb-3" />
   703→            <p className="text-sm text-muted-foreground">
   704→              Analizando {parsed?.rows.length || 0} filas con el mapeo elegido…
   705→            </p>
   706→          </div>
   707→        )}
   708→
   709→        {/* ════════════════════════════════════════════════════════════════════════
   710→                PASO 3: PREVIEW DE ITEMS                                                 */}
   711→        {phase === "preview" && preview && (
   712→          <div className="space-y-3 flex-1 overflow-hidden flex flex-col">
   713→            <div className="flex items-center justify-between gap-2 flex-wrap shrink-0">
   714→              <div className="flex items-center gap-2 flex-wrap">
   715→                <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100">
   716→                  <CheckCircle2 className="w-3 h-3 mr-1" />
   717→                  {preview.summary.create} a crear
   718→                </Badge>
   719→                <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100">
   720→                  <AlertTriangle className="w-3 h-3 mr-1" />
   721→                  {preview.summary.update} a actualizar
   722→                </Badge>
   723→                {preview.summary.error > 0 && (
   724→                  <Badge className="bg-red-100 text-red-700 hover:bg-red-100">
   725→                    <XCircle className="w-3 h-3 mr-1" />
   726→                    {preview.summary.error} con error
   727→                  </Badge>
   728→                )}
   729→              </div>
   730→              <div className="flex items-center gap-1 text-xs">
   731→                <Button variant="ghost" size="sm" onClick={selectAll}>
   732→                  Todos
   733→                </Button>
   734→                <Button variant="ghost" size="sm" onClick={selectOnlyCreates}>
   735→                  Solo nuevos
   736→                </Button>
   737→                <Button variant="ghost" size="sm" onClick={selectNone}>
   738→                  Ninguno
   739→                </Button>
   740→              </div>
   741→            </div>
   742→
   743→            <ScrollArea className="flex-1 border rounded-md">
   744→              <Table>
   745→                <TableHeader>
   746→                  <TableRow>
   747→                    <TableHead className="w-10"></TableHead>
   748→                    <TableHead className="w-12">#</TableHead>
   749→                    <TableHead>Acción</TableHead>
   750→                    <TableHead>Nombre</TableHead>
   751→                    <TableHead className="text-muted-foreground">
   752→                      Detalle
   753→                    </TableHead>
   754→                  </TableRow>
   755→                </TableHeader>
   756→                <TableBody>
   757→                  {preview.items.slice(0, 200).map((item) => {
   758→                    const selectable =
   759→                      item.action === "create" || item.action === "update";
   760→                    const checked = selectedRowIndices.has(item.rowIndex);
   761→                    return (
   762→                      <TableRow
   763→                        key={item.rowIndex}
   764→                        className={
   765→                          !selectable
   766→                            ? "opacity-50"
   767→                            : checked
   768→                            ? "bg-indigo-50/50"
   769→                            : ""
   770→                        }
   771→                      >
   772→                        <TableCell>
   773→                          {selectable && (
   774→                            <Checkbox
   775→                              checked={checked}
   776→                              onCheckedChange={() => toggleRow(item.rowIndex)}
   777→                            />
   778→                          )}
   779→                        </TableCell>
   780→                        <TableCell className="text-xs text-muted-foreground">
   781→                          {item.rowIndex}
   782→                        </TableCell>
   783→                        <TableCell>
   784→                          {item.action === "create" && (
   785→                            <Badge
   786→                              variant="outline"
   787→                              className="bg-emerald-50 text-emerald-700 border-emerald-200"
   788→                            >
   789→                              Crear
   790→                            </Badge>
   791→                          )}
   792→                          {item.action === "update" && (
   793→                            <Badge
   794→                              variant="outline"
   795→                              className="bg-amber-50 text-amber-700 border-amber-200"
   796→                            >
   797→                              Actualizar
   798→                            </Badge>
   799→                          )}
   800→                          {item.action === "error" && (
   801→                            <Badge
   802→                              variant="outline"
   803→                              className="bg-red-50 text-red-700 border-red-200"
   804→                            >
   805→                              Error
   806→                            </Badge>
   807→                          )}
   808→                        </TableCell>
   809→                        <TableCell className="font-medium">
   810→                          {item.name || (
   811→                            <span className="text-muted-foreground italic">
   812→                              sin nombre
   813→                            </span>
   814→                          )}
   815→                        </TableCell>
   816→                        <TableCell className="text-xs text-muted-foreground">
   817→                          {item.action === "update" && (
   818→                            <span>
   819→                              Ya existe como{" "}
   820→                              <span className="font-medium">
   821→                                {item.existingName}
   822→                              </span>{" "}
   823→                              (coincidencia por{" "}
   824→                              <span className="font-mono">{item.matchBy}</span>)
   825→                            </span>
   826→                          )}
   827→                          {item.action === "error" && (
   828→                            <span className="text-red-600">{item.error}</span>
   829→                          )}
   830→                          {item.action === "create" &&
   831→                            item.data?.barcode &&
   832→                            `Código: ${item.data.barcode}`}
   833→                          {item.action === "create" &&
   834→                            !item.data?.barcode &&
   835→                            item.data?.cuit &&
   836→                            `CUIT: ${item.data.cuit}`}
   837→                        </TableCell>
   838→                      </TableRow>
   839→                    );
   840→                  })}
   841→                </TableBody>
   842→              </Table>
   843→              {preview.items.length > 200 && (
   844→                <div className="p-2 text-center text-xs text-muted-foreground">
   845→                  Mostrando primeros 200 de {preview.items.length} items
   846→                </div>
   847→              )}
   848→            </ScrollArea>
   849→
   850→            <div className="text-xs text-muted-foreground shrink-0">
   851→              {selectedRowIndices.size} de{" "}
   852→              {preview.summary.create + preview.summary.update} {entityLabelPlural}{" "}
   853→              seleccionados
   854→            </div>
   855→
   856→            {errorMsg && (
   857→              <Alert variant="destructive" className="shrink-0">
   858→                <XCircle className="h-4 w-4" />
   859→                <AlertDescription>{errorMsg}</AlertDescription>
   860→              </Alert>
   861→            )}
   862→          </div>
   863→        )}
   864→
   865→        {/* ════════════════════════════════════════════════════════════════════════
   866→                Fase committing                                                        */}
   867→        {phase === "committing" && (
   868→          <div className="py-12 text-center">
   869→            <Loader2 className="w-8 h-8 animate-spin mx-auto text-indigo-600 mb-3" />
   870→            <p className="text-sm text-muted-foreground">
   871→              Guardando {selectedRowIndices.size} {entityLabelPlural}…
   872→            </p>
   873→          </div>
   874→        )}
   875→
   876→        {/* ════════════════════════════════════════════════════════════════════════
   877→                PASO 4: RESULTADO                                                       */}
   878→        {phase === "done" && commitResult && (
   879→          <div className="space-y-4 py-2">
   880→            <div className="grid grid-cols-3 gap-3">
   881→              <div className="rounded-lg border bg-emerald-50/50 p-4 text-center">
   882→                <CheckCircle2 className="w-6 h-6 mx-auto text-emerald-600 mb-1" />
   883→                <div className="text-2xl font-bold text-emerald-700">
   884→                  {commitResult.created}
   885→                </div>
   886→                <div className="text-xs text-muted-foreground">creados</div>
   887→              </div>
   888→              <div className="rounded-lg border bg-amber-50/50 p-4 text-center">
   889→                <AlertTriangle className="w-6 h-6 mx-auto text-amber-600 mb-1" />
   890→                <div className="text-2xl font-bold text-amber-700">
   891→                  {commitResult.updated}
   892→                </div>
   893→                <div className="text-xs text-muted-foreground">actualizados</div>

   896→                <XCircle className="w-6 h-6 mx-auto text-red-600 mb-1" />
   897→                <div className="text-2xl font-bold text-red-700">
   898→                  {commitResult.errors.length}
   899→                </div>
   900→                <div className="text-xs text-muted-foreground">errores</div>
   901→              </div>
   902→            </div>
   903→
   904→            {commitResult.errors.length > 0 && (
   905→              <ScrollArea className="h-32 border rounded-md">
   906→                <div className="p-3 space-y-1 text-xs">
   907→                  {commitResult.errors.slice(0, 50).map((e, i) => (
   908→                    <div key={i} className="flex gap-2">
   909→                      <span className="text-muted-foreground">#{e.rowIndex}</span>
   910→                      <span className="font-medium">{e.name}</span>
   911→                      <span className="text-red-600">{e.error}</span>
   912→                    </div>
   913→                  ))}
   914→                  {commitResult.errors.length > 50 && (
   915→                    <div className="text-muted-foreground italic">
   916→                      …y {commitResult.errors.length - 50} errores más
   917→                    </div>
   918→                  )}
   919→                </div>
   920→              </ScrollArea>
   921→            )}
   922→          </div>
   923→        )}
   924→
   925→        {/* ════════════════════════════════════════════════════════════════════════
   926→                FOOTER con acciones según fase                                          */}
   927→        <DialogFooter>
   928→          {phase === "idle" && (
   929→            <Button variant="outline" onClick={() => handleOpenChange(false)}>
   930→              Cancelar
   931→            </Button>
   932→          )}
   933→
   934→          {phase === "mapping" && (
   935→            <>
   936→              <Button
   937→                variant="outline"
   938→                onClick={() => {
   939→                  setPhase("idle");
   940→                  setParsed(null);
   941→                  setColumnMapping({});
   942→                  setErrorMsg("");
   943→                  if (fileInputRef.current) fileInputRef.current.value = "";
   944→                }}
   945→              >
   946→                <ArrowLeft className="w-4 h-4 mr-1" />
   947→                Atrás
   948→              </Button>
   949→              <Button
   950→                onClick={handlePreview}
   951→                disabled={!requiredSatisfied}
   952→                className="bg-indigo-600 hover:bg-indigo-700"
   953→              >
   954→                Continuar a revisión
   955→                <ArrowRight className="w-4 h-4 ml-1" />
   956→              </Button>
   957→            </>
   958→          )}
   959→
   960→          {phase === "preview" && (
   961→            <>
   962→              <Button
   963→                variant="outline"
   964→                onClick={() => {
   965→                  setPhase("mapping");
   966→                  setPreview(null);
   967→                  setErrorMsg("");
   968→                }}
   969→              >
   970→                <ArrowLeft className="w-4 h-4 mr-1" />
   971→                Volver al mapeo
   972→              </Button>
   973→              <Button
   974→                onClick={handleCommit}
   975→                disabled={selectedRowIndices.size === 0}
   976→                className="bg-indigo-600 hover:bg-indigo-700"
   977→              >
   978→                Importar {selectedRowIndices.size} {entityLabelPlural}
   979→              </Button>
   980→            </>
   981→          )}
   982→
   983→          {phase === "done" && (
   984→            <Button
   985→              onClick={() => handleOpenChange(false)}
   986→              className="bg-indigo-600 hover:bg-indigo-700"
   987→            >
   988→              Cerrar
   989→            </Button>
   990→          )}
   991→        </DialogFooter>
   992→      </DialogContent>
   993→    </Dialog>
   994→  );
   995→}
