   198→  const addToCart = useCallback((product: Product) => {
   199→    setCart((prev) => {

   201→      if (existing) {

   203→          toast.error(`Stock máximo alcanzado para ${product.name}`);
   204→          return prev;
   205→        }
   206→        return prev.map((i) =>
   207→          i.product.id === product.id ? { ...i, qty: i.qty + 1 } : i
   208→        );
   209→      }
   210→      if (product.stock <= 0) {
   211→        toast.error(`${product.name} sin stock`);
   212→        return prev;
   213→      }

   221→  function handleSearchEnter(e: React.KeyboardEvent) {
   222→    if (e.key !== "Enter") return;
   223→    const term = search.trim();
   224→    if (!term) return;
   225→    // Match exacto por barcode o SKU
   226→    const exact = products.find(
   227→      (p) => p.barcode === term || (p.sku && p.sku.toLowerCase() === term.toLowerCase())
   228→    );
   229→    if (exact) {
   230→      addToCart(exact);
   231→      setSearch("");
   232→      return;
   233→    }
   234→    // Un solo resultado filtrado → agregar
   235→    if (filtered.length === 1) {
   236→      addToCart(filtered[0]);
   237→      setSearch("");
   238→    }
   239→  }

   241→  function updateQty(productId: string, delta: number) {
   242→    setCart((prev) =>
   243→      prev
   244→        .map((i) => {
   245→          if (i.product.id !== productId) return i;
   246→          const nextQty = i.qty + delta;
   247→          if (nextQty > i.product.stock) {
   248→            toast.error(`Solo hay ${i.product.stock} ${unitLabel(i.product.unit).toLowerCase()}s disponibles`);
   249→            return i;
   250→          }
   251→          return { ...i, qty: nextQty };
   252→        })
   253→        .filter((i) => i.qty > 0)
   254→    );
   255→  }

   257→  function setQty(productId: string, qty: number) {
   258→    setCart((prev) =>
   259→      prev
   260→        .map((i) =>
   261→          i.product.id === productId
   262→            ? { ...i, qty: Math.max(0, Math.min(qty, i.product.stock)) }
   263→            : i
   264→        )
   265→        .filter((i) => i.qty > 0)
   266→    );
   267→  }

   269→  function removeItem(productId: string) {
   270→    setCart((prev) => prev.filter((i) => i.product.id !== productId));
   271→  }

   273→  function clearCart() {
   274→    setCart([]);
   275→    setCustomerId("");
   276→    setDiscount(0);
   277→    setNotes("");
   278→  }
   279→
   280→  const subtotal = cart.reduce((s, i) => s + i.product.salePrice * i.qty, 0);
   281→
   282→  // Promoción aplicada (descuento automático)
   283→  const promotionDiscount = appliedPromotion?.discountAmount || 0;
   284→  // Puntos a canjear (validar máximo permitido)
   285→  const maxRedeemablePoints =
   286→    selectedCustomer && loyaltyProgram?.enabled
   287→      ? calculateMaxRedeemablePoints(
   288→          selectedCustomer.loyaltyPoints || 0,
   289→          subtotal - promotionDiscount,
   290→          {
   291→            ...loyaltyProgram,
   292→            enabled: true,
   293→            roundMode: loyaltyProgram.roundMode as any,
   294→          }
   295→        )
   296→      : 0;
   297→  const effectivePointsToRedeem = Math.min(pointsToRedeem, maxRedeemablePoints);
   298→  const pointsCurrencyDiscount =
   299→    effectivePointsToRedeem > 0 && loyaltyProgram?.enabled
   300→      ? pointsToCurrency(effectivePointsToRedeem, {
   301→          ...loyaltyProgram,
   302→          enabled: true,
   303→          roundMode: loyaltyProgram.roundMode as any,
   304→        })
   305→      : 0;
   306→
   307→  const manualDiscount = Math.min(discount, subtotal);
   308→  const totalDiscount = manualDiscount + promotionDiscount + pointsCurrencyDiscount;
   309→  const taxable = Math.max(0, subtotal - totalDiscount);
   310→  const taxAmount = taxEnabled ? taxable * (taxRate / 100) : 0;
   311→  const selectedMethod = paymentMethods.find((m) => m.id === paymentMethodId);
   312→  const surchargeRate = selectedMethod?.surcharge || 0;
   313→  const surchargeAmount = (taxable + taxAmount) * (surchargeRate / 100);
   314→  const total = taxable + taxAmount + surchargeAmount;
   315→
   316→  // Evaluar promociones disponibles (al cambiar carrito)
   317→  useEffect(() => {
   318→    if (cart.length === 0) {
   319→      setAvailablePromotions([]);

   356→    setPointsToRedeem(0);
   357→  }, [customerId]);

   359→  async function printThermalSale() {
   360→    if (!lastSale?.id) return;
   361→    try {
   362→      const printRes = await safeFetchJSON<any>("/api/print", {
   363→        method: "POST",
   364→        body: JSON.stringify({ type: "TICKET", saleId: lastSale.id }),
   365→      });
   366→      if (!printRes.ok) throw new Error(printRes.error);
   367→
   368→      // Descargar binario .bin que se puede enviar a impresora USB
   369→      const blobRes = await fetch("/api/print", {
   370→        method: "POST",
   371→        headers: { "Content-Type": "application/json" },
   372→        body: JSON.stringify({
   373→          type: "TICKET",
   374→          saleId: lastSale.id,
   375→          returnFormat: "blob",
   376→        }),
   377→      });
   378→      if (!blobRes.ok) {
   379→        throw new Error("No se pudo generar el archivo del ticket");
   380→      }
   381→      const blob = await blobRes.blob();
   382→      const url = URL.createObjectURL(blob);
   383→      const a = document.createElement("a");
   384→      a.href = url;
   385→      a.download = `ticket-${lastSale.id.slice(-6)}.bin`;
   386→      document.body.appendChild(a);
   387→      a.click();
   388→      document.body.removeChild(a);
   389→      URL.revokeObjectURL(url);
   390→
   391→      toast.success("Ticket generado (archivo .bin)");
   392→    } catch (e: any) {
   393→      toast.error("Error al imprimir", { description: e.message });
   394→    }
   395→  }

   397→  async function processSale() {
   398→    if (cart.length === 0) {
   399→      toast.error("El carrito está vacío");
   400→      return;
   401→    }
   402→    if (!paymentMethodId) {
   403→      toast.error("Seleccioná un método de pago");
   404→      return;
   405→    }
   406→    if (selectedMethod?.type === "CUENTA" && !customerId) {
   407→      toast.error("Para cuenta corriente tenés que seleccionar un cliente");
   408→      return;
   409→    }
   410→    setProcessing(true);
   411→    try {
   412→      const saleRes = await safeFetchJSON<any>("/api/sales", {
   413→        method: "POST",
   414→        body: JSON.stringify({
   415→          items: cart.map((i) => ({
   416→            productId: i.product.id,
   417→            quantity: i.qty,
   418→          })),
   419→          customerId: customerId || null,
   420→          discount: manualDiscount,
   421→          paymentMethodId,
   422→          notes,
   423→          taxRate: taxEnabled ? taxRate : 0,
   424→          branchId: branchId || null,
   425→          promotionId: appliedPromotion?.promotionId || null,
   426→          promotionDiscount: appliedPromotion?.discountAmount || 0,
   427→          loyaltyPointsUsed: effectivePointsToRedeem || 0,
   428→        }),
   429→      });
   430→      if (!saleRes.ok || !saleRes.data) {
   431→        throw new Error(saleRes.error || "No se pudo registrar la venta");
   432→      }
   433→      const data = saleRes.data;
   434→      toast.success("Venta registrada!");
   435→      setLastSale({
   436→        ...data,
   437→        items: cart,
   438→        customer: customers.find((c) => c.id === customerId),
   439→        paymentMethod: selectedMethod,
   440→        discount: totalDiscount,
   441→        tax: taxAmount,
   442→        surcharge: surchargeAmount,
   443→        total,
   444→        subtotal,
   445→        appliedPromotion,
   446→        pointsUsed: effectivePointsToRedeem,
   447→        pointsEarned: data.loyaltyPointsEarned || 0,
   448→      });
   449→      setCheckoutOpen(false);
   450→      setReceiptOpen(true);
   451→      clearCart();
   452→      setAppliedPromotion(null);
   453→      setPointsToRedeem(0);
   454→      // Refrescar stock y clientes en paralelo, sin romper si fallan
   455→      // (la venta ya se registró, no queremos que el cajero vea un error
   456→      // acá y crea que la venta no se hizo).
   457→      try {
   458→        const [refreshedRes, refreshedCustRes] = await Promise.all([
   459→          safeFetchJSON<Product[]>("/api/products"),
   460→          safeFetchJSON<Customer[]>("/api/customers"),
   461→        ]);
   462→        const refreshed = Array.isArray(refreshedRes.data) ? refreshedRes.data : [];
   463→        setProducts(refreshed.filter((x: Product) => x.active));
   464→        if (Array.isArray(refreshedCustRes.data)) {
   465→          setCustomers(refreshedCustRes.data);
   466→        }
   467→      } catch {
   468→        // Silencioso: el refresh es best-effort.
   469→      }
   470→    } catch (e: any) {
   471→      toast.error(e.message);
   472→    } finally {
   473→      setProcessing(false);
   474→    }
   475→  }
   476→
   477→  return (
   478→    <div className="space-y-4 h-full">
   479→      <div className="flex items-end justify-between flex-wrap gap-3">
   480→        <div>
   481→          <h1 className="text-2xl font-bold tracking-tight">Punto de Venta</h1>
   482→          <p className="text-sm text-muted-foreground">
   483→            Buscá productos, agregalos al carrito y registrá la venta
   484→          </p>
   485→        </div>
   486→        {branches.length > 1 && (
   487→          <div className="flex items-center gap-2">
   488→            <Label className="text-xs text-muted-foreground">Sucursal:</Label>
   489→            <Select value={branchId} onValueChange={setBranchId}>
   490→              <SelectTrigger className="w-48 h-9">
   491→                <SelectValue />
   492→              </SelectTrigger>
   493→              <SelectContent>
   494→                {branches
   495→                  .filter((b) => b.active)
   496→                  .map((b) => (
   497→                    <SelectItem key={b.id} value={b.id}>
   498→                      <span className="inline-flex items-center gap-1.5">
   499→                        {b.isMain && <Star className="w-3 h-3 fill-amber-400 text-amber-400" />}
   500→                        {b.name} ({b.code})
   501→                      </span>
   502→                    </SelectItem>
   503→                  ))}
   504→              </SelectContent>
   505→            </Select>
   506→          </div>
   507→        )}
   508→      </div>
   509→
   510→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-4">
   511→        {/* Productos - 3 cols */}
   512→        <div className="lg:col-span-3 space-y-3">
   513→          <Card>
   514→            <CardContent className="p-3 space-y-2">
   515→              <div className="relative">
   516→                <Barcode className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   517→                <Input
   518→                  placeholder="Escanear código o buscar producto..."
   519→                  value={search}
   520→                  onChange={(e) => setSearch(e.target.value)}
   521→                  onKeyDown={handleSearchEnter}
   522→                  className="pl-9 h-11 text-base"
   523→                  autoFocus
   524→                />
   525→              </div>
   526→              <div className="flex gap-2 overflow-x-auto pb-1">
   527→                <Button
   528→                  variant={filterCat === "all" ? "default" : "outline"}
   529→                  size="sm"
   530→                  onClick={() => setFilterCat("all")}
   531→                  className={
   532→                    filterCat === "all" ? "bg-indigo-600 hover:bg-indigo-700" : ""
   533→                  }
   534→                >
   535→                  Todos
   536→                </Button>
   537→                {categories.map((c) => (
   538→                  <Button
   539→                    key={c.id}
   540→                    variant={filterCat === c.id ? "default" : "outline"}
   541→                    size="sm"
   542→                    onClick={() => setFilterCat(c.id)}
   543→                    className={
   544→                      filterCat === c.id ? "bg-indigo-600 hover:bg-indigo-700 whitespace-nowrap" : "whitespace-nowrap"
   545→                    }
   546→                  >
   547→                    {c.name}
   548→                  </Button>
   549→                ))}
   550→              </div>
   551→            </CardContent>
   552→          </Card>
   553→
   554→          {loading ? (
   555→            <div className="p-12 text-center">
   556→              <Loader2 className="w-6 h-6 animate-spin mx-auto text-indigo-600" />
   557→            </div>
   558→          ) : (
   559→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2 max-h-[60vh] overflow-y-auto pr-1">
   560→              {filtered.map((p) => {
   561→                const labels = p.labels
   562→                  ? p.labels.split(",").map((s) => s.trim()).filter(Boolean)
   563→                  : [];
   564→                const allergens = p.allergens
   565→                  ? p.allergens.split(",").map((s) => s.trim()).filter(Boolean)
   566→                  : [];
   567→                return (
   568→                <button
   569→                  key={p.id}
