     3→     3→/**
     4→     4→ * Dialog de importación masiva reutilizable con mapeo manual de columnas.
     5→     5→ *
     6→     6→ * Flujo (wizard de 4 pasos):
     7→     7→ *   1. SUBIR: usuario elige un archivo CSV/XLSX/JSON.
     8→     8→ *   2. MAPEAR: el sistema detecta automáticamente las columnas y sugiere
     9→     9→ *      un mapeo. El usuario puede cambiar qué columna va a cada campo.
    10→    10→ *      Una columna del archivo puede no estar mapeada; un campo puede
    11→    11→ *      quedar sin mapear (salvo los obligatorios).
    12→    12→ *   3. PREVIEW: el servidor procesa con el mapeo elegido y muestra la
    13→    13→ *      lista de items a crear / actualizar / con error. El usuario
    14→    14→ *      selecciona qué items confirmar.
    15→    15→ *   4. RESULTADO: estadísticas finales (creados, actualizados, errores).
    16→    16→ *
    17→    17→ * Props:
    18→    18→ *   - endpoint:       URL del API route ("/api/products/import" | "/api/customers/import")
    19→    19→ *   - fields:         lista de ImportField (de @/lib/import-config) que definen
    20→    20→ *                     qué campos puede mapear el usuario.
    21→    21→ *   - templateHeaders: columnas sugeridas para la plantilla descargable
    22→    22→ *   - templateName:    nombre del archivo de plantilla
    23→    23→ *   - entityLabel:     "producto" | "cliente"
    24→    24→ *   - entityLabelPlural: "productos" | "clientes"
    25→    25→ *   - onImported:      callback al terminar (para que el view recargue la lista)
    26→    26→ */

    84→    84→
    85→    85→export interface ImportPreviewItem {
    86→    86→  action: "create" | "update" | "error";
    87→    87→  rowIndex: number;
    88→    88→  name: string;
    89→    89→  existingId?: string;
    90→    90→  existingName?: string;
    91→    91→  matchBy?: string;

    95→    95→
    96→    96→export interface ImportPreviewResponse {

    98→    98→  summary: {
    99→    99→    create: number;
   100→   100→    update: number;
   101→   101→    error: number;
   102→   102→  };
   103→   103→  columnMapping?: Record<string, number>;
   104→   104→}
   105→   105→
   106→   106→export interface ImportCommitResponse {
   107→   107→  created: number;
   108→   108→  updated: number;
   109→   109→  errors: { rowIndex: number; name: string; error: string }[];
   110→   110→  total: number;
   111→   111→}
   112→   112→
   113→   113→interface ImportDialogProps {
   114→   114→  open: boolean;
   115→   115→  onOpenChange: (open: boolean) => void;
   116→   116→  endpoint: string;
   117→   117→  fields: ImportField[];
   118→   118→  templateHeaders: string[];
   119→   119→  templateName: string;
   120→   120→  entityLabel: string;
   121→   121→  entityLabelPlural: string;
   122→   122→  onImported?: () => void;
   123→   123→}
   124→   124→
   125→   125→type Phase = "idle" | "parsing" | "mapping" | "previewing" | "committing" | "done";
   126→   126→
   127→   127→export function ImportDialog({
   128→   128→  open,
   129→   129→  onOpenChange,
   130→   130→  endpoint,
   131→   131→  fields,
   132→   132→  templateHeaders,
   133→   133→  templateName,
   134→   134→  entityLabel,
   135→   135→  entityLabelPlural,
   136→   136→  onImported,
   137→   137→}: ImportDialogProps) {
   138→   138→  const [phase, setPhase] = useState<Phase>("idle");
   139→   139→  const [fileName, setFileName] = useState("");
   140→   140→  const [parsed, setParsed] = useState<ParsedFile | null>(null);
   141→   141→  // columnMapping: { fieldKey: columnIndex }
   142→   142→  // columnIndex = -1 significa "no mapeado" (lo usamos internamente para
   143→   143→  // distinguir de un campo que todavía no fue tocado por el usuario).
   144→   144→  const [columnMapping, setColumnMapping] = useState<Record<string, number>>({});
   145→   145→  const [preview, setPreview] = useState<ImportPreviewResponse | null>(null);
   146→   146→  const [selectedRowIndices, setSelectedRowIndices] = useState<Set<number>>(
   147→   147→    new Set()
   148→   148→  );
   149→   149→  const [commitResult, setCommitResult] = useState<ImportCommitResponse | null>(
   150→   150→    null
   151→   151→  );
   152→   152→  const [errorMsg, setErrorMsg] = useState("");
   153→   153→  const fileInputRef = useRef<HTMLInputElement>(null);
   154→   154→
   155→   155→  const reset = useCallback(() => {
   156→   156→    setPhase("idle");
   157→   157→    setFileName("");
   158→   158→    setParsed(null);
   159→   159→    setColumnMapping({});
   160→   160→    setPreview(null);
   161→   161→    setSelectedRowIndices(new Set());
   162→   162→    setCommitResult(null);
   163→   163→    setErrorMsg("");
   164→   164→    if (fileInputRef.current) fileInputRef.current.value = "";
   165→   165→  }, []);
   166→   166→
   167→   167→  function handleOpenChange(next: boolean) {
   168→   168→    if (!next) reset();
   169→   169→    onOpenChange(next);
   170→   170→  }
   171→   171→
   172→   172→  function handleTemplate() {
   173→   173→    const csv = generateCSV(templateHeaders);
   174→   174→    downloadTextFile(templateName, csv);
   175→   175→    toast.success("Plantilla descargada", {
   176→   176→      description: `Abrí ${templateName} en Excel o Google Sheets`,
   177→   177→    });
   178→   178→  }
   179→   179→
   180→   180→  // ─── Paso 1 → 2: parsear archivo y saltar a mapeo ───────────────────────────
   181→   181→  async function handleFile(file: File) {
   182→   182→    setPhase("parsing");
   183→   183→    setErrorMsg("");
   184→   184→    setFileName(file.name);
   185→   185→
   186→   186→    let result: ParsedFile;
   187→   187→    try {
   188→   188→      result = await parseFile(file);
   189→   189→    } catch (e: any) {
   190→   190→      setErrorMsg(e?.message || "No se pudo leer el archivo");
   191→   191→      setPhase("idle");
   192→   192→      return;
   193→   193→    }
   194→   194→
   195→   195→    if (!result.headers.length || !result.rows.length) {
   196→   196→      setErrorMsg("El archivo está vacío o no tiene filas de datos");
   197→   197→      setPhase("idle");
   198→   198→      return;
   199→   199→    }
   200→   200→
   201→   201→    setParsed(result);
   202→   202→    // Sugerir mapeo automático
   203→   203→    const suggested = suggestColumnMapping(result.headers, fields);
   204→   204→    setColumnMapping(suggested);
   205→   205→    setPhase("mapping");
   206→   206→  }
   207→   207→
   208→   208→  // ─── Helpers de mapeo ────────────────────────────────────────────────────────
   209→   209→  function setFieldMapping(fieldKey: string, columnIndex: number) {
   210→   210→    setColumnMapping((prev) => {
   211→   211→      const next = { ...prev };
   212→   212→      if (columnIndex < 0) {
   213→   213→        delete next[fieldKey];
   214→   214→      } else {
   215→   215→        next[fieldKey] = columnIndex;
   216→   216→      }
   217→   217→      return next;
   218→   218→    });
   219→   219→  }
   220→   220→
   221→   221→  function autoDetectAll() {
   222→   222→    if (!parsed) return;
   223→   223→    const suggested = suggestColumnMapping(parsed.headers, fields);
   224→   224→    setColumnMapping(suggested);
   225→   225→    toast.success("Mapeo automático aplicado", {
   226→   226→      description: "Revisá que las columnas estén bien asignadas",
   227→   227→    });
   228→   228→  }
   229→   229→
   230→   230→  function clearAllMappings() {
   231→   231→    setColumnMapping({});
   232→   232→  }
   233→   233→
   234→   234→  // Stats del mapeo para mostrar en la UI
   235→   235→  const mappingStats = useMemo(() => {
   236→   236→    const mapped = Object.keys(columnMapping).length;
   237→   237→    const required = fields.filter((f) => f.required);
   238→   238→    const requiredMapped = required.filter((r) => r.key in columnMapping).length;
   239→   239→    const totalRequired = required.length;
   240→   240→    return { mapped, requiredMapped, totalRequired };
   241→   241→  }, [columnMapping, fields]);
   242→   242→
   243→   243→  const requiredSatisfied = mappingStats.requiredMapped === mappingStats.totalRequired;
   244→   244→
   245→   245→  // ─── Paso 2 → 3: enviar mapeo al servidor para preview ──────────────────────
   246→   246→  async function handlePreview() {
   247→   247→    if (!parsed) return;
   248→   248→    if (!requiredSatisfied) {
   249→   249→      toast.error("Faltan campos obligatorios por mapear");
   250→   250→      return;
   251→   251→    }
   252→   252→
   253→   253→    setPhase("previewing");
   254→   254→    setErrorMsg("");
   255→   255→    try {
   256→   256→      const res = await fetch(endpoint, {
   257→   257→        method: "POST",
   258→   258→        headers: { "Content-Type": "application/json" },
   259→   259→        body: JSON.stringify({
   260→   260→          mode: "preview",
   261→   261→          headers: parsed.headers,
   262→   262→          rows: parsed.rows,
   263→   263→          columnMapping,
   264→   264→        }),
   265→   265→      });
   266→   266→      const data = await res.json();
   267→   267→      if (!res.ok) throw new Error(data.error || "Error en el servidor");
   268→   268→
   269→   269→      const previewData = data as ImportPreviewResponse;
   270→   270→      setPreview(previewData);
   271→   271→
   272→   272→      // Por defecto: seleccionar todos los create y update, excluir errors
   273→   273→      const selected = new Set<number>();
   274→   274→      previewData.items.forEach((item) => {
   275→   275→        if (item.action === "create" || item.action === "update") {
   276→   276→          selected.add(item.rowIndex);
   277→   277→        }
   278→   278→      });
   279→   279→      setSelectedRowIndices(selected);
   280→   280→      setPhase("preview");
   281→   281→    } catch (e: any) {
   282→   282→      setErrorMsg(e?.message || "Error al analizar el archivo");
   283→   283→      setPhase("mapping");
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  // ─── Selección de items en el preview ───────────────────────────────────────
   288→   288→  function toggleRow(rowIndex: number) {
   289→   289→    setSelectedRowIndices((prev) => {
   290→   290→      const next = new Set(prev);
   291→   291→      if (next.has(rowIndex)) next.delete(rowIndex);
   292→   292→      else next.add(rowIndex);
   293→   293→      return next;
   294→   294→    });
   295→   295→  }
   296→   296→  function selectAll() {
   297→   297→    if (!preview) return;
   298→   298→    const next = new Set<number>();
   299→   299→    preview.items.forEach((item) => {
   300→   300→      if (item.action === "create" || item.action === "update") {
   301→   301→        next.add(item.rowIndex);
   302→   302→      }
   303→   303→    });
   304→   304→    setSelectedRowIndices(next);
   305→   305→  }
   306→   306→  function selectNone() {
   307→   307→    setSelectedRowIndices(new Set());
   308→   308→  }
   309→   309→  function selectOnlyCreates() {
   310→   310→    if (!preview) return;
   311→   311→    const next = new Set<number>();
   312→   312→    preview.items.forEach((item) => {
   313→   313→      if (item.action === "create") next.add(item.rowIndex);
   314→   314→    });
   315→   315→    setSelectedRowIndices(next);
   316→   316→  }
   317→   317→
   318→   318→  // ─── Paso 3 → 4: confirmar y commitear ──────────────────────────────────────
   319→   319→  async function handleCommit() {
   320→   320→    if (!preview) return;
   321→   321→    const itemsToCommit = preview.items.filter((i) =>
   322→   322→      selectedRowIndices.has(i.rowIndex)
   323→   323→    );
   324→   324→    if (itemsToCommit.length === 0) {
   325→   325→      toast.error("No seleccionaste ningún item para importar");
   326→   326→      return;
   327→   327→    }
   328→   328→
   329→   329→    setPhase("committing");
   330→   330→    try {
   331→   331→      const res = await fetch(endpoint, {
   332→   332→        method: "POST",
   333→   333→        headers: { "Content-Type": "application/json" },
   334→   334→        body: JSON.stringify({
   335→   335→          mode: "commit",
   336→   336→          items: itemsToCommit,
   337→   337→        }),
   338→   338→      });
   339→   339→      const data = await res.json();
   340→   340→      if (!res.ok) throw new Error(data.error || "Error en el servidor");
   341→   341→
   342→   342→      const result = data as ImportCommitResponse;
   343→   343→      setCommitResult(result);
   344→   344→      setPhase("done");
   345→   345→      onImported?.();
   346→   346→
   347→   347→      const parts: string[] = [];
   348→   348→      if (result.created) parts.push(`${result.created} creados`);
   349→   349→      if (result.updated) parts.push(`${result.updated} actualizados`);
   350→   350→      if (result.errors.length) parts.push(`${result.errors.length} con error`);
   351→   351→      toast.success(`Importación completa`, {
   352→   352→        description: parts.join(" · ") || "Sin cambios",
   353→   353→      });
   354→   354→    } catch (e: any) {
   355→   355→      setErrorMsg(e?.message || "Error al importar");
   356→   356→      setPhase("preview");
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  // ─── Helpers de render del mapeo ─────────────────────────────────────────────
   361→   361→  function columnLabel(idx: number): string {
   362→   362→    if (!parsed) return "";
   363→   363→    return parsed.headers[idx] || `(columna ${idx + 1})`;
   364→   364→  }
   365→   365→  function sampleValues(idx: number): string[] {
   366→   366→    if (!parsed) return [];
   367→   367→    return parsed.rows.slice(0, 3).map((r) => {
   368→   368→      const v = r[idx];
   369→   369→      if (v === null || v === undefined || v === "") return "—";
   370→   370→      return String(v).slice(0, 20);
   371→   371→    });
   372→   372→  }
   373→   373→
   374→   374→  // ─── Render ──────────────────────────────────────────────────────────────────
   375→   375→  return (
   376→   376→    <Dialog open={open} onOpenChange={handleOpenChange}>
   377→   377→      <DialogContent className="max-w-5xl w-[95vw] max-h-[92vh] overflow-hidden flex flex-col">
   378→   378→        <DialogHeader>
   379→   379→          <DialogTitle className="flex items-center gap-2">
   380→   380→            <Upload className="w-5 h-5 text-indigo-600" />
   381→   381→            Importar {entityLabelPlural}
   382→   382→          </DialogTitle>
   383→   383→          <DialogDescription>
   384→   384→            {phase === "idle" && "Subí un archivo CSV, Excel o JSON para comenzar."}
   385→   385→            {phase === "mapping" &&
   386→   386→              "Asigná cada columna del archivo a un campo del sistema. Podés dejar columnas sin mapear."}
   387→   387→            {phase === "preview" &&
   388→   388→              "Revisá el resultado del mapeo y elegí qué items importar."}
   389→   389→            {phase === "done" && "Resultado de la importación."}
   390→   390→          </DialogDescription>
   391→   391→        </DialogHeader>
   392→   392→
   393→   393→        {/* Indicador de pasos */}
   394→   394→        {(phase === "mapping" || phase === "preview" || phase === "previewing") && (
   395→   395→          <div className="flex items-center gap-1 text-xs shrink-0">
   396→   396→            <StepBadge n={1} active={phase === "mapping"} done={phase !== "mapping"}>
   397→   397→              Mapear
   398→   398→            </StepBadge>
   399→   399→            <ArrowRight className="w-3 h-3 text-muted-foreground" />
   400→   400→            <StepBadge n={2} active={phase === "preview"} done={false}>
   401→   401→              Revisar
   402→   402→            </StepBadge>
   403→   403→            <ArrowRight className="w-3 h-3 text-muted-foreground" />
   404→   404→            <StepBadge n={3} active={false} done={false}>
   405→   405→              Confirmar
   406→   406→            </StepBadge>
   407→   407→          </div>
   408→   408→        )}
   409→   409→
   410→   410→        {/* ════════════════════════════════════════════════════════════════════════
   411→   411→                PASO 1: SUBIR ARCHIVO                                                   */}
   412→   412→        {phase === "idle" && (
   413→   413→          <div className="space-y-4 py-2">
   414→   414→            <div className="flex flex-wrap gap-2">
   415→   415→              <Button
   416→   416→                variant="outline"
   417→   417→                size="sm"
   418→   418→                onClick={handleTemplate}
   419→   419→                className="border-dashed"
   420→   420→              >
   421→   421→                <Download className="w-4 h-4 mr-2" />
   422→   422→                Descargar plantilla CSV
   423→   423→              </Button>
   424→   424→            </div>
   425→   425→
   426→   426→            <Alert>
   427→   427→              <FileSpreadsheet className="h-4 w-4" />
   428→   428→              <AlertDescription>
   429→   429→                <div className="space-y-1">
   430→   430→                  <p className="font-medium">Cómo funciona</p>
   431→   431→                  <p className="text-xs text-muted-foreground">
   432→   432→                    Subí tu archivo. En el próximo paso vas a poder mapear las
   433→   433→                    columnas de tu planilla a los campos del sistema, sin
   434→   434→                    importar cómo se llamen. Aceptamos <code>.csv</code>,{" "}
   435→   435→                    <code>.xlsx</code> y <code>.json</code>.
   436→   436→                  </p>
   437→   437→                </div>
   438→   438→              </AlertDescription>
   439→   439→            </Alert>
   440→   440→
   441→   441→            <div
   442→   442→              className="border-2 border-dashed rounded-lg p-8 text-center cursor-pointer hover:bg-muted/50 transition-colors"
   443→   443→              onClick={() => fileInputRef.current?.click()}
   444→   444→              onDragOver={(e) => e.preventDefault()}
   445→   445→              onDrop={(e) => {
   446→   446→                e.preventDefault();
   447→   447→                const file = e.dataTransfer.files?.[0];
   448→   448→                if (file) handleFile(file);
   449→   449→              }}
   450→   450→            >
   451→   451→              <Upload className="w-10 h-10 mx-auto text-muted-foreground mb-2" />
   452→   452→              <p className="text-sm font-medium">
   453→   453→                Hacé click o arrastrá un archivo aquí
   454→   454→              </p>
   455→   455→              <p className="text-xs text-muted-foreground mt-1">
   456→   456→                CSV, XLSX o JSON · máx ~5MB recomendado
   457→   457→              </p>
   458→   458→              <Input
   459→   459→                ref={fileInputRef}
   460→   460→                type="file"
   461→   461→                accept=".csv,.xlsx,.xls,.json"
   462→   462→                className="hidden"
   463→   463→                onChange={(e) => {
   464→   464→                  const file = e.target.files?.[0];
   465→   465→                  if (file) handleFile(file);
   466→   466→                }}
   467→   467→              />
   468→   468→            </div>
   469→   469→
   470→   470→            {errorMsg && (
   471→   471→              <Alert variant="destructive">
   472→   472→                <XCircle className="h-4 w-4" />
   473→   473→                <AlertDescription>{errorMsg}</AlertDescription>
   474→   474→              </Alert>
   475→   475→            )}
   476→   476→          </div>
   477→   477→        )}
   478→   478→
   479→   479→        {/* ════════════════════════════════════════════════════════════════════════
   480→   480→                PASO 2: MAPEAR COLUMNAS                                                 */}
   481→   481→        {(phase === "mapping" || phase === "previewing") && parsed && (
   482→   482→          <div className="space-y-3 flex-1 overflow-hidden flex flex-col">
   483→   483→            <div className="flex items-center justify-between gap-2 flex-wrap shrink-0">
