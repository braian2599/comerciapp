     1→"use client";
     2→
     3→import { useEffect, useState, useMemo } from "react";
     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";
     8→import { Textarea } from "@/components/ui/textarea";
     9→import { Badge } from "@/components/ui/badge";
    10→import {
    11→  Dialog,
    12→  DialogContent,
    13→  DialogDescription,
    14→  DialogFooter,
    15→  DialogHeader,
    16→  DialogTitle,
    17→  DialogClose,
    18→} from "@/components/ui/dialog";
    19→import {
    20→  Table,
    21→  TableBody,
    22→  TableCell,
    23→  TableHead,
    24→  TableHeader,
    25→  TableRow,
    26→} from "@/components/ui/table";
    27→import {

    31→  AlertDialogContent,
    32→  AlertDialogDescription,
    33→  AlertDialogFooter,
    34→  AlertDialogHeader,
    35→  AlertDialogTitle,
    36→} from "@/components/ui/alert-dialog";
    37→import {
    38→  Select,
    39→  SelectContent,
    40→  SelectItem,
    41→  SelectTrigger,
    42→  SelectValue,
    43→} from "@/components/ui/select";
    44→import { toast } from "sonner";
    45→import {
    46→  Plus,
    47→  Search,
    48→  Pencil,
    49→  Trash2,
    50→  Users,
    51→  Phone,
    52→  Mail,
    53→  Loader2,
    54→  Wallet,
    55→  ReceiptText,
    56→  ArrowDownCircle,
    57→  ArrowUpCircle,
    58→  Star,
    59→  Award,
    60→  Gift,
    61→  Upload,
    62→} from "lucide-react";
    63→import { ImportDialog } from "@/components/import-dialog";
    64→import { useAppStore } from "@/store/app-store";
    65→import { formatCurrency, formatDateTime, PAYMENT_METHOD_TYPES } from "@/lib/constants";
    66→import { safeFetchJSON, safeFetchArray } from "@/lib/fetch";
    67→import { CUSTOMER_IMPORT_FIELDS } from "@/lib/import-config";
    68→
    69→interface Customer {
    70→  id: string;
    71→  name: string;
    72→  phone?: string;
    73→  email?: string;
    74→  address?: string;
    75→  notes?: string;
    76→  creditLimit?: number;
    77→  saldo?: number;
    78→  _count?: { sales: number };
    79→  loyaltyPoints?: number;
    80→  loyaltyTier?: string;
    81→  totalSpent?: number;
    82→  totalSales?: number;
    83→}
    84→
    85→interface LedgerItem {
    86→  id: string;
    87→  date: string;
    88→  type: "DEBE" | "HABER";
    89→  description: string;
    90→  amount: number;
    91→  user?: string;
    92→  balanceAfter: number;
    93→}
    94→
    95→interface AccountState {
    96→  customer: Customer;
    97→  saldo: number;
    98→  creditLimit: number;
    99→  disponible: number;
   100→  movimientos: LedgerItem[];
   101→  totalVentas: number;
   102→  totalPagos: number;
   103→}
   104→
   105→const emptyForm = {
   106→  id: "",
   107→  name: "",
   108→  phone: "",
   109→  email: "",
   110→  address: "",
   111→  notes: "",
   112→  creditLimit: 0,
   113→};
   114→
   115→export function CustomersView() {
   116→  const { store } = useAppStore();
   117→  const symbol = store?.currencySymbol || "$";
   118→  const [customers, setCustomers] = useState<Customer[]>([]);
   119→  const [loading, setLoading] = useState(true);
   120→  const [search, setSearch] = useState("");
   121→  const [formOpen, setFormOpen] = useState(false);
   122→  const [form, setForm] = useState<any>(emptyForm);
   123→  const [saving, setSaving] = useState(false);
   124→  const [deleteId, setDeleteId] = useState<string | null>(null);
   125→  const [importOpen, setImportOpen] = useState(false);
   126→
   127→  // Cuenta corriente
   128→  const [accountOpen, setAccountOpen] = useState(false);
   129→  const [accountData, setAccountData] = useState<AccountState | null>(null);
   130→  const [accountLoading, setAccountLoading] = useState(false);
   131→  const [payOpen, setPayOpen] = useState(false);
   132→  const [payAmount, setPayAmount] = useState(0);
   133→  const [payMethod, setPayMethod] = useState("EFECTIVO");
   134→  const [payNotes, setPayNotes] = useState("");
   135→  const [paySaving, setPaySaving] = useState(false);
   136→
   137→  // Fidelización - puntos
   138→  const [pointsOpen, setPointsOpen] = useState(false);
   139→  const [pointsData, setPointsData] = useState<any | null>(null);
   140→  const [pointsLoading, setPointsLoading] = useState(false);
   141→  const [adjustPoints, setAdjustPoints] = useState(0);
   142→  const [adjustNotes, setAdjustNotes] = useState("");
   143→  const [adjustSaving, setAdjustSaving] = useState(false);
   144→
   145→  // Programa de fidelización (cargado una vez)
   146→  const [loyaltyEnabled, setLoyaltyEnabled] = useState(false);
   147→
   148→  async function load() {
   149→    setLoading(true);
   150→    try {
   151→      const [custRes, loyRes] = await Promise.all([
   152→        safeFetchArray<Customer>("/api/customers"),
   153→        safeFetchJSON<any>("/api/loyalty"),
   154→      ]);
   155→      setCustomers(custRes);
   156→      const loy = loyRes.data;
   157→      setLoyaltyEnabled(!!loy?.enabled && typeof loy === "object" && !Array.isArray(loy));
   158→    } catch {
   159→      toast.error("No se pudieron cargar los clientes");
   160→      setCustomers([]);
   161→    } finally {
   162→      setLoading(false);
   163→    }
   164→  }
   165→
   166→  useEffect(() => {
   167→    load();
   168→  }, []);
   169→
   170→  const filtered = useMemo(() => {
   171→    return customers.filter((c) => {
   172→      if (!search) return true;
   173→      const s = search.toLowerCase();
   174→      return (
   175→        c.name.toLowerCase().includes(s) ||
   176→        c.phone?.toLowerCase().includes(s) ||
   177→        c.email?.toLowerCase().includes(s)
   178→      );
   179→    });
   180→  }, [customers, search]);
   181→
   182→  const saldoTotal = customers.reduce((s, c) => s + (c.saldo || 0), 0);
   183→
   184→  function openNew() {
   185→    setForm({ ...emptyForm });
   186→    setFormOpen(true);
   187→  }
   188→
   189→  function openEdit(c: Customer) {
   190→    setForm({ ...c, creditLimit: c.creditLimit || 0 });
   191→    setFormOpen(true);
   192→  }
   193→
   194→  async function openAccount(c: Customer) {
   195→    setAccountOpen(true);
   196→    setAccountLoading(true);
   197→    setAccountData(null);
   198→    try {
   199→      const res = await fetch(`/api/customers/account?customerId=${c.id}`);
   200→      const data = await res.json();
   201→      setAccountData(data);
   202→    } catch {
   203→      toast.error("Error al cargar cuenta");
   204→    } finally {
   205→      setAccountLoading(false);
   206→    }
   207→  }
   208→
   209→  async function openPoints(c: Customer) {
   210→    setPointsOpen(true);
   211→    setPointsLoading(true);
   212→    setPointsData(null);
   213→    setAdjustPoints(0);
   214→    setAdjustNotes("");
   215→    try {
   216→      const res = await fetch(`/api/loyalty/points?customerId=${c.id}`);
   217→      const data = await res.json();
   218→      setPointsData(data);
   219→    } catch {
   220→      toast.error("Error al cargar puntos");
   221→    } finally {
   222→      setPointsLoading(false);
   223→    }
   224→  }
   225→
   226→  async function handleAdjustPoints() {
   227→    if (!pointsData?.customer?.id) return;
   228→    if (adjustPoints === 0) {
   229→      toast.error("Indica puntos (positivo o negativo)");
   230→      return;
   231→    }
   232→    setAdjustSaving(true);
   233→    try {
   234→      const res = await fetch("/api/loyalty/points", {
   235→        method: "POST",
   236→        headers: { "Content-Type": "application/json" },
   237→        body: JSON.stringify({
   238→          customerId: pointsData.customer.id,
   239→          points: adjustPoints,
   240→          description: adjustNotes,
   241→        }),
   242→      });
   243→      const data = await res.json();
   244→      if (!res.ok) throw new Error(data.error);
   245→      toast.success("Puntos ajustados");
   246→      // Recargar
   247→      await openPoints(pointsData.customer);
   248→      load();
   249→    } catch (e: any) {
   250→      toast.error(e.message);
   251→    } finally {
   252→      setAdjustSaving(false);
   253→    }
   254→  }
   255→
   256→  async function handleSave() {
   257→    if (!form.name) {
   258→      toast.error("El nombre es obligatorio");
   259→      return;
   260→    }
   261→    setSaving(true);
   262→    try {
   263→      const method = form.id ? "PUT" : "POST";
   264→      const res = await fetch("/api/customers", {
   265→        method,
   266→        headers: { "Content-Type": "application/json" },
   267→        body: JSON.stringify(form),
   268→      });
   269→      const data = await res.json();
   270→      if (!res.ok) throw new Error(data.error);
   271→      toast.success(form.id ? "Cliente actualizado" : "Cliente creado");
   272→      setFormOpen(false);
   273→      load();
   274→    } catch (e: any) {
   275→      toast.error(e.message);
   276→    } finally {
   277→      setSaving(false);
   278→    }
   279→  }
   280→
   281→  async function handleDelete() {
   282→    if (!deleteId) return;
   283→    try {
   284→      const res = await fetch(`/api/customers?id=${deleteId}`, { method: "DELETE" });
   285→      const data = await res.json();
   286→      if (!res.ok) throw new Error(data.error);
   287→      toast.success("Cliente eliminado");
   288→      setDeleteId(null);
   289→      load();
   290→    } catch (e: any) {
   291→      toast.error(e.message);
   292→    }
   293→  }
   294→
   295→  async function handlePayment() {
   296→    if (!accountData) return;
   297→    if (payAmount <= 0) {
   298→      toast.error("El monto debe ser mayor a 0");
   299→      return;
   300→    }
   301→    setPaySaving(true);
   302→    try {
   303→      const res = await fetch("/api/customers/account", {
   304→        method: "POST",
   305→        headers: { "Content-Type": "application/json" },
   306→        body: JSON.stringify({
   307→          customerId: accountData.customer.id,
   308→          amount: payAmount,
   309→          paymentMethod: payMethod,
   310→          notes: payNotes,
   311→        }),
   312→      });
   313→      const data = await res.json();
   314→      if (!res.ok) throw new Error(data.error);
   315→      toast.success("Pago registrado");
   316→      setPayOpen(false);
   317→      setPayAmount(0);
   318→      setPayNotes("");
   319→      // Recargar cuenta y lista
   320→      await openAccount(accountData.customer);
   321→      load();
   322→    } catch (e: any) {
   323→      toast.error(e.message);
   324→    } finally {
   325→      setPaySaving(false);
   326→    }
   327→  }
   328→
   329→  return (
   330→    <div className="space-y-4">
   331→      <ImportDialog
   332→        open={importOpen}
   333→        onOpenChange={setImportOpen}
   334→        endpoint="/api/customers/import"
   335→        fields={CUSTOMER_IMPORT_FIELDS}
   336→        templateHeaders={[
   337→          "name",
   338→          "phone",
   339→          "email",
   340→          "address",
   341→          "cuit",
   342→          "taxType",
   343→          "creditLimit",
   344→          "notes",
   345→        ]}
   346→        templateName="plantilla_clientes.csv"
   347→        entityLabel="cliente"
   348→        entityLabelPlural="clientes"
   349→        onImported={load}
   350→      />
   351→      <div className="flex items-center justify-between flex-wrap gap-3">
   352→        <div>
   353→          <h1 className="text-2xl font-bold tracking-tight">Clientes</h1>
   354→          <p className="text-sm text-muted-foreground">
   355→            {customers.length} clientes · Saldo total en cuentas:{" "}
   356→            <span className={saldoTotal > 0 ? "text-amber-700 font-medium" : ""}>
   357→              {formatCurrency(saldoTotal, symbol)}
   358→            </span>
   359→          </p>
   360→        </div>
   361→        <div className="flex gap-2">
   362→          <Button
   363→            variant="outline"
   364→            onClick={() => setImportOpen(true)}
   365→          >
   366→            <Upload className="w-4 h-4 mr-2" />
   367→            Importar
   368→          </Button>
   369→          <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   370→            <Plus className="w-4 h-4 mr-2" />
   371→            Nuevo cliente
   372→          </Button>
   373→        </div>
   374→      </div>
   375→
   376→      <Card>
   377→        <CardContent className="p-3">
   378→          <div className="relative">
   379→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   380→            <Input
   381→              placeholder="Buscar por nombre, teléfono o email..."
   382→              value={search}
   383→              onChange={(e) => setSearch(e.target.value)}
   384→              className="pl-9"
   385→            />
   386→          </div>
   387→        </CardContent>
   388→      </Card>
   389→
   390→      <Card>
   391→        <CardContent className="p-0">
   392→          {loading ? (
   393→            <div className="p-8 text-center">
   394→              <Loader2 className="w-5 h-5 animate-spin inline mr-2 text-indigo-600" />
   395→              Cargando clientes...
   396→            </div>
   397→          ) : filtered.length === 0 ? (
   398→            <div className="p-12 text-center">
   399→              <Users className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   400→              <p className="text-sm text-muted-foreground">
   401→                {search
   402→                  ? "No se encontraron clientes"
   403→                  : "Aún no tenés clientes registrados"}
   404→              </p>
   405→            </div>
   406→          ) : (
   407→            <div className="overflow-x-auto">
   408→              <Table>
   409→                <TableHeader>
   410→                  <TableRow>
   411→                    <TableHead>Nombre</TableHead>
   412→                    <TableHead className="hidden sm:table-cell">Contacto</TableHead>
   413→                    <TableHead className="text-center">Compras</TableHead>
   414→                    {loyaltyEnabled && (
   415→                      <TableHead className="text-center">Puntos / Tier</TableHead>
   416→                    )}
   417→                    <TableHead className="text-right">Saldo cta.</TableHead>
   418→                    <TableHead className="text-right">Acciones</TableHead>
   419→                  </TableRow>
   420→                </TableHeader>
   421→                <TableBody>
   422→                  {filtered.map((c) => {
   423→                    const saldo = c.saldo || 0;
   424→                    return (
   425→                      <TableRow key={c.id}>
   426→                        <TableCell>
   427→                          <p className="font-medium">{c.name}</p>
   428→                          {c.notes && (
   429→                            <p className="text-xs text-muted-foreground line-clamp-1">
   430→                              {c.notes}
   431→                            </p>
   432→                          )}
   433→                        </TableCell>
   434→                        <TableCell className="hidden sm:table-cell">
   435→                          <div className="text-sm space-y-0.5">
   436→                            {c.phone && (
   437→                              <p className="flex items-center gap-1">
   438→                                <Phone className="w-3 h-3" />
   439→                                {c.phone}
   440→                              </p>
   441→                            )}
   442→                            {c.email && (
   443→                              <p className="flex items-center gap-1 text-muted-foreground">
   444→                                <Mail className="w-3 h-3" />
   445→                                {c.email}
   446→                              </p>
   447→                            )}
   448→                            {!c.phone && !c.email && (
   449→                              <span className="text-muted-foreground text-xs">—</span>
   450→                            )}
   451→                          </div>
   452→                        </TableCell>
   453→                        <TableCell className="text-center">
   454→                          <Badge variant="outline">{c._count?.sales || 0}</Badge>
   455→                        </TableCell>
   456→                        {loyaltyEnabled && (
   457→                          <TableCell className="text-center">
   458→                            {c.loyaltyTier && (
   459→                              <Badge
   460→                                variant="outline"
   461→                                className={
   462→                                  c.loyaltyTier === "PLATINO"
   463→                                    ? "bg-gray-100 text-gray-800"
   464→                                    : c.loyaltyTier === "ORO"
   465→                                    ? "bg-yellow-100 text-yellow-800"
   466→                                    : c.loyaltyTier === "PLATA"
   467→                                    ? "bg-slate-100 text-slate-700"
   468→                                    : "bg-amber-50 text-amber-800"
   469→                                }
   470→                              >
   471→                                <Star className="w-3 h-3 mr-1" />
   472→                                {c.loyaltyTier}
   473→                              </Badge>
   474→                            )}
   475→                            <p className="text-xs mt-0.5">
   476→                              {Math.floor(c.loyaltyPoints || 0)} pts
   477→                            </p>
   478→                          </TableCell>
   479→                        )}
   480→                        <TableCell className="text-right">
   481→                          {saldo > 0 ? (
   482→                            <Badge className="bg-amber-100 text-amber-800 hover:bg-amber-100">
   483→                              {formatCurrency(saldo, symbol)}
   484→                            </Badge>
   485→                          ) : (
   486→                            <span className="text-xs text-muted-foreground">Saldada</span>
   487→                          )}
   488→                        </TableCell>
   489→                        <TableCell className="text-right">
   490→                          <Button
   491→                            variant="ghost"
   492→                            size="sm"
   493→                            className="h-8"
   494→                            onClick={() => openAccount(c)}
   495→                          >
   496→                            <Wallet className="w-3.5 h-3.5 mr-1" />
   497→                            Cuenta
   498→                          </Button>
   499→                          {loyaltyEnabled && (
   500→                            <Button
   501→                              variant="ghost"
   502→                              size="sm"
   503→                              className="h-8"
   504→                              onClick={() => openPoints(c)}
   505→                            >
   506→                              <Award className="w-3.5 h-3.5 mr-1" />
   507→                              Puntos
   508→                            </Button>
   509→                          )}
   510→                          <Button
   511→                            variant="ghost"
   512→                            size="icon"
   513→                            className="h-8 w-8"
   514→                            onClick={() => openEdit(c)}
   515→                          >
   516→                            <Pencil className="w-3.5 h-3.5" />
   517→                          </Button>
   518→                          <Button
   519→                            variant="ghost"
   520→                            size="icon"
   521→                            className="h-8 w-8 text-red-600"
   522→                            onClick={() => setDeleteId(c.id)}
   523→                          >
   524→                            <Trash2 className="w-3.5 h-3.5" />
   525→                          </Button>
   526→                        </TableCell>
   527→                      </TableRow>
   528→                    );
   529→                  })}
   530→                </TableBody>
   531→              </Table>
   532→            </div>
   533→          )}
   534→        </CardContent>
   535→      </Card>
   536→
   537→      {/* Diálogo cuenta corriente */}
   538→      <Dialog open={accountOpen} onOpenChange={setAccountOpen}>
   539→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   540→          <DialogHeader>
   541→            <DialogTitle className="flex items-center gap-2">
   542→              <ReceiptText className="w-5 h-5 text-indigo-600" />
   543→              Cuenta corriente
   544→            </DialogTitle>
   545→            <DialogDescription>
   546→              {accountData?.customer?.name}
   547→              {accountData?.customer?.phone && ` · ${accountData.customer.phone}`}
   548→            </DialogDescription>
   549→          </DialogHeader>
   550→
   551→          {accountLoading ? (
   552→            <div className="py-12 text-center">
   553→              <Loader2 className="w-6 h-6 animate-spin inline text-indigo-600" />
   554→            </div>
   555→          ) : accountData ? (
   556→            <div className="space-y-4">
   557→              {/* Resumen */}
   558→              <div className="grid grid-cols-3 gap-3">
   559→                <Card className="bg-amber-50 border-amber-200">
   560→                  <CardContent className="p-3">
   561→                    <p className="text-xs text-amber-700">Saldo actual</p>
   562→                    <p className="text-xl font-bold text-amber-800">
   563→                      {formatCurrency(accountData.saldo, symbol)}
   564→                    </p>
   565→                  </CardContent>
   566→                </Card>
   567→                <Card>
   568→                  <CardContent className="p-3">
   569→                    <p className="text-xs text-muted-foreground">Límite</p>
   570→                    <p className="text-xl font-bold">
   571→                      {formatCurrency(accountData.creditLimit, symbol)}
   572→                    </p>
   573→                  </CardContent>
   574→                </Card>
   575→                <Card>
   576→                  <CardContent className="p-3">
   577→                    <p className="text-xs text-muted-foreground">Disponible</p>
   578→                    <p className="text-xl font-bold text-indigo-700">
   579→                      {formatCurrency(
   580→                        accountData.creditLimit > 0
   581→                          ? Math.max(0, accountData.disponible)
   582→                          : 0,
   583→                        symbol
   584→                      )}
   585→                    </p>
   586→                  </CardContent>
   587→                </Card>
   588→              </div>
   589→
   590→              <div className="flex justify-end">
   591→                <Button
   592→                  size="sm"
   593→                  onClick={() => {
   594→                    setPayAmount(accountData.saldo > 0 ? accountData.saldo : 0);
   595→                    setPayMethod("EFECTIVO");
   596→                    setPayNotes("");
   597→                    setPayOpen(true);
   598→                  }}
   599→                  disabled={accountData.saldo <= 0}
   600→                  className="bg-indigo-600 hover:bg-indigo-700"
   601→                >
   602→                  <ArrowDownCircle className="w-4 h-4 mr-2" />
   603→                  Registrar pago
   604→                </Button>
   605→              </div>
   606→
   607→              {/* Movimientos */}
   608→              <div className="border rounded-md">
   609→                <Table>
   610→                  <TableHeader>
   611→                    <TableRow>
   612→                      <TableHead className="w-32">Fecha</TableHead>
   613→                      <TableHead>Tipo</TableHead>
   614→                      <TableHead>Detalle</TableHead>
   615→                      <TableHead className="text-right">Debe</TableHead>
   616→                      <TableHead className="text-right">Haber</TableHead>
   617→                      <TableHead className="text-right">Saldo</TableHead>

   978→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>

   989→            <AlertDialogCancel>Cancelar</AlertDialogCancel>
