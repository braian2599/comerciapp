     1→"use client";
     2→
     3→import { useEffect, useState, useMemo } from "react";
     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";
     8→import { Textarea } from "@/components/ui/textarea";
     9→import { Badge } from "@/components/ui/badge";
    10→import {
    11→  Dialog,
    12→  DialogContent,
    13→  DialogDescription,
    14→  DialogFooter,
    15→  DialogHeader,
    16→  DialogTitle,
    17→  DialogClose,
    18→} from "@/components/ui/dialog";
    19→import {
    20→  Table,
    21→  TableBody,
    22→  TableCell,
    23→  TableHead,
    24→  TableHeader,
    25→  TableRow,
    26→} from "@/components/ui/table";
    27→import {
    28→  AlertDialog,
    29→  AlertDialogAction,
    30→  AlertDialogCancel,
    31→  AlertDialogContent,
    32→  AlertDialogDescription,
    33→  AlertDialogFooter,
    34→  AlertDialogHeader,
    35→  AlertDialogTitle,
    36→} from "@/components/ui/alert-dialog";
    37→import {
    38→  Select,
    39→  SelectContent,
    40→  SelectItem,
    41→  SelectTrigger,
    42→  SelectValue,
    43→} from "@/components/ui/select";
    44→import { toast } from "sonner";
    45→import {
    46→  Plus,
    47→  Search,
    48→  Pencil,
    49→  Trash2,
    50→  Users,
    51→  Phone,
    52→  Mail,
    53→  Loader2,
    54→  Wallet,
    55→  ReceiptText,
    56→  ArrowDownCircle,
    57→  ArrowUpCircle,
    58→  Star,
    59→  Award,
    60→  Gift,
    61→} from "lucide-react";
    62→import { useAppStore } from "@/store/app-store";
    63→import { formatCurrency, formatDateTime, PAYMENT_METHOD_TYPES } from "@/lib/constants";
    64→import { safeFetchJSON, safeFetchArray } from "@/lib/fetch";
    65→
    66→interface Customer {
    67→  id: string;
    68→  name: string;
    69→  phone?: string;
    70→  email?: string;
    71→  address?: string;
    72→  notes?: string;
    73→  creditLimit?: number;
    74→  saldo?: number;
    75→  _count?: { sales: number };
    76→  loyaltyPoints?: number;
    77→  loyaltyTier?: string;
    78→  totalSpent?: number;
    79→  totalSales?: number;
    80→}
    81→
    82→interface LedgerItem {
    83→  id: string;
    84→  date: string;
    85→  type: "DEBE" | "HABER";
    86→  description: string;
    87→  amount: number;
    88→  user?: string;
    89→  balanceAfter: number;
    90→}
    91→
    92→interface AccountState {
    93→  customer: Customer;
    94→  saldo: number;
    95→  creditLimit: number;
    96→  disponible: number;
    97→  movimientos: LedgerItem[];
    98→  totalVentas: number;
    99→  totalPagos: number;
   100→}
   101→
   102→const emptyForm = {
   103→  id: "",
   104→  name: "",
   105→  phone: "",
   106→  email: "",
   107→  address: "",
   108→  notes: "",
   109→  creditLimit: 0,
   110→};
   111→
   112→export function CustomersView() {
   113→  const { store } = useAppStore();
   114→  const symbol = store?.currencySymbol || "$";
   115→  const [customers, setCustomers] = useState<Customer[]>([]);
   116→  const [loading, setLoading] = useState(true);
   117→  const [search, setSearch] = useState("");
   118→  const [formOpen, setFormOpen] = useState(false);
   119→  const [form, setForm] = useState<any>(emptyForm);
   120→  const [saving, setSaving] = useState(false);
   121→  const [deleteId, setDeleteId] = useState<string | null>(null);
   122→
   123→  // Cuenta corriente
   124→  const [accountOpen, setAccountOpen] = useState(false);
   125→  const [accountData, setAccountData] = useState<AccountState | null>(null);
   126→  const [accountLoading, setAccountLoading] = useState(false);
   127→  const [payOpen, setPayOpen] = useState(false);
   128→  const [payAmount, setPayAmount] = useState(0);
   129→  const [payMethod, setPayMethod] = useState("EFECTIVO");
   130→  const [payNotes, setPayNotes] = useState("");
   131→  const [paySaving, setPaySaving] = useState(false);
   132→
   133→  // Fidelización - puntos
   134→  const [pointsOpen, setPointsOpen] = useState(false);
   135→  const [pointsData, setPointsData] = useState<any | null>(null);
   136→  const [pointsLoading, setPointsLoading] = useState(false);
   137→  const [adjustPoints, setAdjustPoints] = useState(0);
   138→  const [adjustNotes, setAdjustNotes] = useState("");
   139→  const [adjustSaving, setAdjustSaving] = useState(false);
   140→
   141→  // Programa de fidelización (cargado una vez)
   142→  const [loyaltyEnabled, setLoyaltyEnabled] = useState(false);
   143→
   144→  async function load() {
   145→    setLoading(true);
   146→    try {
   147→      const [custRes, loyRes] = await Promise.all([
   148→        safeFetchArray<Customer>("/api/customers"),
   149→        safeFetchJSON<any>("/api/loyalty"),
   150→      ]);
   151→      setCustomers(custRes);
   152→      const loy = loyRes.data;
   153→      setLoyaltyEnabled(!!loy?.enabled && typeof loy === "object" && !Array.isArray(loy));
   154→    } catch {
   155→      toast.error("No se pudieron cargar los clientes");
   156→      setCustomers([]);
   157→    } finally {
   158→      setLoading(false);
   159→    }
   160→  }
   161→
   162→  useEffect(() => {
   163→    load();
   164→  }, []);
   165→
   166→  const filtered = useMemo(() => {
   167→    return customers.filter((c) => {
   168→      if (!search) return true;
   169→      const s = search.toLowerCase();
   170→      return (
   171→        c.name.toLowerCase().includes(s) ||
   172→        c.phone?.toLowerCase().includes(s) ||
   173→        c.email?.toLowerCase().includes(s)
   174→      );
   175→    });
   176→  }, [customers, search]);
   177→
   178→  const saldoTotal = customers.reduce((s, c) => s + (c.saldo || 0), 0);
   179→
   180→  function openNew() {
   181→    setForm({ ...emptyForm });
   182→    setFormOpen(true);
   183→  }
   184→
   185→  function openEdit(c: Customer) {
   186→    setForm({ ...c, creditLimit: c.creditLimit || 0 });
   187→    setFormOpen(true);
   188→  }
   189→
   190→  async function openAccount(c: Customer) {
   191→    setAccountOpen(true);
   192→    setAccountLoading(true);
   193→    setAccountData(null);
   194→    try {
   195→      const res = await fetch(`/api/customers/account?customerId=${c.id}`);
   196→      const data = await res.json();
   197→      setAccountData(data);
   198→    } catch {
   199→      toast.error("Error al cargar cuenta");
   200→    } finally {
   201→      setAccountLoading(false);
   202→    }
   203→  }
   204→
   205→  async function openPoints(c: Customer) {
   206→    setPointsOpen(true);
   207→    setPointsLoading(true);
   208→    setPointsData(null);
   209→    setAdjustPoints(0);
   210→    setAdjustNotes("");
   211→    try {
   212→      const res = await fetch(`/api/loyalty/points?customerId=${c.id}`);
   213→      const data = await res.json();
   214→      setPointsData(data);
   215→    } catch {
   216→      toast.error("Error al cargar puntos");
   217→    } finally {
   218→      setPointsLoading(false);
   219→    }
   220→  }
   221→
   222→  async function handleAdjustPoints() {
   223→    if (!pointsData?.customer?.id) return;
   224→    if (adjustPoints === 0) {
   225→      toast.error("Indica puntos (positivo o negativo)");
   226→      return;
   227→    }
   228→    setAdjustSaving(true);
   229→    try {
   230→      const res = await fetch("/api/loyalty/points", {
   231→        method: "POST",
   232→        headers: { "Content-Type": "application/json" },
   233→        body: JSON.stringify({
   234→          customerId: pointsData.customer.id,
   235→          points: adjustPoints,
   236→          description: adjustNotes,
   237→        }),
   238→      });
   239→      const data = await res.json();
   240→      if (!res.ok) throw new Error(data.error);
   241→      toast.success("Puntos ajustados");
   242→      // Recargar
   243→      await openPoints(pointsData.customer);
   244→      load();
   245→    } catch (e: any) {
   246→      toast.error(e.message);
   247→    } finally {
   248→      setAdjustSaving(false);
   249→    }
   250→  }
   251→
   252→  async function handleSave() {
   253→    if (!form.name) {
   254→      toast.error("El nombre es obligatorio");
   255→      return;
   256→    }
   257→    setSaving(true);
   258→    try {
   259→      const method = form.id ? "PUT" : "POST";
   260→      const res = await fetch("/api/customers", {
   261→        method,
   262→        headers: { "Content-Type": "application/json" },
   263→        body: JSON.stringify(form),
   264→      });
   265→      const data = await res.json();
   266→      if (!res.ok) throw new Error(data.error);
   267→      toast.success(form.id ? "Cliente actualizado" : "Cliente creado");
   268→      setFormOpen(false);
   269→      load();
   270→    } catch (e: any) {
   271→      toast.error(e.message);
   272→    } finally {
   273→      setSaving(false);
   274→    }
   275→  }
   276→
   277→  async function handleDelete() {
   278→    if (!deleteId) return;
   279→    try {
   280→      const res = await fetch(`/api/customers?id=${deleteId}`, { method: "DELETE" });
   281→      const data = await res.json();
   282→      if (!res.ok) throw new Error(data.error);
   283→      toast.success("Cliente eliminado");
   284→      setDeleteId(null);
   285→      load();
   286→    } catch (e: any) {
   287→      toast.error(e.message);
   288→    }
   289→  }
   290→
   291→  async function handlePayment() {
   292→    if (!accountData) return;
   293→    if (payAmount <= 0) {
   294→      toast.error("El monto debe ser mayor a 0");
   295→      return;
   296→    }
   297→    setPaySaving(true);
   298→    try {
   299→      const res = await fetch("/api/customers/account", {
   300→        method: "POST",
   301→        headers: { "Content-Type": "application/json" },
   302→        body: JSON.stringify({
   303→          customerId: accountData.customer.id,
   304→          amount: payAmount,
   305→          paymentMethod: payMethod,
   306→          notes: payNotes,
   307→        }),
   308→      });
   309→      const data = await res.json();
   310→      if (!res.ok) throw new Error(data.error);
   311→      toast.success("Pago registrado");
   312→      setPayOpen(false);
   313→      setPayAmount(0);
   314→      setPayNotes("");
   315→      // Recargar cuenta y lista
   316→      await openAccount(accountData.customer);
   317→      load();
   318→    } catch (e: any) {
   319→      toast.error(e.message);
   320→    } finally {
   321→      setPaySaving(false);
   322→    }
   323→  }
   324→
   325→  return (
   326→    <div className="space-y-4">
   327→      <div className="flex items-center justify-between flex-wrap gap-3">
   328→        <div>
   329→          <h1 className="text-2xl font-bold tracking-tight">Clientes</h1>
   330→          <p className="text-sm text-muted-foreground">
   331→            {customers.length} clientes · Saldo total en cuentas:{" "}
   332→            <span className={saldoTotal > 0 ? "text-amber-700 font-medium" : ""}>
   333→              {formatCurrency(saldoTotal, symbol)}
   334→            </span>
   335→          </p>
   336→        </div>
   337→        <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   338→          <Plus className="w-4 h-4 mr-2" />
   339→          Nuevo cliente
   340→        </Button>
   341→      </div>
   342→
   343→      <Card>
   344→        <CardContent className="p-3">
   345→          <div className="relative">
   346→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   347→            <Input
   348→              placeholder="Buscar por nombre, teléfono o email..."
   349→              value={search}
   350→              onChange={(e) => setSearch(e.target.value)}
   351→              className="pl-9"
   352→            />
   353→          </div>
   354→        </CardContent>
   355→      </Card>
   356→
   357→      <Card>
   358→        <CardContent className="p-0">
   359→          {loading ? (
   360→            <div className="p-8 text-center">
   361→              <Loader2 className="w-5 h-5 animate-spin inline mr-2 text-indigo-600" />
   362→              Cargando clientes...
   363→            </div>
   364→          ) : filtered.length === 0 ? (
   365→            <div className="p-12 text-center">
   366→              <Users className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   367→              <p className="text-sm text-muted-foreground">
   368→                {search
   369→                  ? "No se encontraron clientes"
   370→                  : "Aún no tenés clientes registrados"}
   371→              </p>
   372→            </div>
   373→          ) : (
   374→            <div className="overflow-x-auto">
   375→              <Table>
   376→                <TableHeader>
   377→                  <TableRow>
   378→                    <TableHead>Nombre</TableHead>
   379→                    <TableHead className="hidden sm:table-cell">Contacto</TableHead>
   380→                    <TableHead className="text-center">Compras</TableHead>
   381→                    {loyaltyEnabled && (
   382→                      <TableHead className="text-center">Puntos / Tier</TableHead>
   383→                    )}
   384→                    <TableHead className="text-right">Saldo cta.</TableHead>
   385→                    <TableHead className="text-right">Acciones</TableHead>
   386→                  </TableRow>
   387→                </TableHeader>
   388→                <TableBody>
   389→                  {filtered.map((c) => {
   390→                    const saldo = c.saldo || 0;
   391→                    return (
   392→                      <TableRow key={c.id}>
   393→                        <TableCell>
   394→                          <p className="font-medium">{c.name}</p>
   395→                          {c.notes && (
   396→                            <p className="text-xs text-muted-foreground line-clamp-1">
   397→                              {c.notes}
   398→                            </p>
   399→                          )}
   400→                        </TableCell>
   401→                        <TableCell className="hidden sm:table-cell">
   402→                          <div className="text-sm space-y-0.5">
   403→                            {c.phone && (
   404→                              <p className="flex items-center gap-1">
   405→                                <Phone className="w-3 h-3" />
   406→                                {c.phone}
   407→                              </p>
   408→                            )}
   409→                            {c.email && (
   410→                              <p className="flex items-center gap-1 text-muted-foreground">
   411→                                <Mail className="w-3 h-3" />
   412→                                {c.email}
   413→                              </p>
   414→                            )}
   415→                            {!c.phone && !c.email && (
   416→                              <span className="text-muted-foreground text-xs">—</span>
   417→                            )}
   418→                          </div>
   419→                        </TableCell>
   420→                        <TableCell className="text-center">
   421→                          <Badge variant="outline">{c._count?.sales || 0}</Badge>
   422→                        </TableCell>
   423→                        {loyaltyEnabled && (
   424→                          <TableCell className="text-center">
   425→                            {c.loyaltyTier && (
   426→                              <Badge
   427→                                variant="outline"
   428→                                className={
   429→                                  c.loyaltyTier === "PLATINO"
   430→                                    ? "bg-gray-100 text-gray-800"
   431→                                    : c.loyaltyTier === "ORO"
   432→                                    ? "bg-yellow-100 text-yellow-800"
   433→                                    : c.loyaltyTier === "PLATA"
   434→                                    ? "bg-slate-100 text-slate-700"
   435→                                    : "bg-amber-50 text-amber-800"
   436→                                }
   437→                              >
   438→                                <Star className="w-3 h-3 mr-1" />
   439→                                {c.loyaltyTier}
   440→                              </Badge>
   441→                            )}
   442→                            <p className="text-xs mt-0.5">
   443→                              {Math.floor(c.loyaltyPoints || 0)} pts
   444→                            </p>
   445→                          </TableCell>
   446→                        )}
   447→                        <TableCell className="text-right">
   448→                          {saldo > 0 ? (
   449→                            <Badge className="bg-amber-100 text-amber-800 hover:bg-amber-100">
   450→                              {formatCurrency(saldo, symbol)}
   451→                            </Badge>
   452→                          ) : (
   453→                            <span className="text-xs text-muted-foreground">Saldada</span>
   454→                          )}
   455→                        </TableCell>
   456→                        <TableCell className="text-right">
   457→                          <Button
   458→                            variant="ghost"
   459→                            size="sm"
   460→                            className="h-8"
   461→                            onClick={() => openAccount(c)}
   462→                          >
   463→                            <Wallet className="w-3.5 h-3.5 mr-1" />
   464→                            Cuenta
   465→                          </Button>
   466→                          {loyaltyEnabled && (
   467→                            <Button
   468→                              variant="ghost"
   469→                              size="sm"
   470→                              className="h-8"
   471→                              onClick={() => openPoints(c)}
   472→                            >
   473→                              <Award className="w-3.5 h-3.5 mr-1" />
   474→                              Puntos
   475→                            </Button>
   476→                          )}
   477→                          <Button
   478→                            variant="ghost"
   479→                            size="icon"
   480→                            className="h-8 w-8"
   481→                            onClick={() => openEdit(c)}
   482→                          >
   483→                            <Pencil className="w-3.5 h-3.5" />
   484→                          </Button>
   485→                          <Button
   486→                            variant="ghost"
   487→                            size="icon"
   488→                            className="h-8 w-8 text-red-600"
   489→                            onClick={() => setDeleteId(c.id)}
   490→                          >
   491→                            <Trash2 className="w-3.5 h-3.5" />
   492→                          </Button>
   493→                        </TableCell>
   494→                      </TableRow>
   495→                    );
   496→                  })}
   497→                </TableBody>
   498→              </Table>
   499→            </div>
   500→          )}
   501→        </CardContent>
   502→      </Card>
   503→
   504→      {/* Diálogo cuenta corriente */}
   505→      <Dialog open={accountOpen} onOpenChange={setAccountOpen}>
   506→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   507→          <DialogHeader>
   508→            <DialogTitle className="flex items-center gap-2">
   509→              <ReceiptText className="w-5 h-5 text-indigo-600" />
   510→              Cuenta corriente
   511→            </DialogTitle>
   512→            <DialogDescription>
   513→              {accountData?.customer?.name}
   514→              {accountData?.customer?.phone && ` · ${accountData.customer.phone}`}
   515→            </DialogDescription>
   516→          </DialogHeader>
   517→
   518→          {accountLoading ? (
   519→            <div className="py-12 text-center">
   520→              <Loader2 className="w-6 h-6 animate-spin inline text-indigo-600" />
   521→            </div>
   522→          ) : accountData ? (
   523→            <div className="space-y-4">
   524→              {/* Resumen */}
   525→              <div className="grid grid-cols-3 gap-3">
   526→                <Card className="bg-amber-50 border-amber-200">
   527→                  <CardContent className="p-3">
   528→                    <p className="text-xs text-amber-700">Saldo actual</p>
   529→                    <p className="text-xl font-bold text-amber-800">
   530→                      {formatCurrency(accountData.saldo, symbol)}
   531→                    </p>
   532→                  </CardContent>
   533→                </Card>
   534→                <Card>
   535→                  <CardContent className="p-3">
   536→                    <p className="text-xs text-muted-foreground">Límite</p>
   537→                    <p className="text-xl font-bold">
   538→                      {formatCurrency(accountData.creditLimit, symbol)}
   539→                    </p>
   540→                  </CardContent>
   541→                </Card>
   542→                <Card>
   543→                  <CardContent className="p-3">
   544→                    <p className="text-xs text-muted-foreground">Disponible</p>
   545→                    <p className="text-xl font-bold text-indigo-700">
   546→                      {formatCurrency(
   547→                        accountData.creditLimit > 0
   548→                          ? Math.max(0, accountData.disponible)
   549→                          : 0,
   550→                        symbol
   551→                      )}
   552→                    </p>
   553→                  </CardContent>
   554→                </Card>
   555→              </div>
   556→
   557→              <div className="flex justify-end">
   558→                <Button
   559→                  size="sm"
   560→                  onClick={() => {
   561→                    setPayAmount(accountData.saldo > 0 ? accountData.saldo : 0);
   562→                    setPayMethod("EFECTIVO");
   563→                    setPayNotes("");
   564→                    setPayOpen(true);
   565→                  }}
   566→                  disabled={accountData.saldo <= 0}
   567→                  className="bg-indigo-600 hover:bg-indigo-700"
   568→                >
   569→                  <ArrowDownCircle className="w-4 h-4 mr-2" />
   570→                  Registrar pago
   571→                </Button>
   572→              </div>
   573→
   574→              {/* Movimientos */}
   575→              <div className="border rounded-md">
   576→                <Table>
   577→                  <TableHeader>
   578→                    <TableRow>
   579→                      <TableHead className="w-32">Fecha</TableHead>
   580→                      <TableHead>Tipo</TableHead>
   581→                      <TableHead>Detalle</TableHead>
   582→                      <TableHead className="text-right">Debe</TableHead>
   583→                      <TableHead className="text-right">Haber</TableHead>
   584→                      <TableHead className="text-right">Saldo</TableHead>
   585→                    </TableRow>
   586→                  </TableHeader>
   587→                  <TableBody>
   588→                    {accountData.movimientos.length === 0 ? (
   589→                      <TableRow>
   590→                        <TableCell colSpan={6} className="text-center text-muted-foreground py-6">
   591→                          Sin movimientos
   592→                        </TableCell>
   593→                      </TableRow>
   594→                    ) : (
   595→                      accountData.movimientos.map((m) => (
   596→                        <TableRow key={m.id + m.type}>
   597→                          <TableCell className="text-xs">
   598→                            {formatDateTime(m.date)}
   599→                          </TableCell>
   600→                          <TableCell>
   601→                            {m.type === "DEBE" ? (
   602→                              <Badge variant="outline" className="bg-red-50 text-red-700 border-red-200">
   603→                                <ArrowUpCircle className="w-3 h-3 mr-1" />
   604→                                Venta
   605→                              </Badge>
   606→                            ) : (
   607→                              <Badge variant="outline" className="bg-indigo-50 text-indigo-700 border-indigo-200">
   608→                                <ArrowDownCircle className="w-3 h-3 mr-1" />
   609→                                Pago
