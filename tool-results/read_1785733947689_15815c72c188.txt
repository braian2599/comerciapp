    68→  name: string;

    75→  salePrice: number;
    76→  stock: number;

    83→  allergens?: string;
    84→  imageUrl?: string;

    87→interface Category {
    88→  id: string;
    89→  name: string;
    90→  _count?: { products: number };

    93→const emptyForm = {
    94→  id: "",
    95→  name: "",
    96→  description: "",
    97→  barcode: "",
    98→  sku: "",
    99→  categoryId: "",
   100→  costPrice: 0,
   101→  salePrice: 0,
   102→  stock: 0,
   103→  minStock: 5,
   104→  unit: "UNIDAD",
   105→  active: true,
   106→  brand: "",
   107→  labels: "",
   108→  ingredients: "",
   109→  allergens: "",
   110→  imageUrl: "",
   111→};
   112→
   113→export function ProductsView() {
   114→  const { store } = useAppStore();
   115→  const [products, setProducts] = useState<Product[]>([]);
   116→  const [categories, setCategories] = useState<Category[]>([]);
   117→  const [loading, setLoading] = useState(true);
   118→  const [search, setSearch] = useState("");
   119→  const [filterCat, setFilterCat] = useState("all");
   120→  const [filterStock, setFilterStock] = useState("all");
   121→
   122→  const [formOpen, setFormOpen] = useState(false);
   123→  const [form, setForm] = useState<any>(emptyForm);
   124→  const [saving, setSaving] = useState(false);
   125→  const [deleteId, setDeleteId] = useState<string | null>(null);
   126→  const [catDialogOpen, setCatDialogOpen] = useState(false);
   127→  const [newCatName, setNewCatName] = useState("");
   128→
   129→  // Lookup por código de barras
   130→  const [lookupLoading, setLookupLoading] = useState(false);
   131→  const [lookupResult, setLookupResult] = useState<
   132→    | {
   133→        status: "idle" | "loading" | "found" | "notfound" | "error";
   134→        source?: string;
   135→        imageUrl?: string;
   136→        labels?: string[];
   137→        allergens?: string[];
   138→      }
   139→    | null
   140→  >(null);
   141→  // Sección colapsable de datos adicionales
   142→  const [showExtraFields, setShowExtraFields] = useState(false);
   143→  // Etiquetas y alérgenos editados como chips
   144→  const [labelChips, setLabelChips] = useState<string[]>([]);
   145→  const [allergenChips, setAllergenChips] = useState<string[]>([]);
   146→  const [labelInput, setLabelInput] = useState("");
   147→  const [allergenInput, setAllergenInput] = useState("");
   148→
   149→  async function handleBarcodeLookup() {
   150→    const code = (form.barcode || "").trim();
   151→    if (!code) {
   152→      toast.error("Ingresá un código de barras primero");
   153→      return;
   154→    }
   155→    setLookupLoading(true);
   156→    setLookupResult({ status: "loading" });
   157→    try {
   158→      const res = await fetch(`/api/products/lookup?barcode=${encodeURIComponent(code)}`);
   159→      const data = await res.json();
   160→      if (data.found) {
   161→        // Solo autocompletamos campos que el usuario aún no llenó.
   162→        setForm((prev: any) => ({
   163→          ...prev,
   164→          name: prev.name || data.name || "",
   165→          description: prev.description || data.description || data.brand || "",
   166→          barcode: prev.barcode || code,
   167→          brand: prev.brand || data.brand || "",
   168→          ingredients: prev.ingredients || data.ingredients || "",
   169→          imageUrl: prev.imageUrl || data.imageUrl || "",
   170→        }));
   171→        // Merge de etiquetas y alérgenos (sin duplicar)
   172→        if (Array.isArray(data.labels) && data.labels.length > 0) {
   173→          setLabelChips((prev) =>
   174→            Array.from(new Set([...prev, ...data.labels])).slice(0, 15)
   175→          );
   176→          setShowExtraFields(true);
   177→        }
   178→        if (Array.isArray(data.allergens) && data.allergens.length > 0) {
   179→          setAllergenChips((prev) =>
   180→            Array.from(new Set([...prev, ...data.allergens])).slice(0, 15)
   181→          );
   182→          setShowExtraFields(true);
   183→        }
   184→        setLookupResult({
   185→          status: "found",
   186→          source: data.source,
   187→          imageUrl: data.imageUrl,
   188→          labels: data.labels,
   189→          allergens: data.allergens,
   190→        });
   191→        const sourceLabel =
   192→          data.source === "openfoodfacts" ? "Open Food Facts" : "UPC Item DB";
   193→        toast.success(`Producto encontrado en ${sourceLabel}`, {
   194→          description: data.name,
   195→        });
   196→      } else {
   197→        setLookupResult({ status: "notfound" });
   198→        toast.info("No se encontró el código en las bases de datos públicas", {
   199→          description: "Completá los datos manualmente",
   200→        });
   201→      }
   202→    } catch (e: any) {
   203→      setLookupResult({ status: "error" });
   204→      toast.error("No se pudo consultar la base de productos");
   205→    } finally {
   206→      setLookupLoading(false);
   207→    }
   208→  }
   209→
   210→  function resetLookup() {
   211→    setLookupResult(null);
   212→  }
   213→
   214→  const symbol = store?.currencySymbol || "$";
   215→
   216→  async function load() {
   217→    setLoading(true);
   218→    const [p, c] = await Promise.all([
   219→      fetch("/api/products").then((r) => r.json()),
   220→      fetch("/api/categories").then((r) => r.json()),
   221→    ]);
   222→    setProducts(p);
   223→    setCategories(c);
   224→    setLoading(false);
   225→  }
   226→
   227→  useEffect(() => {
   228→    load();
   229→  }, []);
   230→
   231→  const filtered = useMemo(() => {
   232→    return products.filter((p) => {
   233→      const matchSearch =
   234→        !search ||
   235→        p.name.toLowerCase().includes(search.toLowerCase()) ||
   236→        p.barcode?.includes(search) ||
   237→        p.sku?.toLowerCase().includes(search.toLowerCase());
   238→      const matchCat =
   239→        filterCat === "all" || p.categoryId === filterCat;
   240→      const matchStock =
   241→        filterStock === "all" ||
   242→        (filterStock === "low" && p.stock <= p.minStock) ||
   243→        (filterStock === "out" && p.stock <= 0) ||
   244→        (filterStock === "ok" && p.stock > p.minStock);
   245→      return matchSearch && matchCat && matchStock;
   246→    });
   247→  }, [products, search, filterCat, filterStock]);
   248→
   249→  function openNew() {
   250→    setForm({ ...emptyForm });
   251→    resetLookup();
   252→    setLabelChips([]);
   253→    setAllergenChips([]);
   254→    setShowExtraFields(false);
   255→    setFormOpen(true);
   256→  }
   257→
   258→  function openEdit(p: Product) {
   259→    setForm({
   260→      ...p,
   261→      costPrice: p.costPrice,
   262→      salePrice: p.salePrice,
   263→      stock: p.stock,
   264→      minStock: p.minStock,
   265→      adjustReason: "",
   266→    });
   267→    resetLookup();
   268→    setLabelChips(p.labels ? p.labels.split(",").map((s) => s.trim()).filter(Boolean) : []);
   269→    setAllergenChips(p.allergens ? p.allergens.split(",").map((s) => s.trim()).filter(Boolean) : []);
   270→    setShowExtraFields(
   271→      !!(p.brand || p.ingredients || p.labels || p.allergens || p.imageUrl)
   272→    );
   273→    setFormOpen(true);
   274→  }
   275→
   276→  async function handleSave() {
   277→    if (!form.name || !form.salePrice) {
   278→      toast.error("Nombre y precio de venta son obligatorios");
   279→      return;
   280→    }
   281→    setSaving(true);
   282→    try {
   283→      const method = form.id ? "PUT" : "POST";
   284→      const payload = {
   285→        ...form,
   286→        labels: labelChips.join(","),
   287→        allergens: allergenChips.join(","),
   288→      };
   289→      const res = await fetch("/api/products", {
   290→        method,
   291→        headers: { "Content-Type": "application/json" },
   292→        body: JSON.stringify(payload),
   293→      });
   294→      const data = await res.json();
   295→      if (!res.ok) throw new Error(data.error);
   296→      toast.success(form.id ? "Producto actualizado" : "Producto creado");
   297→      setFormOpen(false);
   298→      load();
   299→    } catch (e: any) {
   300→      toast.error(e.message);
   301→    } finally {
   302→      setSaving(false);
   303→    }
   304→  }
   305→
   306→  async function handleDelete() {
   307→    if (!deleteId) return;
   308→    try {
   309→      const res = await fetch(`/api/products?id=${deleteId}`, { method: "DELETE" });
   310→      const data = await res.json();
   311→      if (!res.ok) throw new Error(data.error);
   312→      toast.success("Producto desactivado");
   313→      setDeleteId(null);
   314→      load();
   315→    } catch (e: any) {
   316→      toast.error(e.message);
   317→    }
   318→  }
   319→
   320→  async function handleAddCategory() {
   321→    if (!newCatName.trim()) return;
   322→    try {
   323→      const res = await fetch("/api/categories", {
   324→        method: "POST",
   325→        headers: { "Content-Type": "application/json" },
   326→        body: JSON.stringify({ name: newCatName }),
   327→      });
   328→      const data = await res.json();
   329→      if (!res.ok) throw new Error(data.error);
   330→      toast.success("Categoría creada");
   331→      setNewCatName("");
   332→      setCatDialogOpen(false);
   333→      load();
   334→    } catch (e: any) {
   335→      toast.error(e.message);
   336→    }
   337→  }
   338→
   339→  async function handleDeleteCategory(id: string, name: string) {
   340→    if (!confirm(`¿Eliminar la categoría "${name}"? Los productos quedarán sin categoría.`)) return;
   341→    try {
   342→      const res = await fetch(`/api/categories?id=${id}`, { method: "DELETE" });
   343→      const data = await res.json();
   344→      if (!res.ok) throw new Error(data.error);
   345→      toast.success("Categoría eliminada");
   346→      load();
   347→    } catch (e: any) {
   348→      toast.error(e.message);
   349→    }
   350→  }
   351→
   352→  return (
   353→    <div className="space-y-4">
   354→      <div className="flex items-center justify-between flex-wrap gap-3">
   355→        <div>
   356→          <h1 className="text-2xl font-bold tracking-tight">Productos</h1>
   357→          <p className="text-sm text-muted-foreground">
   358→            {products.length} productos · {categories.length} categorías
   359→          </p>
   360→        </div>
   361→        <div className="flex gap-2">
   362→          <Dialog open={catDialogOpen} onOpenChange={setCatDialogOpen}>
   363→            <DialogTrigger asChild>
   364→              <Button variant="outline">
   365→                <Tag className="w-4 h-4 mr-2" />
   366→                Categorías
   367→              </Button>
   368→            </DialogTrigger>
   369→            <DialogContent>
   370→              <DialogHeader>
   371→                <DialogTitle>Gestionar categorías</DialogTitle>
   372→                <DialogDescription>
   373→                  Agregá o eliminá categorías para organizar tus productos
   374→                </DialogDescription>
   375→              </DialogHeader>
   376→              <div className="flex gap-2">
   377→                <Input
   378→                  placeholder="Nueva categoría..."
   379→                  value={newCatName}
   380→                  onChange={(e) => setNewCatName(e.target.value)}
   381→                  onKeyDown={(e) => e.key === "Enter" && handleAddCategory()}
   382→                />
   383→                <Button onClick={handleAddCategory} className="bg-indigo-600 hover:bg-indigo-700">
   384→                  <Plus className="w-4 h-4" />
   385→                </Button>
   386→              </div>
   387→              <div className="max-h-60 overflow-y-auto space-y-1 mt-2">
   388→                {categories.length === 0 ? (
   389→                  <p className="text-sm text-muted-foreground text-center py-4">
   390→                    Sin categorías
   391→                  </p>
   392→                ) : (
   393→                  categories.map((c) => (
   394→                    <div
   395→                      key={c.id}
   396→                      className="flex items-center justify-between p-2 rounded-md bg-muted/50"
   397→                    >
   398→                      <div>
   399→                        <p className="text-sm font-medium">{c.name}</p>
   400→                        <p className="text-xs text-muted-foreground">
   401→                          {c._count?.products || 0} productos
   402→                        </p>
   403→                      </div>
   404→                      <Button
   405→                        variant="ghost"
   406→                        size="icon"
   407→                        className="h-7 w-7 text-red-600"
   408→                        onClick={() => handleDeleteCategory(c.id, c.name)}
   409→                      >
   410→                        <Trash2 className="w-3.5 h-3.5" />
   411→                      </Button>
   412→                    </div>
   413→                  ))
   414→                )}
   415→              </div>
   416→            </DialogContent>
   417→          </Dialog>
   418→
   419→          <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   420→            <Plus className="w-4 h-4 mr-2" />
   421→            Nuevo producto
   422→          </Button>
   423→        </div>
   424→      </div>
   425→
   426→      {/* Filtros */}
   427→      <Card>
   428→        <CardContent className="p-3">
   429→          <div className="grid gap-2 grid-cols-1 sm:grid-cols-3">
   430→            <div className="relative">
   431→              <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   432→              <Input
   433→                placeholder="Buscar por nombre, código o SKU..."
   434→                value={search}
   435→                onChange={(e) => setSearch(e.target.value)}
   436→                className="pl-9"
   437→              />
   438→            </div>
   439→            <Select value={filterCat} onValueChange={setFilterCat}>
   440→              <SelectTrigger>
   441→                <SelectValue placeholder="Todas las categorías" />
   442→              </SelectTrigger>
   443→              <SelectContent>
   444→                <SelectItem value="all">Todas las categorías</SelectItem>
   445→                {categories.map((c) => (
   446→                  <SelectItem key={c.id} value={c.id}>
   447→                    {c.name}
   448→                  </SelectItem>
   449→                ))}
   450→              </SelectContent>
   451→            </Select>
   452→            <Select value={filterStock} onValueChange={setFilterStock}>
   453→              <SelectTrigger>
   454→                <SelectValue />
   455→              </SelectTrigger>
   456→              <SelectContent>
   457→                <SelectItem value="all">Todo el stock</SelectItem>
   458→                <SelectItem value="ok">Stock OK</SelectItem>
   459→                <SelectItem value="low">Stock bajo</SelectItem>
   460→                <SelectItem value="out">Sin stock</SelectItem>
   461→              </SelectContent>
   462→            </Select>
   463→          </div>
   464→        </CardContent>
   465→      </Card>
   466→
   467→      {/* Tabla */}
   468→      <Card>
   469→        <CardContent className="p-0">
   470→          {loading ? (
   471→            <div className="p-8 text-center text-muted-foreground">
   472→              <Loader2 className="w-5 h-5 animate-spin inline mr-2" />
   473→              Cargando productos...
   474→            </div>
   475→          ) : filtered.length === 0 ? (
   476→            <div className="p-12 text-center">
   477→              <Package className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   478→              <p className="text-sm text-muted-foreground">
   479→                {search || filterCat !== "all" || filterStock !== "all"
   480→                  ? "No se encontraron productos con esos filtros"
   481→                  : "Aún no tenés productos. Creá el primero!"}
   482→              </p>
   483→            </div>
   484→          ) : (
   485→            <div className="overflow-x-auto">
   486→              <Table>
   487→                <TableHeader>
   488→                  <TableRow>
   489→                    <TableHead>Nombre</TableHead>
   490→                    <TableHead className="hidden md:table-cell">Categoría</TableHead>
   491→                    <TableHead className="text-right">Costo</TableHead>
   492→                    <TableHead className="text-right">Precio</TableHead>
   493→                    <TableHead className="text-right">Stock</TableHead>
   494→                    <TableHead className="text-center">Estado</TableHead>
   495→                    <TableHead className="text-right">Acciones</TableHead>
   496→                  </TableRow>
   497→                </TableHeader>
   498→                <TableBody>
   499→                  {filtered.map((p) => (
   500→                    <TableRow key={p.id} className={!p.active ? "opacity-50" : ""}>
   501→                      <TableCell>
   502→                        <div>
   503→                          <p className="font-medium">{p.name}</p>
   504→                          {(p.barcode || p.sku) && (
   505→                            <p className="text-xs text-muted-foreground">
   506→                              {p.barcode && `Cód: ${p.barcode}`}
   507→                              {p.barcode && p.sku && " · "}
   508→                              {p.sku && `SKU: ${p.sku}`}
   509→                            </p>
   510→                          )}
   511→                        </div>
   512→                      </TableCell>
   513→                      <TableCell className="hidden md:table-cell">
   514→                        {p.category ? (
   515→                          <Badge variant="outline">{p.category.name}</Badge>
   516→                        ) : (
   517→                          <span className="text-xs text-muted-foreground">—</span>
   518→                        )}
   519→                      </TableCell>
   520→                      <TableCell className="text-right text-muted-foreground">
   521→                        {formatCurrency(p.costPrice, symbol)}
   522→                      </TableCell>
   523→                      <TableCell className="text-right font-semibold">
   524→                        {formatCurrency(p.salePrice, symbol)}
   525→                      </TableCell>
   526→                      <TableCell className="text-right">
   527→                        <span
   528→                          className={
   529→                            p.stock <= 0
   530→                              ? "text-red-600 font-medium"
   531→                              : p.stock <= p.minStock
   532→                              ? "text-amber-600 font-medium"
   533→                              : "text-indigo-700"
   534→                          }
   535→                        >
   536→                          {p.stock} {p.unit === "KG" ? "kg" : "u"}
   537→                        </span>
   538→                      </TableCell>
   539→                      <TableCell className="text-center">
   540→                        {p.active ? (
   541→                          <Badge className="bg-indigo-100 text-indigo-700 hover:bg-indigo-100">
   542→                            Activo
   543→                          </Badge>
   544→                        ) : (
   545→                          <Badge variant="secondary">Inactivo</Badge>
   546→                        )}
   547→                      </TableCell>
   548→                      <TableCell className="text-right">
   549→                        <Button
   550→                          variant="ghost"
   551→                          size="icon"
   552→                          className="h-8 w-8"
   553→                          onClick={() => openEdit(p)}
   554→                        >
   555→                          <Pencil className="w-3.5 h-3.5" />
   556→                        </Button>
   557→                        <Button
   558→                          variant="ghost"
   559→                          size="icon"
   560→                          className="h-8 w-8 text-red-600"
   561→                          onClick={() => setDeleteId(p.id)}
   562→                        >
   563→                          <Trash2 className="w-3.5 h-3.5" />
   564→                        </Button>
   565→                      </TableCell>
   566→                    </TableRow>
   567→                  ))}
   568→                </TableBody>
   569→              </Table>
   570→            </div>
   571→          )}
   572→        </CardContent>
   573→      </Card>
   574→
   575→      {/* Form dialog */}
   576→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   577→        <DialogContent className="sm:max-w-4xl w-[95vw] max-h-[92vh] overflow-y-auto">
   578→          <DialogHeader>
   579→            <DialogTitle>{form.id ? "Editar producto" : "Nuevo producto"}</DialogTitle>
   580→            <DialogDescription>
   581→              {form.id
   582→                ? "Modificá los datos del producto. Si cambiás el stock, se registrará un ajuste."
   583→                : "Completá los datos del nuevo producto."}
   584→            </DialogDescription>
   585→          </DialogHeader>
   586→
   587→          <div className="grid gap-4 grid-cols-1 sm:grid-cols-2">
   588→            <div className="space-y-2 sm:col-span-2">
   589→              <Label htmlFor="name">Nombre *</Label>
   590→              <Input
   591→                id="name"
   592→                value={form.name}
   593→                onChange={(e) => setForm({ ...form, name: e.target.value })}
