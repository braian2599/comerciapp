   116→   116→  const { store } = useAppStore();
   117→   117→  const symbol = store?.currencySymbol || "$";
   118→   118→  const [customers, setCustomers] = useState<Customer[]>([]);
   119→   119→  const [loading, setLoading] = useState(true);
   120→   120→  const [search, setSearch] = useState("");
   121→   121→  const [formOpen, setFormOpen] = useState(false);
   122→   122→  const [form, setForm] = useState<any>(emptyForm);
   123→   123→  const [saving, setSaving] = useState(false);
   124→   124→  const [deleteId, setDeleteId] = useState<string | null>(null);
   125→   125→  const [importOpen, setImportOpen] = useState(false);
   126→   126→
   127→   127→  // Cuenta corriente
   128→   128→  const [accountOpen, setAccountOpen] = useState(false);
   129→   129→  const [accountData, setAccountData] = useState<AccountState | null>(null);
   130→   130→  const [accountLoading, setAccountLoading] = useState(false);
   131→   131→  const [payOpen, setPayOpen] = useState(false);
   132→   132→  const [payAmount, setPayAmount] = useState(0);
   133→   133→  const [payMethod, setPayMethod] = useState("EFECTIVO");
   134→   134→  const [payNotes, setPayNotes] = useState("");
   135→   135→  const [paySaving, setPaySaving] = useState(false);
   136→   136→
   137→   137→  // Fidelización - puntos
   138→   138→  const [pointsOpen, setPointsOpen] = useState(false);
   139→   139→  const [pointsData, setPointsData] = useState<any | null>(null);
   140→   140→  const [pointsLoading, setPointsLoading] = useState(false);
   141→   141→  const [adjustPoints, setAdjustPoints] = useState(0);
   142→   142→  const [adjustNotes, setAdjustNotes] = useState("");
   143→   143→  const [adjustSaving, setAdjustSaving] = useState(false);
   144→   144→
   145→   145→  // Programa de fidelización (cargado una vez)
   146→   146→  const [loyaltyEnabled, setLoyaltyEnabled] = useState(false);
   147→   147→
   148→   148→  async function load() {
   149→   149→    setLoading(true);
   150→   150→    try {
   151→   151→      const [custRes, loyRes] = await Promise.all([
   152→   152→        safeFetchArray<Customer>("/api/customers"),
   153→   153→        safeFetchJSON<any>("/api/loyalty"),
   154→   154→      ]);
   155→   155→      setCustomers(custRes);
   156→   156→      const loy = loyRes.data;
   157→   157→      setLoyaltyEnabled(!!loy?.enabled && typeof loy === "object" && !Array.isArray(loy));
   158→   158→    } catch {
   159→   159→      toast.error("No se pudieron cargar los clientes");
   160→   160→      setCustomers([]);
   161→   161→    } finally {
   162→   162→      setLoading(false);
   163→   163→    }
   164→   164→  }
   165→   165→
   166→   166→  useEffect(() => {
   167→   167→    load();
   168→   168→  }, []);
   169→   169→
   170→   170→  const filtered = useMemo(() => {
   171→   171→    return customers.filter((c) => {
   172→   172→      if (!search) return true;
   173→   173→      const s = search.toLowerCase();
   174→   174→      return (
   175→   175→        c.name.toLowerCase().includes(s) ||
   176→   176→        c.phone?.toLowerCase().includes(s) ||
   177→   177→        c.email?.toLowerCase().includes(s)
   178→   178→      );
   179→   179→    });
   180→   180→  }, [customers, search]);
   181→   181→
   182→   182→  const saldoTotal = customers.reduce((s, c) => s + (c.saldo || 0), 0);
   183→   183→
   184→   184→  function openNew() {
   185→   185→    setForm({ ...emptyForm });
   186→   186→    setFormOpen(true);
   187→   187→  }
   188→   188→
   189→   189→  function openEdit(c: Customer) {
   190→   190→    setForm({ ...c, creditLimit: c.creditLimit || 0 });
   191→   191→    setFormOpen(true);
   192→   192→  }
   193→   193→
   194→   194→  async function openAccount(c: Customer) {
   195→   195→    setAccountOpen(true);
   196→   196→    setAccountLoading(true);
   197→   197→    setAccountData(null);
   198→   198→    try {
   199→   199→      const res = await fetch(`/api/customers/account?customerId=${c.id}`);
   200→   200→      const data = await res.json();
   201→   201→      setAccountData(data);
   202→   202→    } catch {
   203→   203→      toast.error("Error al cargar cuenta");
   204→   204→    } finally {
   205→   205→      setAccountLoading(false);
   206→   206→    }
   207→   207→  }
   208→   208→
   209→   209→  async function openPoints(c: Customer) {
   210→   210→    setPointsOpen(true);
   211→   211→    setPointsLoading(true);
   212→   212→    setPointsData(null);
   213→   213→    setAdjustPoints(0);
   214→   214→    setAdjustNotes("");
   215→   215→    try {
   216→   216→      const res = await fetch(`/api/loyalty/points?customerId=${c.id}`);
   217→   217→      const data = await res.json();
   218→   218→      setPointsData(data);
   219→   219→    } catch {
   220→   220→      toast.error("Error al cargar puntos");
   221→   221→    } finally {
   222→   222→      setPointsLoading(false);
   223→   223→    }
   224→   224→  }
   225→   225→
   226→   226→  async function handleAdjustPoints() {
   227→   227→    if (!pointsData?.customer?.id) return;
   228→   228→    if (adjustPoints === 0) {
   229→   229→      toast.error("Indica puntos (positivo o negativo)");
   230→   230→      return;
   231→   231→    }
   232→   232→    setAdjustSaving(true);
   233→   233→    try {
   234→   234→      const res = await fetch("/api/loyalty/points", {
   235→   235→        method: "POST",
   236→   236→        headers: { "Content-Type": "application/json" },
   237→   237→        body: JSON.stringify({
   238→   238→          customerId: pointsData.customer.id,
   239→   239→          points: adjustPoints,
   240→   240→          description: adjustNotes,
   241→   241→        }),
   242→   242→      });
   243→   243→      const data = await res.json();
   244→   244→      if (!res.ok) throw new Error(data.error);
   245→   245→      toast.success("Puntos ajustados");
   246→   246→      // Recargar
   247→   247→      await openPoints(pointsData.customer);
   248→   248→      load();
   249→   249→    } catch (e: any) {
   250→   250→      toast.error(e.message);
   251→   251→    } finally {
   252→   252→      setAdjustSaving(false);
   253→   253→    }
   254→   254→  }
   255→   255→
   256→   256→  async function handleSave() {
   257→   257→    if (!form.name) {
   258→   258→      toast.error("El nombre es obligatorio");
   259→   259→      return;
   260→   260→    }
   261→   261→    setSaving(true);
   262→   262→    try {
   263→   263→      const method = form.id ? "PUT" : "POST";
   264→   264→      const res = await fetch("/api/customers", {
   265→   265→        method,
   266→   266→        headers: { "Content-Type": "application/json" },
   267→   267→        body: JSON.stringify(form),
   268→   268→      });
   269→   269→      const data = await res.json();
   270→   270→      if (!res.ok) throw new Error(data.error);
   271→   271→      toast.success(form.id ? "Cliente actualizado" : "Cliente creado");
   272→   272→      setFormOpen(false);
   273→   273→      load();
   274→   274→    } catch (e: any) {
   275→   275→      toast.error(e.message);
   276→   276→    } finally {
   277→   277→      setSaving(false);
   278→   278→    }
   279→   279→  }
   280→   280→
   281→   281→  async function handleDelete() {
   282→   282→    if (!deleteId) return;
   283→   283→    try {
   284→   284→      const res = await fetch(`/api/customers?id=${deleteId}`, { method: "DELETE" });
   285→   285→      const data = await res.json();
   286→   286→      if (!res.ok) throw new Error(data.error);
   287→   287→      toast.success("Cliente eliminado");
   288→   288→      setDeleteId(null);
   289→   289→      load();
   290→   290→    } catch (e: any) {
   291→   291→      toast.error(e.message);
   292→   292→    }
   293→   293→  }
   294→   294→
   295→   295→  async function handlePayment() {
   296→   296→    if (!accountData) return;
   297→   297→    if (payAmount <= 0) {
   298→   298→      toast.error("El monto debe ser mayor a 0");
   299→   299→      return;
   300→   300→    }
   301→   301→    setPaySaving(true);
   302→   302→    try {
   303→   303→      const res = await fetch("/api/customers/account", {
   304→   304→        method: "POST",
   305→   305→        headers: { "Content-Type": "application/json" },
   306→   306→        body: JSON.stringify({
   307→   307→          customerId: accountData.customer.id,
   308→   308→          amount: payAmount,
   309→   309→          paymentMethod: payMethod,
   310→   310→          notes: payNotes,
   311→   311→        }),
   312→   312→      });
   313→   313→      const data = await res.json();
   314→   314→      if (!res.ok) throw new Error(data.error);
   315→   315→      toast.success("Pago registrado");
   316→   316→      setPayOpen(false);
   317→   317→      setPayAmount(0);
   318→   318→      setPayNotes("");
   319→   319→      // Recargar cuenta y lista
   320→   320→      await openAccount(accountData.customer);
   321→   321→      load();
   322→   322→    } catch (e: any) {
   323→   323→      toast.error(e.message);
   324→   324→    } finally {
   325→   325→      setPaySaving(false);
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  return (
   330→   330→    <div className="space-y-4">
   331→   331→      <ImportDialog
   332→   332→        open={importOpen}
   333→   333→        onOpenChange={setImportOpen}
   334→   334→        endpoint="/api/customers/import"
   335→   335→        fields={CUSTOMER_IMPORT_FIELDS}
   336→   336→        templateHeaders={[
   337→   337→          "name",
   338→   338→          "phone",
   339→   339→          "email",
   340→   340→          "address",
   341→   341→          "cuit",
   342→   342→          "taxType",
   343→   343→          "creditLimit",
   344→   344→          "notes",
   345→   345→        ]}
   346→   346→        templateName="plantilla_clientes.csv"
   347→   347→        entityLabel="cliente"
   348→   348→        entityLabelPlural="clientes"
   349→   349→        onImported={load}
   350→   350→      />
   351→   351→      <div className="flex items-center justify-between flex-wrap gap-3">
   352→   352→        <div>
   353→   353→          <h1 className="text-2xl font-bold tracking-tight">Clientes</h1>
   354→   354→          <p className="text-sm text-muted-foreground">
   355→   355→            {customers.length} clientes · Saldo total en cuentas:{" "}
   356→   356→            <span className={saldoTotal > 0 ? "text-amber-700 font-medium" : ""}>
   357→   357→              {formatCurrency(saldoTotal, symbol)}
   358→   358→            </span>
   359→   359→          </p>
   360→   360→        </div>
   361→   361→        <div className="flex gap-2">
   362→   362→          <Button
   363→   363→            variant="outline"
   364→   364→            onClick={() => setImportOpen(true)}
   365→   365→          >
   366→   366→            <Upload className="w-4 h-4 mr-2" />
   367→   367→            Importar
   368→   368→          </Button>
   369→   369→          <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   370→   370→            <Plus className="w-4 h-4 mr-2" />
   371→   371→            Nuevo cliente
   372→   372→          </Button>
   373→   373→        </div>
   374→   374→      </div>
   375→   375→
   376→   376→      <Card>
   377→   377→        <CardContent className="p-3">
   378→   378→          <div className="relative">

   381→   381→              placeholder="Buscar por nombre, teléfono o email..."
   382→   382→              value={search}
   383→   383→              onChange={(e) => setSearch(e.target.value)}
   384→   384→              className="pl-9"
   385→   385→            />
   386→   386→          </div>
   387→   387→        </CardContent>
   388→   388→      </Card>
   389→   389→
   390→   390→      <Card>
   391→   391→        <CardContent className="p-0">
   392→   392→          {loading ? (
   393→   393→            <div className="p-8 text-center">
   394→   394→              <Loader2 className="w-5 h-5 animate-spin inline mr-2 text-indigo-600" />
   395→   395→              Cargando clientes...
   396→   396→            </div>
   397→   397→          ) : filtered.length === 0 ? (
   398→   398→            <div className="p-12 text-center">
   399→   399→              <Users className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   400→   400→              <p className="text-sm text-muted-foreground">
   401→   401→                {search
   402→   402→                  ? "No se encontraron clientes"
   403→   403→                  : "Aún no tenés clientes registrados"}
   404→   404→              </p>
   405→   405→            </div>
   406→   406→          ) : (
   407→   407→            <div className="overflow-x-auto">
   408→   408→              <Table>
   409→   409→                <TableHeader>
   410→   410→                  <TableRow>
   411→   411→                    <TableHead>Nombre</TableHead>
   412→   412→                    <TableHead className="hidden sm:table-cell">Contacto</TableHead>
   413→   413→                    <TableHead className="text-center">Compras</TableHead>
   414→   414→                    {loyaltyEnabled && (
   415→   415→                      <TableHead className="text-center">Puntos / Tier</TableHead>
   416→   416→                    )}
   417→   417→                    <TableHead className="text-right">Saldo cta.</TableHead>
   418→   418→                    <TableHead className="text-right">Acciones</TableHead>
   419→   419→                  </TableRow>
   420→   420→                </TableHeader>
   421→   421→                <TableBody>
   422→   422→                  {filtered.map((c) => {
   423→   423→                    const saldo = c.saldo || 0;
   424→   424→                    return (
   425→   425→                      <TableRow key={c.id}>
   426→   426→                        <TableCell>
   427→   427→                          <p className="font-medium">{c.name}</p>
   428→   428→                          {c.notes && (
   429→   429→                            <p className="text-xs text-muted-foreground line-clamp-1">
   430→   430→                              {c.notes}
   431→   431→                            </p>
   432→   432→                          )}
   433→   433→                        </TableCell>
   434→   434→                        <TableCell className="hidden sm:table-cell">
   435→   435→                          <div className="text-sm space-y-0.5">
   436→   436→                            {c.phone && (
   437→   437→                              <p className="flex items-center gap-1">
   438→   438→                                <Phone className="w-3 h-3" />
   439→   439→                                {c.phone}
   440→   440→                              </p>
   441→   441→                            )}
   442→   442→                            {c.email && (

   449→   449→                              <span className="text-muted-foreground text-xs">—</span>
   450→   450→                            )}
   451→   451→                          </div>
   452→   452→                        </TableCell>
   453→   453→                        <TableCell className="text-center">
   454→   454→                          <Badge variant="outline">{c._count?.sales || 0}</Badge>
   455→   455→                        </TableCell>
   456→   456→                        {loyaltyEnabled && (
   457→   457→                          <TableCell className="text-center">
   458→   458→                            {c.loyaltyTier && (
   459→   459→                              <Badge
   460→   460→                                variant="outline"
   461→   461→                                className={
   462→   462→                                  c.loyaltyTier === "PLATINO"
   463→   463→                                    ? "bg-gray-100 text-gray-800"
   464→   464→                                    : c.loyaltyTier === "ORO"
   465→   465→                                    ? "bg-yellow-100 text-yellow-800"
   466→   466→                                    : c.loyaltyTier === "PLATA"
   467→   467→                                    ? "bg-slate-100 text-slate-700"
   468→   468→                                    : "bg-amber-50 text-amber-800"
   469→   469→                                }
   470→   470→                              >
   471→   471→                                <Star className="w-3 h-3 mr-1" />
   472→   472→                                {c.loyaltyTier}
   473→   473→                              </Badge>
   474→   474→                            )}
   475→   475→                            <p className="text-xs mt-0.5">
   476→   476→                              {Math.floor(c.loyaltyPoints || 0)} pts
   477→   477→                            </p>
   478→   478→                          </TableCell>
   479→   479→                        )}
   480→   480→                        <TableCell className="text-right">
   481→   481→                          {saldo > 0 ? (
   482→   482→                            <Badge className="bg-amber-100 text-amber-800 hover:bg-amber-100">
   483→   483→                              {formatCurrency(saldo, symbol)}
   484→   484→                            </Badge>
   485→   485→                          ) : (
   486→   486→                            <span className="text-xs text-muted-foreground">Saldada</span>
   487→   487→                          )}
   488→   488→                        </TableCell>
   489→   489→                        <TableCell className="text-right">
   490→   490→                          <Button
   491→   491→                            variant="ghost"
   492→   492→                            size="sm"
   493→   493→                            className="h-8"
   494→   494→                            onClick={() => openAccount(c)}
   495→   495→                          >
   496→   496→                            <Wallet className="w-3.5 h-3.5 mr-1" />
   497→   497→                            Cuenta
   498→   498→                          </Button>
   499→   499→                          {loyaltyEnabled && (
   500→   500→                            <Button
   501→   501→                              variant="ghost"
   502→   502→                              size="sm"
   503→   503→                              className="h-8"
   504→   504→                              onClick={() => openPoints(c)}
