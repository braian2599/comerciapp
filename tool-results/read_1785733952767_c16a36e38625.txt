   149→   149→  async function handleBarcodeLookup() {
   150→   150→    const code = (form.barcode || "").trim();
   151→   151→    if (!code) {
   152→   152→      toast.error("Ingresá un código de barras primero");
   153→   153→      return;
   154→   154→    }
   155→   155→    setLookupLoading(true);
   156→   156→    setLookupResult({ status: "loading" });
   157→   157→    try {
   158→   158→      const res = await fetch(`/api/products/lookup?barcode=${encodeURIComponent(code)}`);
   159→   159→      const data = await res.json();
   160→   160→      if (data.found) {
   161→   161→        // Solo autocompletamos campos que el usuario aún no llenó.
   162→   162→        setForm((prev: any) => ({
   163→   163→          ...prev,
   164→   164→          name: prev.name || data.name || "",
   165→   165→          description: prev.description || data.description || data.brand || "",
   166→   166→          barcode: prev.barcode || code,
   167→   167→          brand: prev.brand || data.brand || "",
   168→   168→          ingredients: prev.ingredients || data.ingredients || "",
   169→   169→          imageUrl: prev.imageUrl || data.imageUrl || "",
   170→   170→        }));
   171→   171→        // Merge de etiquetas y alérgenos (sin duplicar)
   172→   172→        if (Array.isArray(data.labels) && data.labels.length > 0) {
   173→   173→          setLabelChips((prev) =>
   174→   174→            Array.from(new Set([...prev, ...data.labels])).slice(0, 15)
   175→   175→          );
   176→   176→          setShowExtraFields(true);
   177→   177→        }
   178→   178→        if (Array.isArray(data.allergens) && data.allergens.length > 0) {
   179→   179→          setAllergenChips((prev) =>
   180→   180→            Array.from(new Set([...prev, ...data.allergens])).slice(0, 15)
   181→   181→          );
   182→   182→          setShowExtraFields(true);
   183→   183→        }
   184→   184→        setLookupResult({
   185→   185→          status: "found",
   186→   186→          source: data.source,
   187→   187→          imageUrl: data.imageUrl,
   188→   188→          labels: data.labels,
   189→   189→          allergens: data.allergens,
   190→   190→        });
   191→   191→        const sourceLabel =
   192→   192→          data.source === "openfoodfacts" ? "Open Food Facts" : "UPC Item DB";
   193→   193→        toast.success(`Producto encontrado en ${sourceLabel}`, {
   194→   194→          description: data.name,
   195→   195→        });
   196→   196→      } else {
   197→   197→        setLookupResult({ status: "notfound" });
   198→   198→        toast.info("No se encontró el código en las bases de datos públicas", {
   199→   199→          description: "Completá los datos manualmente",
   200→   200→        });
   201→   201→      }
   202→   202→    } catch (e: any) {
   203→   203→      setLookupResult({ status: "error" });
   204→   204→      toast.error("No se pudo consultar la base de productos");
   205→   205→    } finally {
   206→   206→      setLookupLoading(false);
   207→   207→    }
   208→   208→  }
