    69→  name: string;
    70→  description?: string;
    71→  barcode?: string;
    72→  sku?: string;
    73→  categoryId?: string;
    74→  category?: { id: string; name: string };
    75→  costPrice: number;
    76→  salePrice: number;
    77→  stock: number;
    78→  minStock: number;
    79→  unit: string;
    80→  active: boolean;
    81→  brand?: string;
    82→  labels?: string;
    83→  ingredients?: string;
    84→  allergens?: string;
    85→  imageUrl?: string;

    88→interface Category {
    89→  id: string;
    90→  name: string;
    91→  _count?: { products: number };
    92→}

    94→const emptyForm = {
    95→  id: "",
    96→  name: "",
    97→  description: "",
    98→  barcode: "",
    99→  sku: "",
   100→  categoryId: "",
   101→  costPrice: 0,
   102→  salePrice: 0,
   103→  stock: 0,
   104→  minStock: 5,
   105→  unit: "UNIDAD",
   106→  active: true,
   107→  brand: "",
   108→  labels: "",
   109→  ingredients: "",
   110→  allergens: "",
   111→  imageUrl: "",
   112→};
   113→
   114→export function ProductsView() {
   115→  const { store } = useAppStore();
   116→  const [products, setProducts] = useState<Product[]>([]);
   117→  const [categories, setCategories] = useState<Category[]>([]);
   118→  const [loading, setLoading] = useState(true);
   119→  const [search, setSearch] = useState("");
   120→  const [filterCat, setFilterCat] = useState("all");
   121→  const [filterStock, setFilterStock] = useState("all");
   122→
   123→  const [formOpen, setFormOpen] = useState(false);
   124→  const [form, setForm] = useState<any>(emptyForm);
   125→  const [saving, setSaving] = useState(false);
   126→  const [deleteId, setDeleteId] = useState<string | null>(null);
   127→  const [catDialogOpen, setCatDialogOpen] = useState(false);
   128→  const [newCatName, setNewCatName] = useState("");
   129→
   130→  // Lookup por código de barras
   131→  const [lookupLoading, setLookupLoading] = useState(false);
   132→  const [lookupResult, setLookupResult] = useState<
   133→    | {
   134→        status: "idle" | "loading" | "found" | "notfound" | "error";
   135→        source?: string;
   136→        imageUrl?: string;
   137→        labels?: string[];
   138→        allergens?: string[];
   139→      }
   140→    | null
   141→  >(null);
   142→  // Sección colapsable de datos adicionales
   143→  const [showExtraFields, setShowExtraFields] = useState(false);
   144→  // Etiquetas y alérgenos editados como chips
   145→  const [labelChips, setLabelChips] = useState<string[]>([]);
   146→  const [allergenChips, setAllergenChips] = useState<string[]>([]);
   147→  const [labelInput, setLabelInput] = useState("");
   148→  const [allergenInput, setAllergenInput] = useState("");
   149→
   150→  async function handleBarcodeLookup() {
   151→    const code = (form.barcode || "").trim();
   152→    if (!code) {
   153→      toast.error("Ingresá un código de barras primero");
   154→      return;
   155→    }
   156→    setLookupLoading(true);
   157→    setLookupResult({ status: "loading" });
   158→    try {
   159→      const { ok, data, error } = await safeFetchJSON<any>(
   160→        `/api/products/lookup?barcode=${encodeURIComponent(code)}`
   161→      );
   162→      if (!ok || !data) {
   163→        throw new Error(error || "No se pudo consultar la base de productos");
   164→      }
   165→      if (data.found) {
   166→        // Solo autocompletamos campos que el usuario aún no llenó.
   167→        setForm((prev: any) => ({
   168→          ...prev,
   169→          name: prev.name || data.name || "",
   170→          description: prev.description || data.description || data.brand || "",
   171→          barcode: prev.barcode || code,
   172→          brand: prev.brand || data.brand || "",
   173→          ingredients: prev.ingredients || data.ingredients || "",
   174→          imageUrl: prev.imageUrl || data.imageUrl || "",
   175→        }));
   176→        // Merge de etiquetas y alérgenos (sin duplicar)
   177→        if (Array.isArray(data.labels) && data.labels.length > 0) {
   178→          setLabelChips((prev) =>
   179→            Array.from(new Set([...prev, ...data.labels])).slice(0, 15)
   180→          );
   181→          setShowExtraFields(true);
   182→        }
   183→        if (Array.isArray(data.allergens) && data.allergens.length > 0) {
   184→          setAllergenChips((prev) =>
   185→            Array.from(new Set([...prev, ...data.allergens])).slice(0, 15)
   186→          );
   187→          setShowExtraFields(true);
   188→        }
   189→        setLookupResult({
   190→          status: "found",
   191→          source: data.source,
   192→          imageUrl: data.imageUrl,
   193→          labels: data.labels,
   194→          allergens: data.allergens,
   195→        });
   196→        const sourceLabel =
   197→          data.source === "openfoodfacts" ? "Open Food Facts" : "UPC Item DB";
   198→        toast.success(`Producto encontrado en ${sourceLabel}`, {
   199→          description: data.name,
   200→        });
   201→      } else {
   202→        setLookupResult({ status: "notfound" });
   203→        toast.info("No se encontró el código en las bases de datos públicas", {
   204→          description: "Completá los datos manualmente",
   205→        });
   206→      }
   207→    } catch (e: any) {
   208→      setLookupResult({ status: "error" });
   209→      toast.error("No se pudo consultar la base de productos");
   210→    } finally {
   211→      setLookupLoading(false);
   212→    }
   213→  }
   214→
   215→  function resetLookup() {
   216→    setLookupResult(null);
   217→  }
   218→
   219→  const symbol = store?.currencySymbol || "$";
   220→
   221→  async function load() {
   222→    setLoading(true);
   223→    try {
   224→      const [p, c] = await Promise.all([
   225→        safeFetchArray<Product>("/api/products"),
   226→        safeFetchArray<Category>("/api/categories"),
   227→      ]);
   228→      setProducts(p);
   229→      setCategories(c);
   230→    } catch {
   231→      toast.error("No se pudieron cargar los productos");
   232→      setProducts([]);
   233→      setCategories([]);
   234→    } finally {
   235→      setLoading(false);
   236→    }
   237→  }
   238→
   239→  useEffect(() => {
   240→    load();
   241→  }, []);
   242→
   243→  const filtered = useMemo(() => {
   244→    return products.filter((p) => {
   245→      const matchSearch =
   246→        !search ||
   247→        p.name.toLowerCase().includes(search.toLowerCase()) ||
   248→        p.barcode?.includes(search) ||
   249→        p.sku?.toLowerCase().includes(search.toLowerCase());
   250→      const matchCat =
   251→        filterCat === "all" || p.categoryId === filterCat;
   252→      const matchStock =
   253→        filterStock === "all" ||
   254→        (filterStock === "low" && p.stock <= p.minStock) ||
   255→        (filterStock === "out" && p.stock <= 0) ||
   256→        (filterStock === "ok" && p.stock > p.minStock);
   257→      return matchSearch && matchCat && matchStock;
   258→    });
   259→  }, [products, search, filterCat, filterStock]);
   260→
   261→  function openNew() {
   262→    setForm({ ...emptyForm });
   263→    resetLookup();
   264→    setLabelChips([]);
   265→    setAllergenChips([]);
   266→    setShowExtraFields(false);
   267→    setFormOpen(true);
   268→  }
   269→
   270→  function openEdit(p: Product) {
   271→    setForm({
   272→      ...p,
   273→      costPrice: p.costPrice,
   274→      salePrice: p.salePrice,
   275→      stock: p.stock,
   276→      minStock: p.minStock,
   277→      adjustReason: "",
   278→    });
   279→    resetLookup();
   280→    setLabelChips(p.labels ? p.labels.split(",").map((s) => s.trim()).filter(Boolean) : []);
   281→    setAllergenChips(p.allergens ? p.allergens.split(",").map((s) => s.trim()).filter(Boolean) : []);
   282→    setShowExtraFields(
   283→      !!(p.brand || p.ingredients || p.labels || p.allergens || p.imageUrl)
   284→    );
   285→    setFormOpen(true);
   286→  }
   287→
   288→  async function handleSave() {
   289→    if (!form.name || !form.salePrice) {
   290→      toast.error("Nombre y precio de venta son obligatorios");
   291→      return;
   292→    }
   293→    setSaving(true);
   294→    try {
   295→      const method = form.id ? "PUT" : "POST";
   296→      const payload = {
   297→        ...form,
   298→        labels: labelChips.join(","),
   299→        allergens: allergenChips.join(","),
   300→      };
   301→      const { ok, error } = await safeFetchJSON("/api/products", {
   302→        method,
   303→        body: JSON.stringify(payload),
   304→      });
   305→      if (!ok) throw new Error(error);
   306→      toast.success(form.id ? "Producto actualizado" : "Producto creado");
   307→      setFormOpen(false);
   308→      load();
   309→    } catch (e: any) {
   310→      const msg = e?.message || "No se pudo guardar el producto";
   311→      toast.error("Error al guardar el producto", { description: msg });
   312→    } finally {
   313→      setSaving(false);
   314→    }
   315→  }
   316→
   317→  async function handleDelete() {
   318→    if (!deleteId) return;
   319→    try {
   320→      const { ok, error } = await safeFetchJSON(
   321→        `/api/products?id=${deleteId}`,
   322→        { method: "DELETE" }
   323→      );
   324→      if (!ok) throw new Error(error);
   325→      toast.success("Producto desactivado");
   326→      setDeleteId(null);
   327→      load();
   328→    } catch (e: any) {
   329→      toast.error("Error al eliminar", { description: e.message });
   330→    }
   331→  }
   332→
   333→  async function handleAddCategory() {
   334→    if (!newCatName.trim()) return;
   335→    try {
   336→      const { ok, error } = await safeFetchJSON("/api/categories", {
   337→        method: "POST",
   338→        body: JSON.stringify({ name: newCatName }),
   339→      });
   340→      if (!ok) throw new Error(error);
   341→      toast.success("Categoría creada");
   342→      setNewCatName("");
   343→      setCatDialogOpen(false);
   344→      load();
   345→    } catch (e: any) {
   346→      toast.error("Error al crear categoría", { description: e.message });
   347→    }
   348→  }
   349→
   350→  async function handleDeleteCategory(id: string, name: string) {
   351→    if (!confirm(`¿Eliminar la categoría "${name}"? Los productos quedarán sin categoría.`)) return;
   352→    try {
   353→      const { ok, error } = await safeFetchJSON(`/api/categories?id=${id}`, {
   354→        method: "DELETE",
   355→      });
   356→      if (!ok) throw new Error(error);
   357→      toast.success("Categoría eliminada");
   358→      load();
   359→    } catch (e: any) {
   360→      toast.error("Error al eliminar categoría", { description: e.message });
   361→    }
   362→  }
   363→
   364→  return (
   365→    <div className="space-y-4">
   366→      <div className="flex items-center justify-between flex-wrap gap-3">
   367→        <div>
   368→          <h1 className="text-2xl font-bold tracking-tight">Productos</h1>
   369→          <p className="text-sm text-muted-foreground">
   370→            {products.length} productos · {categories.length} categorías
   371→          </p>
   372→        </div>
   373→        <div className="flex gap-2">
   374→          <Dialog open={catDialogOpen} onOpenChange={setCatDialogOpen}>
   375→            <DialogTrigger asChild>
   376→              <Button variant="outline">
   377→                <Tag className="w-4 h-4 mr-2" />
   378→                Categorías
   379→              </Button>
   380→            </DialogTrigger>
   381→            <DialogContent>
   382→              <DialogHeader>
   383→                <DialogTitle>Gestionar categorías</DialogTitle>
   384→                <DialogDescription>
   385→                  Agregá o eliminá categorías para organizar tus productos
   386→                </DialogDescription>
   387→              </DialogHeader>
   388→              <div className="flex gap-2">
   389→                <Input
   390→                  placeholder="Nueva categoría..."
   391→                  value={newCatName}
   392→                  onChange={(e) => setNewCatName(e.target.value)}
   393→                  onKeyDown={(e) => e.key === "Enter" && handleAddCategory()}
   394→                />
   395→                <Button onClick={handleAddCategory} className="bg-indigo-600 hover:bg-indigo-700">
   396→                  <Plus className="w-4 h-4" />
   397→                </Button>
   398→              </div>
   399→              <div className="max-h-60 overflow-y-auto space-y-1 mt-2">
   400→                {categories.length === 0 ? (
   401→                  <p className="text-sm text-muted-foreground text-center py-4">
   402→                    Sin categorías
   403→                  </p>
   404→                ) : (
   405→                  categories.map((c) => (
   406→                    <div
   407→                      key={c.id}
   408→                      className="flex items-center justify-between p-2 rounded-md bg-muted/50"
   409→                    >
   410→                      <div>
   411→                        <p className="text-sm font-medium">{c.name}</p>
   412→                        <p className="text-xs text-muted-foreground">
   413→                          {c._count?.products || 0} productos
   414→                        </p>
   415→                      </div>
   416→                      <Button
   417→                        variant="ghost"
   418→                        size="icon"
   419→                        className="h-7 w-7 text-red-600"
   420→                        onClick={() => handleDeleteCategory(c.id, c.name)}
   421→                      >
   422→                        <Trash2 className="w-3.5 h-3.5" />
   423→                      </Button>
   424→                    </div>
   425→                  ))
   426→                )}
   427→              </div>
   428→            </DialogContent>
   429→          </Dialog>
   430→
   431→          <Button onClick={openNew} className="bg-indigo-600 hover:bg-indigo-700">
   432→            <Plus className="w-4 h-4 mr-2" />
   433→            Nuevo producto
   434→          </Button>
   435→        </div>
   436→      </div>
   437→
   438→      {/* Filtros */}
   439→      <Card>
   440→        <CardContent className="p-3">
   441→          <div className="grid gap-2 grid-cols-1 sm:grid-cols-3">
   442→            <div className="relative">
   443→              <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   444→              <Input
   445→                placeholder="Buscar por nombre, código o SKU..."
   446→                value={search}
   447→                onChange={(e) => setSearch(e.target.value)}
   448→                className="pl-9"
   449→              />
   450→            </div>
   451→            <Select value={filterCat} onValueChange={setFilterCat}>
   452→              <SelectTrigger>
   453→                <SelectValue placeholder="Todas las categorías" />
   454→              </SelectTrigger>
   455→              <SelectContent>
   456→                <SelectItem value="all">Todas las categorías</SelectItem>
   457→                {categories.map((c) => (
   458→                  <SelectItem key={c.id} value={c.id}>
   459→                    {c.name}
   460→                  </SelectItem>
   461→                ))}
   462→              </SelectContent>
   463→            </Select>
   464→            <Select value={filterStock} onValueChange={setFilterStock}>
   465→              <SelectTrigger>
   466→                <SelectValue />
   467→              </SelectTrigger>
   468→              <SelectContent>
   469→                <SelectItem value="all">Todo el stock</SelectItem>
   470→                <SelectItem value="ok">Stock OK</SelectItem>
   471→                <SelectItem value="low">Stock bajo</SelectItem>
   472→                <SelectItem value="out">Sin stock</SelectItem>
   473→              </SelectContent>
   474→            </Select>
   475→          </div>
   476→        </CardContent>
   477→      </Card>
   478→
   479→      {/* Tabla */}
   480→      <Card>
   481→        <CardContent className="p-0">
   482→          {loading ? (
   483→            <div className="p-8 text-center text-muted-foreground">
   484→              <Loader2 className="w-5 h-5 animate-spin inline mr-2" />
   485→              Cargando productos...
   486→            </div>
   487→          ) : filtered.length === 0 ? (
   488→            <div className="p-12 text-center">
   489→              <Package className="w-10 h-10 mx-auto text-muted-foreground/50 mb-3" />
   490→              <p className="text-sm text-muted-foreground">
   491→                {search || filterCat !== "all" || filterStock !== "all"
   492→                  ? "No se encontraron productos con esos filtros"
   493→                  : "Aún no tenés productos. Creá el primero!"}
   494→              </p>
   495→            </div>
   496→          ) : (
   497→            <div className="overflow-x-auto">
   498→              <Table>
   499→                <TableHeader>
   500→                  <TableRow>
   501→                    <TableHead>Nombre</TableHead>
   502→                    <TableHead className="hidden md:table-cell">Categoría</TableHead>
   503→                    <TableHead className="text-right">Costo</TableHead>
   504→                    <TableHead className="text-right">Precio</TableHead>
   505→                    <TableHead className="text-right">Stock</TableHead>
   506→                    <TableHead className="text-center">Estado</TableHead>
   507→                    <TableHead className="text-right">Acciones</TableHead>
   508→                  </TableRow>
   509→                </TableHeader>
   510→                <TableBody>
   511→                  {filtered.map((p) => (
   512→                    <TableRow key={p.id} className={!p.active ? "opacity-50" : ""}>
   513→                      <TableCell>
   514→                        <div>
   515→                          <p className="font-medium">{p.name}</p>
   516→                          {(p.barcode || p.sku) && (
   517→                            <p className="text-xs text-muted-foreground">
   518→                              {p.barcode && `Cód: ${p.barcode}`}
   519→                              {p.barcode && p.sku && " · "}
   520→                              {p.sku && `SKU: ${p.sku}`}
   521→                            </p>
   522→                          )}
   523→                        </div>
   524→                      </TableCell>
   525→                      <TableCell className="hidden md:table-cell">
   526→                        {p.category ? (
   527→                          <Badge variant="outline">{p.category.name}</Badge>
   528→                        ) : (
   529→                          <span className="text-xs text-muted-foreground">—</span>
   530→                        )}
   531→                      </TableCell>
   532→                      <TableCell className="text-right text-muted-foreground">
   533→                        {formatCurrency(p.costPrice, symbol)}
   534→                      </TableCell>
   535→                      <TableCell className="text-right font-semibold">
   536→                        {formatCurrency(p.salePrice, symbol)}
   537→                      </TableCell>
   538→                      <TableCell className="text-right">
   539→                        <span
   540→                          className={
   541→                            p.stock <= 0
   542→                              ? "text-red-600 font-medium"
   543→                              : p.stock <= p.minStock
   544→                              ? "text-amber-600 font-medium"
   545→                              : "text-indigo-700"
   546→                          }
   547→                        >
   548→                          {p.stock} {p.unit === "KG" ? "kg" : "u"}
   549→                        </span>
   550→                      </TableCell>
   551→                      <TableCell className="text-center">
   552→                        {p.active ? (
   553→                          <Badge className="bg-indigo-100 text-indigo-700 hover:bg-indigo-100">
   554→                            Activo
   555→                          </Badge>
   556→                        ) : (
   557→                          <Badge variant="secondary">Inactivo</Badge>
   558→                        )}
   559→                      </TableCell>
   560→                      <TableCell className="text-right">
   561→                        <Button
   562→                          variant="ghost"
   563→                          size="icon"
   564→                          className="h-8 w-8"
   565→                          onClick={() => openEdit(p)}
   566→                        >
   567→                          <Pencil className="w-3.5 h-3.5" />
   568→                        </Button>
   569→                        <Button
   570→                          variant="ghost"
   571→                          size="icon"
   572→                          className="h-8 w-8 text-red-600"
   573→                          onClick={() => setDeleteId(p.id)}
   574→                        >
   575→                          <Trash2 className="w-3.5 h-3.5" />
   576→                        </Button>
   577→                      </TableCell>
   578→                    </TableRow>
   579→                  ))}
   580→                </TableBody>
   581→              </Table>
   582→            </div>
   583→          )}
   584→        </CardContent>
   585→      </Card>
   586→
   587→      {/* Form dialog */}
   588→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   589→        <DialogContent className="sm:max-w-4xl w-[95vw] max-h-[92vh] overflow-y-auto">
   590→          <DialogHeader>
   591→            <DialogTitle>{form.id ? "Editar producto" : "Nuevo producto"}</DialogTitle>
   592→            <DialogDescription>
   593→              {form.id
   594→                ? "Modificá los datos del producto. Si cambiás el stock, se registrará un ajuste."
   595→                : "Completá los datos del nuevo producto."}
   596→            </DialogDescription>
