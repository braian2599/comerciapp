     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";

     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";

     9→import {

    55→
    56→type TabKey = "sales" | "profits" | "taxes" | "products" | "customers" | "cashflow";
    57→
    58→const COLORS = ["#6366f1", "#3b82f6", "#f59e0b", "#ef4444", "#8b5cf6", "#ec4899", "#06b6d4", "#84cc16"];
    59→
    60→export function ReportsView() {
    61→  const { store } = useAppStore();
    62→  const symbol = store?.currencySymbol || "$";
    63→  const [tab, setTab] = useState<TabKey>("sales");
    64→  const [from, setFrom] = useState("");
    65→  const [to, setTo] = useState("");
    66→  const [loading, setLoading] = useState(false);
    67→  const [data, setData] = useState<any>(null);
    68→
    69→  // Defaults: últimos 30 días
    70→  useEffect(() => {
    71→    if (!from && !to) {
    72→      const end = new Date();
    73→      const start = new Date();
    74→      start.setDate(start.getDate() - 30);
    75→      setFrom(start.toISOString().slice(0, 10));
    76→      setTo(end.toISOString().slice(0, 10));
    77→    }
    78→  }, []);
    79→
    80→  async function loadReport() {
    81→    if (!from || !to) return;
    82→    setLoading(true);
    83→    setData(null); // reset para que no se renderice data de otro tab
    84→    try {
    85→      const params = new URLSearchParams({ from, to });
    86→      const res = await fetch(`/api/reports/${tab}?${params}`);
    87→      const json = await res.json();
    88→      if (!res.ok) {
    89→        toast.error(json.error || "Error cargando reporte");
    90→        return;
    91→      }
    92→      setData(json);
    93→    } catch (err) {
    94→      toast.error("Error de conexión");
    95→    } finally {
    96→      setLoading(false);
    97→    }
    98→  }
    99→
   100→  useEffect(() => {
   101→    if (from && to) loadReport();
   102→  }, [tab, from, to]);
   103→
   104→  function setPeriod(days: number) {
   105→    const end = new Date();
   106→    const start = new Date();
   107→    start.setDate(start.getDate() - days);
   108→    setFrom(start.toISOString().slice(0, 10));
   109→    setTo(end.toISOString().slice(0, 10));
   110→  }
   111→
   112→  function exportarCSV() {
   113→    if (!data) return;
   114→    let rows: string[][] = [];
   115→    let filename: string = tab;
   116→
   117→    if (tab === "sales" && data.series) {
   118→      rows = [["Período", "Cantidad", "Total", "Descuentos", "Recargos"]];
   119→      data.series.forEach((s: any) => {
   120→        rows.push([s.period, s.count, s.total, s.discount, s.surcharge]);
   121→      });
   122→      filename = "ventas";
   123→    } else if (tab === "profits" && data.series) {
   124→      rows = [["Fecha", "Ventas", "Costo", "Gastos", "Ganancia Bruta", "Ganancia Neta"]];
   125→      data.series.forEach((s: any) => {
   126→        rows.push([s.date, s.ventas, s.costo, s.gastos, s.gananciaBruta, s.gananciaNeta]);
   127→      });
   128→      filename = "ganancias";
   129→    } else if (tab === "products" && data.ranking) {
   130→      rows = [["Producto", "Cantidad", "Ingresos", "Costo", "Ganancia", "Margen %", "Stock Actual"]];
   131→      data.ranking.forEach((p: any) => {
   132→        rows.push([p.productName, p.quantitySold, p.revenue, p.cost, p.profit, p.margin, p.currentStock]);
   133→      });
   134→      filename = "productos";
   135→    } else if (tab === "customers" && data.topClientes) {
   136→      rows = [["Cliente", "Compras", "Cantidad ventas", "Fiado", "Pagado"]];
   137→      data.topClientes.forEach((c: any) => {
   138→        rows.push([c.customerName, c.totalCompras, c.cantidadVentas, c.totalFiado, c.totalPagado]);
   139→      });
   140→      filename = "clientes";
   141→    } else if (tab === "taxes" && data.byTipo) {
   142→      rows = [["Tipo", "Cantidad", "Neto", "IVA", "Total"]];
   143→      data.byTipo.forEach((t: any) => {
   144→        rows.push([t.tipo, t.count, t.netoGravado, t.iva, t.total]);
   145→      });
   146→      filename = "fiscal";
   147→    } else if (tab === "cashflow" && data.series) {
   148→      rows = [["Fecha", "Ventas Efectivo", "Cobros Cuenta", "Ing. Manuales", "Gastos", "Egresos Manuales", "Flujo Neto"]];
   149→      data.series.forEach((s: any) => {
   150→        rows.push([s.date, s.ventasEfectivo, s.cobrosCuenta, s.ingresosManuales, s.gastosEfectivo, s.egresosManuales, s.flujoNeto]);
   151→      });
   152→      filename = "flujo-caja";
   153→    }
   154→
   155→    const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n");
   156→    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
   157→    const url = URL.createObjectURL(blob);
   158→    const a = document.createElement("a");
   159→    a.href = url;
   160→    a.download = `reporte-${filename}-${new Date().toISOString().slice(0, 10)}.csv`;
   161→    a.click();
   162→    URL.revokeObjectURL(url);
   163→  }
   164→
   165→  const TABS: { key: TabKey; label: string; icon: any }[] = [
   166→    { key: "sales", label: "Ventas", icon: Receipt },
   167→    { key: "profits", label: "Ganancias", icon: TrendingUp },
   168→    { key: "taxes", label: "Fiscal", icon: BarChart3 },
   169→    { key: "products", label: "Productos", icon: Package },
   170→    { key: "customers", label: "Clientes", icon: Users },
   171→    { key: "cashflow", label: "Flujo Caja", icon: Wallet },
   172→  ];
   173→
   174→  return (
   175→    <div className="p-4 space-y-4">
   176→      <div className="flex flex-wrap items-center justify-between gap-3">
   177→        <div>
   178→          <h1 className="text-2xl font-bold flex items-center gap-2">
   179→            <BarChart3 className="w-6 h-6 text-indigo-600" />
   180→            Reportes
   181→          </h1>
   182→          <p className="text-sm text-muted-foreground">
   183→            Análisis de ventas, ganancias, fiscal, productos y clientes
   184→          </p>
   185→        </div>
   186→        <Button variant="outline" onClick={exportarCSV} disabled={!data}>
   187→          <Download className="w-4 h-4 mr-2" />
   188→          Exportar CSV
   189→        </Button>
   190→      </div>
   191→
   192→      {/* Tabs */}
   193→      <div className="flex flex-wrap gap-2">
   194→        {TABS.map((t) => {
   195→          const Icon = t.icon;
   196→          return (
   197→            <Button
   198→              key={t.key}
   199→              variant={tab === t.key ? "default" : "outline"}
   200→              size="sm"
   201→              onClick={() => {
   202→                setTab(t.key);
   203→                setData(null); // reset inmediato para no renderizar data de otro tab
   204→              }}
   205→              className={tab === t.key ? "bg-indigo-600 hover:bg-indigo-700" : ""}
   206→            >
   207→              <Icon className="w-4 h-4 mr-1" />
   208→              {t.label}
   209→            </Button>
   210→          );
   211→        })}
   212→      </div>
   213→
   214→      {/* Filtros */}
   215→      <Card>
   216→        <CardContent className="pt-4 flex flex-wrap items-end gap-3">
   217→          <div>
   218→            <Label className="text-xs">Desde</Label>
   219→            <Input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className="w-[160px]" />
   220→          </div>
   221→          <div>
   222→            <Label className="text-xs">Hasta</Label>
   223→            <Input type="date" value={to} onChange={(e) => setTo(e.target.value)} className="w-[160px]" />
   224→          </div>
   225→          <div className="flex gap-1">
   226→            <Button size="sm" variant="outline" onClick={() => setPeriod(7)}>7 días</Button>
   227→            <Button size="sm" variant="outline" onClick={() => setPeriod(30)}>30 días</Button>
   228→            <Button size="sm" variant="outline" onClick={() => setPeriod(90)}>90 días</Button>
   229→            <Button size="sm" variant="outline" onClick={() => setPeriod(365)}>1 año</Button>
   230→          </div>
   231→        </CardContent>
   232→      </Card>
   233→
   234→      {/* Contenido */}
   235→      {loading ? (
   236→        <div className="flex items-center justify-center py-12">
   237→          <Loader2 className="w-8 h-8 animate-spin text-indigo-600" />
   238→        </div>
   239→      ) : !data ? (
   240→        <Card>
   241→          <CardContent className="text-center py-12 text-muted-foreground">
   242→            <Calendar className="w-12 h-12 mx-auto mb-2 opacity-30" />
   243→            <p>Selecciona un rango de fechas</p>
   244→          </CardContent>
   245→        </Card>
   246→      ) : (
   247→        <>
   248→          {/* VENTAS */}
   249→          {tab === "sales" && <SalesReport data={data} symbol={symbol} />}
   250→          {/* GANANCIAS */}
   251→          {tab === "profits" && <ProfitsReport data={data} symbol={symbol} />}
   252→          {/* FISCAL */}
   253→          {tab === "taxes" && <TaxesReport data={data} symbol={symbol} />}
   254→          {/* PRODUCTOS */}
   255→          {tab === "products" && <ProductsReport data={data} symbol={symbol} />}
   256→          {/* CLIENTES */}
   257→          {tab === "customers" && <CustomersReport data={data} symbol={symbol} />}
   258→          {/* FLUJO CAJA */}
   259→          {tab === "cashflow" && <CashFlowReport data={data} symbol={symbol} />}
   260→        </>
   261→      )}
   262→    </div>
   263→  );
   264→}

   266→// ===== REPORTE DE VENTAS =====
   267→function SalesReport({ data, symbol }: { data: any; symbol: string }) {
   268→  const summary = data?.summary || {};
   269→  const series = data?.series || [];
   270→  const byPaymentMethod = data?.byPaymentMethod || [];
   271→  const byUser = data?.byUser || [];
   272→  return (
   273→    <div className="space-y-4">
   274→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   275→        <StatCard label="Total ventas" value={formatCurrency(summary.totalAmount || 0, symbol)} icon={DollarSign} color="indigo" />
   276→        <StatCard label="Cantidad" value={(summary.totalSales || 0).toString()} icon={Receipt} color="blue" />
   277→        <StatCard label="Ticket promedio" value={formatCurrency(summary.averageTicket || 0, symbol)} icon={TrendingUp} color="purple" />
   278→        <StatCard label="Recargos" value={formatCurrency(summary.totalSurcharge || 0, symbol)} icon={TrendingUp} color="amber" />
   279→      </div>
   280→
   281→      <Card>
   282→        <CardHeader>
   283→          <CardTitle className="text-sm">Evolución de ventas</CardTitle>
   284→        </CardHeader>
   285→        <CardContent>
   286→          {series.length === 0 ? (
   287→            <p className="text-sm text-muted-foreground text-center py-4">Sin ventas en el período</p>
   288→          ) : (
   289→            <ResponsiveContainer width="100%" height={300}>
   290→              <LineChart data={series}>
   291→                <CartesianGrid strokeDasharray="3 3" />
   292→                <XAxis dataKey="period" fontSize={11} />
   293→                <YAxis fontSize={11} />
   294→                <Tooltip
   295→                  formatter={(v: any, name: string) => [formatCurrency(v, symbol), name === "total" ? "Total" : name]}
   296→                />
   297→                <Legend />
   298→                <Line type="monotone" dataKey="total" stroke="#6366f1" strokeWidth={2} name="Total" />
   299→                <Line type="monotone" dataKey="count" stroke="#3b82f6" strokeWidth={1} name="Cantidad" hide />
   300→              </LineChart>
   301→            </ResponsiveContainer>
   302→          )}
   303→        </CardContent>
   304→      </Card>
   305→
   306→      <div className="grid md:grid-cols-2 gap-4">
   307→        <Card>
   308→          <CardHeader>
   309→            <CardTitle className="text-sm">Por método de pago</CardTitle>
   310→          </CardHeader>
   311→          <CardContent>
   312→            {byPaymentMethod.length === 0 ? (
   313→              <p className="text-sm text-muted-foreground text-center py-4">Sin datos</p>
   314→            ) : (
   315→              <ResponsiveContainer width="100%" height={250}>
   316→                <PieChart>
   317→                  <Pie
   318→                    data={byPaymentMethod}
   319→                    dataKey="total"
   320→                    nameKey="method"
   321→                    cx="50%"
   322→                    cy="50%"
   323→                    outerRadius={80}
   324→                    label={(e: any) => `${e.method}`}
   325→                  >
   326→                    {byPaymentMethod.map((_: any, i: number) => (
   327→                      <Cell key={i} fill={COLORS[i % COLORS.length]} />
   328→                    ))}
   329→                  </Pie>
   330→                  <Tooltip formatter={(v: any) => formatCurrency(v, symbol)} />
   331→                </PieChart>
   332→              </ResponsiveContainer>
   333→            )}
   334→            <Table>
   335→              <TableHeader>
   336→                <TableRow>
   337→                  <TableHead>Método</TableHead>
   338→                  <TableHead className="text-right">Cantidad</TableHead>
   339→                  <TableHead className="text-right">Total</TableHead>
   340→                </TableRow>
   341→              </TableHeader>
   342→              <TableBody>
   343→                {byPaymentMethod.map((m: any) => (
   344→                  <TableRow key={m.method}>
   345→                    <TableCell className="font-medium">{m.method}</TableCell>
   346→                    <TableCell className="text-right">{m.count}</TableCell>
   347→                    <TableCell className="text-right">{formatCurrency(m.total, symbol)}</TableCell>
   348→                  </TableRow>
   349→                ))}
   350→                {byPaymentMethod.length === 0 && (
   351→                  <TableRow>
   352→                    <TableCell colSpan={3} className="text-center text-muted-foreground">
   353→                      Sin datos
   354→                    </TableCell>
   355→                  </TableRow>
   356→                )}
   357→              </TableBody>
   358→            </Table>
   359→          </CardContent>
   360→        </Card>
   361→
   362→        <Card>
   363→          <CardHeader>
   364→            <CardTitle className="text-sm">Por vendedor</CardTitle>
   365→          </CardHeader>
   366→          <CardContent>
   367→            {byUser.length === 0 ? (
   368→              <p className="text-sm text-muted-foreground text-center py-4">Sin datos</p>
   369→            ) : (
   370→              <ResponsiveContainer width="100%" height={250}>
   371→                <BarChart data={byUser}>
   372→                  <CartesianGrid strokeDasharray="3 3" />
   373→                  <XAxis dataKey="userName" fontSize={11} />
   374→                  <YAxis fontSize={11} />
   375→                  <Tooltip formatter={(v: any) => formatCurrency(v, symbol)} />
   376→                  <Bar dataKey="total" fill="#6366f1" />
   377→                </BarChart>
   378→              </ResponsiveContainer>
   379→            )}
   380→            <Table>
   381→              <TableHeader>
   382→                <TableRow>
   383→                  <TableHead>Vendedor</TableHead>
   384→                  <TableHead className="text-right">Cantidad</TableHead>
   385→                  <TableHead className="text-right">Total</TableHead>
   386→                </TableRow>
   387→              </TableHeader>
   388→              <TableBody>
   389→                {byUser.map((u: any) => (
   390→                  <TableRow key={u.userId}>
   391→                    <TableCell className="font-medium">{u.userName}</TableCell>
   392→                    <TableCell className="text-right">{u.count}</TableCell>
   393→                    <TableCell className="text-right">{formatCurrency(u.total, symbol)}</TableCell>
   394→                  </TableRow>
   395→                ))}
   396→                {byUser.length === 0 && (
   397→                  <TableRow>
   398→                    <TableCell colSpan={3} className="text-center text-muted-foreground">
   399→                      Sin datos
   400→                    </TableCell>
   401→                  </TableRow>
   402→                )}
   403→              </TableBody>
   404→            </Table>
   405→          </CardContent>
   406→        </Card>
   407→      </div>
   408→    </div>
   409→  );
   410→}

   412→// ===== REPORTE DE GANANCIAS =====

   415→  const series = data?.series || [];
   416→  const expensesByCategory = data?.expensesByCategory || [];
   417→  const gananciaNeta = summary.gananciaNeta || 0;
   418→  return (
   419→    <div className="space-y-4">
   420→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   421→        <StatCard label="Ganancia bruta" value={formatCurrency(summary.gananciaBruta || 0, symbol)} icon={TrendingUp} color="indigo" />

   425→      </div>
   426→
   427→      <Card>
   428→        <CardHeader>
   429→          <CardTitle className="text-sm">Ventas vs Costo vs Gastos</CardTitle>

   432→          {series.length === 0 ? (
   433→            <p className="text-sm text-muted-foreground text-center py-4">Sin datos en el período</p>

   435→            <ResponsiveContainer width="100%" height={320}>

   437→                <CartesianGrid strokeDasharray="3 3" />

   451→      <Card>
   452→        <CardHeader>
   453→          <CardTitle className="text-sm">Gastos por categoría</CardTitle>
   454→        </CardHeader>
   455→        <CardContent>
   456→          {expensesByCategory.length === 0 ? (
   457→            <p className="text-sm text-muted-foreground text-center py-4">Sin gastos en el período</p>
   458→          ) : (
   459→            <ResponsiveContainer width="100%" height={250}>
   460→              <PieChart>
   461→                <Pie
   462→                  data={expensesByCategory}
   463→                  dataKey="amount"
   464→                  nameKey="category"
   465→                  cx="50%"
   466→                  cy="50%"
   467→                  outerRadius={80}
   468→                  label={(e: any) => `${e.category}`}
   469→                >
   470→                  {expensesByCategory.map((_: any, i: number) => (
   471→                    <Cell key={i} fill={COLORS[i % COLORS.length]} />
   472→                  ))}
   473→                </Pie>
   474→                <Tooltip formatter={(v: any) => formatCurrency(v, symbol)} />
   475→              </PieChart>
   476→            </ResponsiveContainer>
   477→          )}
   478→        </CardContent>
   479→      </Card>
   480→
   481→      <Card>
   482→        <CardContent className="pt-4 grid grid-cols-2 md:grid-cols-4 gap-3 text-sm">
   483→          <div>
   484→            <p className="text-xs text-muted-foreground">Total ventas</p>
   485→            <p className="font-bold">{formatCurrency(summary.totalVentas || 0, symbol)}</p>
   486→          </div>
   487→          <div>
   488→            <p className="text-xs text-muted-foreground">Total costo</p>
   489→            <p className="font-bold text-red-600">{formatCurrency(summary.totalCosto || 0, symbol)}</p>
   490→          </div>
   491→          <div>
   492→            <p className="text-xs text-muted-foreground">Total descuentos</p>
   493→            <p className="font-bold text-amber-600">{formatCurrency(summary.totalDescuentos || 0, symbol)}</p>
   494→          </div>
   495→          <div>
   496→            <p className="text-xs text-muted-foreground">Margen bruto</p>
   497→            <p className="font-bold text-indigo-600">{summary.margenBruto || 0}%</p>
   498→          </div>
   499→        </CardContent>
   500→      </Card>
   501→    </div>
   502→  );
   503→}

   505→// ===== REPORTE FISCAL =====
   506→function TaxesReport({ data, symbol }: { data: any; symbol: string }) {
   507→  const summary = data?.summary || {};
   508→  const byTipo = data?.byTipo || [];
   509→  return (
   510→    <div className="space-y-4">
   511→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">

   820→  };
   821→  return (
   822→    <Card>
   823→      <CardContent className="pt-4">
   824→        <div className="flex items-center gap-2 mb-1">
   825→          <div className={`w-7 h-7 rounded-full flex items-center justify-center ${colorMap[color] || colorMap.indigo}`}>
   826→            <Icon className="w-4 h-4" />

   829→        </div>

   831→      </CardContent>
   832→    </Card>
