     1→"use client";
     2→
     3→import { useEffect, useState, useMemo, useCallback } from "react";
     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     5→import { Button } from "@/components/ui/button";
     6→import { Input } from "@/components/ui/input";
     7→import { Label } from "@/components/ui/label";
     8→import { Badge } from "@/components/ui/badge";
     9→import {
    10→  Select,
    11→  SelectContent,
    12→  SelectItem,
    13→  SelectTrigger,
    14→  SelectValue,
    15→} from "@/components/ui/select";
    16→import {
    17→  Sheet,
    18→  SheetContent,
    19→  SheetDescription,
    20→  SheetFooter,
    21→  SheetHeader,
    22→  SheetTitle,
    23→} from "@/components/ui/sheet";
    24→import { toast } from "sonner";
    25→import {
    26→  Search,
    27→  ShoppingCart,
    28→  Trash2,
    29→  Plus,
    30→  Minus,
    31→  Barcode,
    32→  Loader2,
    33→  CheckCircle2,
    34→  Printer,
    35→  X,
    36→} from "lucide-react";
    37→import { useAppStore } from "@/store/app-store";
    38→import { formatCurrency, unitLabel } from "@/lib/constants";
    39→
    40→interface Product {
    41→  id: string;
    42→  name: string;
    43→  barcode?: string;
    44→  salePrice: number;
    45→  stock: number;
    46→  unit: string;
    47→  category?: { id: string; name: string };
    48→  categoryId?: string;
    49→}
    50→
    51→interface CartItem {
    52→  product: Product;
    53→  qty: number;
    54→}
    55→
    56→interface Customer {
    57→  id: string;
    58→  name: string;
    59→  phone?: string;
    60→}
    61→
    62→interface PaymentMethod {
    63→  id: string;
    64→  name: string;
    65→  type: string;
    66→  surcharge: number;
    67→  active: boolean;
    68→  isDefault: boolean;
    69→}
    70→
    71→export function PosView() {
    72→  const { store, user } = useAppStore();
    73→  const [products, setProducts] = useState<Product[]>([]);
    74→  const [categories, setCategories] = useState<any[]>([]);
    75→  const [customers, setCustomers] = useState<Customer[]>([]);
    76→  const [paymentMethods, setPaymentMethods] = useState<PaymentMethod[]>([]);
    77→  const [loading, setLoading] = useState(true);
    78→  const [search, setSearch] = useState("");
    79→  const [filterCat, setFilterCat] = useState("all");
    80→  const [cart, setCart] = useState<CartItem[]>([]);
    81→  const [customerId, setCustomerId] = useState<string>("");
    82→  const [discount, setDiscount] = useState(0);
    83→  const [paymentMethodId, setPaymentMethodId] = useState<string>("");
    84→  const [notes, setNotes] = useState("");
    85→  const [checkoutOpen, setCheckoutOpen] = useState(false);
    86→  const [processing, setProcessing] = useState(false);
    87→  const [lastSale, setLastSale] = useState<any>(null);
    88→  const [receiptOpen, setReceiptOpen] = useState(false);
    89→
    90→  const symbol = store?.currencySymbol || "$";
    91→  const taxEnabled = store?.taxEnabled;
    92→  const taxRate = store?.taxRate || 0;
    93→  const isCajero = user?.role === "CAJERO";
    94→
    95→  useEffect(() => {
    96→    Promise.all([
    97→      fetch("/api/products").then((r) => r.json()),
    98→      fetch("/api/categories").then((r) => r.json()),
    99→      fetch("/api/customers").then((r) => r.json()),
   100→      fetch("/api/payment-methods").then((r) => r.json()),
   101→    ]).then(([p, c, cust, pm]) => {
   102→      setProducts(p.filter((x: Product) => x.active));
   103→      setCategories(c);
   104→      setCustomers(cust);
   105→      const activePM = (pm as PaymentMethod[]).filter((m) => m.active);
   106→      setPaymentMethods(activePM);
   107→      const def = activePM.find((m) => m.isDefault) || activePM[0];
   108→      if (def) setPaymentMethodId(def.id);
   109→      setLoading(false);
   110→    });
   111→  }, []);
   112→
   113→  const filtered = useMemo(() => {
   114→    return products.filter((p) => {
   115→      const matchSearch =
   116→        !search ||
   117→        p.name.toLowerCase().includes(search.toLowerCase()) ||
   118→        p.barcode?.includes(search);
   119→      const matchCat = filterCat === "all" || p.categoryId === filterCat;
   120→      return matchSearch && matchCat;
   121→    });
   122→  }, [products, search, filterCat]);
   123→
   124→  const addToCart = useCallback((product: Product) => {
   125→    setCart((prev) => {
   126→      const existing = prev.find((i) => i.product.id === product.id);
   127→      if (existing) {
   128→        if (existing.qty >= product.stock) {
   129→          toast.error(`Stock máximo alcanzado para ${product.name}`);
   130→          return prev;
   131→        }
   132→        return prev.map((i) =>
   133→          i.product.id === product.id ? { ...i, qty: i.qty + 1 } : i
   134→        );
   135→      }
   136→      if (product.stock <= 0) {
   137→        toast.error(`${product.name} sin stock`);
   138→        return prev;
   139→      }
   140→      return [...prev, { product, qty: 1 }];
   141→    });
   142→  }, []);
   143→
   144→  // Atajo: Enter en el buscador si solo hay 1 resultado
   145→  function handleSearchEnter(e: React.KeyboardEvent) {
   146→    if (e.key === "Enter" && filtered.length === 1) {
   147→      addToCart(filtered[0]);
   148→      setSearch("");
   149→    }
   150→  }
   151→
   152→  function updateQty(productId: string, delta: number) {
   153→    setCart((prev) =>
   154→      prev
   155→        .map((i) => {
   156→          if (i.product.id !== productId) return i;
   157→          const nextQty = i.qty + delta;
   158→          if (nextQty > i.product.stock) {
   159→            toast.error(`Solo hay ${i.product.stock} ${unitLabel(i.product.unit).toLowerCase()}s disponibles`);
   160→            return i;
   161→          }
   162→          return { ...i, qty: nextQty };
   163→        })
   164→        .filter((i) => i.qty > 0)
   165→    );
   166→  }
   167→
   168→  function setQty(productId: string, qty: number) {
   169→    setCart((prev) =>
   170→      prev
   171→        .map((i) =>
   172→          i.product.id === productId
   173→            ? { ...i, qty: Math.max(0, Math.min(qty, i.product.stock)) }
   174→            : i
   175→        )
   176→        .filter((i) => i.qty > 0)
   177→    );
   178→  }
   179→
   180→  function removeItem(productId: string) {
   181→    setCart((prev) => prev.filter((i) => i.product.id !== productId));
   182→  }
   183→
   184→  function clearCart() {
   185→    setCart([]);
   186→    setCustomerId("");
   187→    setDiscount(0);
   188→    setNotes("");
   189→  }
   190→
   191→  const subtotal = cart.reduce((s, i) => s + i.product.salePrice * i.qty, 0);
   192→  const discountAmount = Math.min(discount, subtotal);
   193→  const taxable = subtotal - discountAmount;
   194→  const taxAmount = taxEnabled ? taxable * (taxRate / 100) : 0;
   195→  const selectedMethod = paymentMethods.find((m) => m.id === paymentMethodId);
   196→  const surchargeRate = selectedMethod?.surcharge || 0;
   197→  const surchargeAmount = (taxable + taxAmount) * (surchargeRate / 100);
   198→  const total = taxable + taxAmount + surchargeAmount;
   199→
   200→  async function processSale() {
   201→    if (cart.length === 0) {
   202→      toast.error("El carrito está vacío");
   203→      return;
   204→    }
   205→    if (!paymentMethodId) {
   206→      toast.error("Seleccioná un método de pago");
   207→      return;
   208→    }
   209→    setProcessing(true);
   210→    try {
   211→      const res = await fetch("/api/sales", {
   212→        method: "POST",
   213→        headers: { "Content-Type": "application/json" },
   214→        body: JSON.stringify({
   215→          items: cart.map((i) => ({
   216→            productId: i.product.id,
   217→            quantity: i.qty,
   218→          })),
   219→          customerId: customerId || null,
   220→          discount: discountAmount,
   221→          paymentMethodId,
   222→          notes,
   223→          taxRate: taxEnabled ? taxRate : 0,
   224→        }),
   225→      });
   226→      const data = await res.json();
   227→      if (!res.ok) throw new Error(data.error);
   228→      toast.success("Venta registrada!");
   229→      setLastSale({
   230→        ...data,
   231→        items: cart,
   232→        customer: customers.find((c) => c.id === customerId),
   233→        paymentMethod: selectedMethod,
   234→        discount: discountAmount,
   235→        tax: taxAmount,
   236→        surcharge: surchargeAmount,
   237→        total,
   238→        subtotal,
   239→      });
   240→      setCheckoutOpen(false);
   241→      setReceiptOpen(true);
   242→      clearCart();
   243→      // Refrescar stock
   244→      const refreshed = await fetch("/api/products").then((r) => r.json());
   245→      setProducts(refreshed.filter((x: Product) => x.active));
   246→    } catch (e: any) {
   247→      toast.error(e.message);
   248→    } finally {
   249→      setProcessing(false);
   250→    }
   251→  }
   252→
   253→  return (
   254→    <div className="space-y-4 h-full">
   255→      <div>
   256→        <h1 className="text-2xl font-bold tracking-tight">Punto de Venta</h1>
   257→        <p className="text-sm text-muted-foreground">
   258→          Buscá productos, agregalos al carrito y registrá la venta
   259→        </p>
   260→      </div>
   261→
   262→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-4">
   263→        {/* Productos - 3 cols */}
   264→        <div className="lg:col-span-3 space-y-3">
   265→          <Card>
   266→            <CardContent className="p-3 space-y-2">
   267→              <div className="relative">
   268→                <Barcode className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   269→                <Input
   270→                  placeholder="Escanear código o buscar producto..."
   271→                  value={search}
   272→                  onChange={(e) => setSearch(e.target.value)}
   273→                  onKeyDown={handleSearchEnter}
   274→                  className="pl-9 h-11 text-base"
   275→                  autoFocus
   276→                />
   277→              </div>
   278→              <div className="flex gap-2 overflow-x-auto pb-1">
   279→                <Button
   280→                  variant={filterCat === "all" ? "default" : "outline"}
   281→                  size="sm"
   282→                  onClick={() => setFilterCat("all")}
   283→                  className={
   284→                    filterCat === "all" ? "bg-emerald-600 hover:bg-emerald-700" : ""
   285→                  }
   286→                >
   287→                  Todos
   288→                </Button>
   289→                {categories.map((c) => (
   290→                  <Button
   291→                    key={c.id}
   292→                    variant={filterCat === c.id ? "default" : "outline"}
   293→                    size="sm"
   294→                    onClick={() => setFilterCat(c.id)}
   295→                    className={
   296→                      filterCat === c.id ? "bg-emerald-600 hover:bg-emerald-700 whitespace-nowrap" : "whitespace-nowrap"
   297→                    }
   298→                  >
   299→                    {c.name}
   300→                  </Button>
   301→                ))}
   302→              </div>
   303→            </CardContent>
   304→          </Card>
   305→
   306→          {loading ? (
   307→            <div className="p-12 text-center">
   308→              <Loader2 className="w-6 h-6 animate-spin mx-auto text-emerald-600" />
   309→            </div>
   310→          ) : (
   311→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2 max-h-[60vh] overflow-y-auto pr-1">
   312→              {filtered.map((p) => (
   313→                <button
   314→                  key={p.id}
   315→                  onClick={() => addToCart(p)}
   316→                  disabled={p.stock <= 0}
   317→                  className="group text-left p-3 rounded-lg border border-emerald-100 bg-white hover:border-emerald-400 hover:shadow-md transition-all disabled:opacity-50 disabled:cursor-not-allowed"
   318→                >
   319→                  <p className="text-sm font-medium line-clamp-2 min-h-[2.5rem]">
   320→                    {p.name}
   321→                  </p>
   322→                  <p className="text-sm font-bold text-emerald-700 mt-1">
   323→                    {formatCurrency(p.salePrice, symbol)}
   324→                  </p>
   325→                  <p
   326→                    className={`text-xs mt-0.5 ${
   327→                      p.stock <= 0
   328→                        ? "text-red-600"
   329→                        : p.stock <= 5
   330→                        ? "text-amber-600"
   331→                        : "text-muted-foreground"
   332→                    }`}
   333→                  >
   334→                    {p.stock <= 0
   335→                      ? "Sin stock"
   336→                      : `${p.stock} ${p.unit === "KG" ? "kg" : "u"}`}
   337→                  </p>
   338→                </button>
   339→              ))}
   340→              {filtered.length === 0 && (
   341→                <div className="col-span-full p-8 text-center text-muted-foreground">
   342→                  <Search className="w-8 h-8 mx-auto mb-2 opacity-50" />
   343→                  <p className="text-sm">No se encontraron productos</p>
   344→                </div>
   345→              )}
   346→            </div>
   347→          )}
   348→        </div>
   349→
   350→        {/* Carrito - 2 cols */}
   351→        <div className="lg:col-span-2">
   352→          <Card className="sticky top-20">
   353→            <CardHeader className="pb-3">
   354→              <div className="flex items-center justify-between">
   355→                <CardTitle className="text-base flex items-center gap-2">
   356→                  <ShoppingCart className="w-4 h-4" />
   357→                  Carrito ({cart.length})
   358→                </CardTitle>
   359→                {cart.length > 0 && (
   360→                  <Button variant="ghost" size="sm" onClick={clearCart}>
   361→                    Vaciar
   362→                  </Button>
   363→                )}
   364→              </div>
   365→            </CardHeader>
   366→            <CardContent className="space-y-3">
   367→              {cart.length === 0 ? (
   368→                <div className="py-12 text-center text-muted-foreground">
   369→                  <ShoppingCart className="w-10 h-10 mx-auto opacity-30 mb-2" />
   370→                  <p className="text-sm">Agregá productos al carrito</p>
   371→                </div>
   372→              ) : (
   373→                <div className="space-y-2 max-h-[40vh] overflow-y-auto pr-1">
   374→                  {cart.map((i) => (
   375→                    <div
   376→                      key={i.product.id}
   377→                      className="flex items-center gap-2 p-2 rounded-md bg-muted/50"
   378→                    >
   379→                      <div className="flex-1 min-w-0">
   380→                        <p className="text-sm font-medium truncate">{i.product.name}</p>
   381→                        <p className="text-xs text-muted-foreground">
   382→                          {formatCurrency(i.product.salePrice, symbol)} ·{" "}
   383→                          {i.product.unit === "KG" ? "kg" : "u"}
   384→                        </p>
   385→                      </div>
   386→                      <div className="flex items-center gap-1">
   387→                        <Button
   388→                          variant="outline"
   389→                          size="icon"
   390→                          className="h-7 w-7"
   391→                          onClick={() => updateQty(i.product.id, -1)}
   392→                        >
   393→                          <Minus className="w-3 h-3" />
   394→                        </Button>
   395→                        <Input
   396→                          type="number"
   397→                          value={i.qty}
   398→                          onChange={(e) =>
   399→                            setQty(i.product.id, Number(e.target.value))
   400→                          }
   401→                          className="h-7 w-12 text-center px-1"
   402→                        />
   403→                        <Button
   404→                          variant="outline"
   405→                          size="icon"
   406→                          className="h-7 w-7"
   407→                          onClick={() => updateQty(i.product.id, 1)}
   408→                        >
   409→                          <Plus className="w-3 h-3" />
   410→                        </Button>
   411→                        <Button
   412→                          variant="ghost"
   413→                          size="icon"
   414→                          className="h-7 w-7 text-red-600"
   415→                          onClick={() => removeItem(i.product.id)}
   416→                        >
   417→                          <X className="w-3 h-3" />
   418→                        </Button>
   419→                      </div>
   420→                      <p className="text-sm font-semibold w-20 text-right">
   421→                        {formatCurrency(i.product.salePrice * i.qty, symbol)}
   422→                      </p>
   423→                    </div>
   424→                  ))}
   425→                </div>
   426→              )}
   427→
   428→              {cart.length > 0 && (
   429→                <div className="space-y-2 pt-3 border-t">
   430→                  <div className="flex justify-between text-sm">
   431→                    <span className="text-muted-foreground">Subtotal</span>
   432→                    <span>{formatCurrency(subtotal, symbol)}</span>
   433→                  </div>
   434→                  <div className="flex justify-between text-sm items-center">
   435→                    <span className="text-muted-foreground">Descuento</span>
   436→                    <Input
   437→                      type="number"
   438→                      value={discount || ""}
   439→                      placeholder="0"
   440→                      onChange={(e) => setDiscount(Number(e.target.value))}
   441→                      className="h-7 w-24 text-right"
   442→                    />
   443→                  </div>
   444→                  {taxEnabled && taxRate > 0 && (
   445→                    <div className="flex justify-between text-sm">
   446→                      <span className="text-muted-foreground">
   447→                        Impuesto ({taxRate}%)
   448→                      </span>
   449→                      <span>{formatCurrency(taxAmount, symbol)}</span>
   450→                    </div>
   451→                  )}
   452→                  {surchargeAmount > 0 && (
   453→                    <div className="flex justify-between text-sm text-amber-700">
   454→                      <span>
   455→                        Recargo {selectedMethod?.name} ({surchargeRate}%)
   456→                      </span>
   457→                      <span>+{formatCurrency(surchargeAmount, symbol)}</span>
   458→                    </div>
   459→                  )}
   460→                  <div className="flex justify-between text-lg font-bold pt-2 border-t">
   461→                    <span>Total</span>
   462→                    <span className="text-emerald-700">
   463→                      {formatCurrency(total, symbol)}
   464→                    </span>
   465→                  </div>
   466→
   467→                  <Button
   468→                    className="w-full bg-emerald-600 hover:bg-emerald-700 h-11 text-base"
   469→                    onClick={() => setCheckoutOpen(true)}
   470→                  >
   471→                    Cobrar {formatCurrency(total, symbol)}
   472→                  </Button>
   473→                </div>
   474→              )}
   475→            </CardContent>
   476→          </Card>
   477→        </div>
   478→      </div>
   479→
   480→      {/* Checkout sheet */}
   481→      <Sheet open={checkoutOpen} onOpenChange={setCheckoutOpen}>
   482→        <SheetContent className="sm:max-w-md overflow-y-auto">
   483→          <SheetHeader>
   484→            <SheetTitle>Confirmar venta</SheetTitle>
   485→            <SheetDescription>
   486→              Revisá los detalles y registrá la venta
   487→            </SheetDescription>
   488→          </SheetHeader>
   489→
   490→          <div className="space-y-4 py-4">
   491→            <div className="space-y-2">
   492→              <Label>Cliente (opcional)</Label>
   493→              <Select
   494→                value={customerId || "none"}
   495→                onValueChange={(v) =>
   496→                  setCustomerId(v === "none" ? "" : v)
   497→                }
   498→              >
   499→                <SelectTrigger>
   500→                  <SelectValue placeholder="Cliente mostrador" />
   501→                </SelectTrigger>
   502→                <SelectContent>
   503→                  <SelectItem value="none">Cliente mostrador</SelectItem>
   504→                  {customers.map((c) => (
   505→                    <SelectItem key={c.id} value={c.id}>
   506→                      {c.name} {c.phone && `(${c.phone})`}
   507→                    </SelectItem>
   508→                  ))}
   509→                </SelectContent>
   510→              </Select>
   511→            </div>
   512→
   513→            <div className="space-y-2">
   514→              <Label>Método de pago</Label>
   515→              <Select
   516→                value={paymentMethodId}
   517→                onValueChange={setPaymentMethodId}
   518→              >
   519→                <SelectTrigger>
   520→                  <SelectValue placeholder="Seleccionar método..." />
   521→                </SelectTrigger>
   522→                <SelectContent>
   523→                  {paymentMethods.map((m) => (
   524→                    <SelectItem key={m.id} value={m.id}>
   525→                      {m.name}
   526→                      {m.surcharge > 0 && ` (+${m.surcharge}%)`}
   527→                      {m.isDefault && " · predeterminado"}
   528→                    </SelectItem>
   529→                  ))}
   530→                </SelectContent>
   531→              </Select>
   532→              {paymentMethods.length === 0 && (
   533→                <p className="text-xs text-amber-600">
   534→                  No hay métodos de pago configurados. Pedile al admin que los
   535→                  cargue en Configuración.
   536→                </p>
   537→              )}
   538→            </div>
   539→
   540→            <div className="space-y-2">
   541→              <Label>Notas (opcional)</Label>
   542→              <Input
   543→                value={notes}
   544→                onChange={(e) => setNotes(e.target.value)}
   545→                placeholder="Ej: Entrega parcial, pedido especial..."
   546→              />
   547→            </div>
   548→
   549→            <div className="p-3 rounded-md bg-muted/50 space-y-1 text-sm">
   550→              <div className="flex justify-between">
   551→                <span className="text-muted-foreground">Subtotal</span>
   552→                <span>{formatCurrency(subtotal, symbol)}</span>
   553→              </div>
   554→              {discountAmount > 0 && (
   555→                <div className="flex justify-between">
   556→                  <span className="text-muted-foreground">Descuento</span>
   557→                  <span>-{formatCurrency(discountAmount, symbol)}</span>
   558→                </div>
   559→              )}
   560→              {taxAmount > 0 && (
   561→                <div className="flex justify-between">
   562→                  <span className="text-muted-foreground">
   563→                    Impuesto ({taxRate}%)
   564→                  </span>
   565→                  <span>{formatCurrency(taxAmount, symbol)}</span>
   566→                </div>
   567→              )}
   568→              {surchargeAmount > 0 && (
   569→                <div className="flex justify-between text-amber-700">
   570→                  <span>
   571→                    Recargo {selectedMethod?.name} ({surchargeRate}%)
   572→                  </span>
   573→                  <span>+{formatCurrency(surchargeAmount, symbol)}</span>
   574→                </div>
   575→              )}
   576→              <div className="flex justify-between text-base font-bold pt-1 border-t">
   577→                <span>Total</span>
   578→                <span className="text-emerald-700">
   579→                  {formatCurrency(total, symbol)}
   580→                </span>
   581→              </div>
   582→            </div>
   583→          </div>
