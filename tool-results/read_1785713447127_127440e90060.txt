     1→     1→"use client";
     2→     2→
     3→     3→import { useEffect, useState, useMemo, useCallback } from "react";
     4→     4→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     5→     5→import { Button } from "@/components/ui/button";
     6→     6→import { Input } from "@/components/ui/input";

     9→     9→import {

    16→    16→import {

    19→    19→  SheetDescription,
    20→    20→  SheetFooter,
    21→    21→  SheetHeader,
    22→    22→  SheetTitle,
    23→    23→} from "@/components/ui/sheet";
    24→    24→import { toast } from "sonner";
    25→    25→import {
    26→    26→  Search,
    27→    27→  ShoppingCart,
    28→    28→  Trash2,
    29→    29→  Plus,
    30→    30→  Minus,
    31→    31→  Barcode,
    32→    32→  Loader2,
    33→    33→  CheckCircle2,
    34→    34→  Printer,
    35→    35→  X,
    36→    36→} from "lucide-react";
    37→    37→import { useAppStore } from "@/store/app-store";
    38→    38→import { formatCurrency, unitLabel } from "@/lib/constants";
    39→    39→
    40→    40→interface Product {
    41→    41→  id: string;
    42→    42→  name: string;
    43→    43→  barcode?: string;
    44→    44→  salePrice: number;
    45→    45→  stock: number;
    46→    46→  unit: string;
    47→    47→  category?: { id: string; name: string };
    48→    48→  categoryId?: string;
    49→    49→}
    50→    50→
    51→    51→interface CartItem {
    52→    52→  product: Product;
    53→    53→  qty: number;
    54→    54→}
    55→    55→
    56→    56→interface Customer {
    57→    57→  id: string;
    58→    58→  name: string;
    59→    59→  phone?: string;
    60→    60→}
    61→    61→
    62→    62→interface PaymentMethod {
    63→    63→  id: string;
    64→    64→  name: string;
    65→    65→  type: string;
    66→    66→  surcharge: number;
    67→    67→  active: boolean;
    68→    68→  isDefault: boolean;
    69→    69→}
    70→    70→
    71→    71→export function PosView() {
    72→    72→  const { store, user } = useAppStore();
    73→    73→  const [products, setProducts] = useState<Product[]>([]);
    74→    74→  const [categories, setCategories] = useState<any[]>([]);
    75→    75→  const [customers, setCustomers] = useState<Customer[]>([]);
    76→    76→  const [paymentMethods, setPaymentMethods] = useState<PaymentMethod[]>([]);
    77→    77→  const [loading, setLoading] = useState(true);
    78→    78→  const [search, setSearch] = useState("");
    79→    79→  const [filterCat, setFilterCat] = useState("all");
    80→    80→  const [cart, setCart] = useState<CartItem[]>([]);
    81→    81→  const [customerId, setCustomerId] = useState<string>("");
    82→    82→  const [discount, setDiscount] = useState(0);
    83→    83→  const [paymentMethodId, setPaymentMethodId] = useState<string>("");
    84→    84→  const [notes, setNotes] = useState("");
    85→    85→  const [checkoutOpen, setCheckoutOpen] = useState(false);
    86→    86→  const [processing, setProcessing] = useState(false);
    87→    87→  const [lastSale, setLastSale] = useState<any>(null);
    88→    88→  const [receiptOpen, setReceiptOpen] = useState(false);
    89→    89→
    90→    90→  const symbol = store?.currencySymbol || "$";
    91→    91→  const taxEnabled = store?.taxEnabled;
    92→    92→  const taxRate = store?.taxRate || 0;
    93→    93→  const isCajero = user?.role === "CAJERO";
    94→    94→
    95→    95→  useEffect(() => {
    96→    96→    Promise.all([
    97→    97→      fetch("/api/products").then((r) => r.json()),
    98→    98→      fetch("/api/categories").then((r) => r.json()),
    99→    99→      fetch("/api/customers").then((r) => r.json()),
   100→   100→      fetch("/api/payment-methods").then((r) => r.json()),
   101→   101→    ]).then(([p, c, cust, pm]) => {
   102→   102→      setProducts(p.filter((x: Product) => x.active));
   103→   103→      setCategories(c);
   104→   104→      setCustomers(cust);
   105→   105→      const activePM = (pm as PaymentMethod[]).filter((m) => m.active);
   106→   106→      setPaymentMethods(activePM);
   107→   107→      const def = activePM.find((m) => m.isDefault) || activePM[0];
   108→   108→      if (def) setPaymentMethodId(def.id);
   109→   109→      setLoading(false);
   110→   110→    });
   111→   111→  }, []);
   112→   112→
   113→   113→  const filtered = useMemo(() => {
   114→   114→    return products.filter((p) => {
   115→   115→      const matchSearch =
   116→   116→        !search ||
   117→   117→        p.name.toLowerCase().includes(search.toLowerCase()) ||
   118→   118→        p.barcode?.includes(search);
   119→   119→      const matchCat = filterCat === "all" || p.categoryId === filterCat;
   120→   120→      return matchSearch && matchCat;
   121→   121→    });
   122→   122→  }, [products, search, filterCat]);
   123→   123→
   124→   124→  const addToCart = useCallback((product: Product) => {
   125→   125→    setCart((prev) => {
   126→   126→      const existing = prev.find((i) => i.product.id === product.id);
   127→   127→      if (existing) {
   128→   128→        if (existing.qty >= product.stock) {
   129→   129→          toast.error(`Stock máximo alcanzado para ${product.name}`);
   130→   130→          return prev;
   131→   131→        }
   132→   132→        return prev.map((i) =>
   133→   133→          i.product.id === product.id ? { ...i, qty: i.qty + 1 } : i
   134→   134→        );
   135→   135→      }
   136→   136→      if (product.stock <= 0) {
   137→   137→        toast.error(`${product.name} sin stock`);
   138→   138→        return prev;
   139→   139→      }
   140→   140→      return [...prev, { product, qty: 1 }];
   141→   141→    });
   142→   142→  }, []);
   143→   143→
   144→   144→  // Atajo: Enter en el buscador si solo hay 1 resultado
   145→   145→  function handleSearchEnter(e: React.KeyboardEvent) {
   146→   146→    if (e.key === "Enter" && filtered.length === 1) {
   147→   147→      addToCart(filtered[0]);
   148→   148→      setSearch("");
   149→   149→    }
   150→   150→  }
   151→   151→
   152→   152→  function updateQty(productId: string, delta: number) {
   153→   153→    setCart((prev) =>
   154→   154→      prev
   155→   155→        .map((i) => {
   156→   156→          if (i.product.id !== productId) return i;
   157→   157→          const nextQty = i.qty + delta;
   158→   158→          if (nextQty > i.product.stock) {
   159→   159→            toast.error(`Solo hay ${i.product.stock} ${unitLabel(i.product.unit).toLowerCase()}s disponibles`);
   160→   160→            return i;
   161→   161→          }
   162→   162→          return { ...i, qty: nextQty };
   163→   163→        })
   164→   164→        .filter((i) => i.qty > 0)
   165→   165→    );
   166→   166→  }
   167→   167→
   168→   168→  function setQty(productId: string, qty: number) {
   169→   169→    setCart((prev) =>
   170→   170→      prev
   171→   171→        .map((i) =>
   172→   172→          i.product.id === productId
   173→   173→            ? { ...i, qty: Math.max(0, Math.min(qty, i.product.stock)) }
   174→   174→            : i
   175→   175→        )
   176→   176→        .filter((i) => i.qty > 0)
   177→   177→    );
   178→   178→  }
   179→   179→
   180→   180→  function removeItem(productId: string) {
   181→   181→    setCart((prev) => prev.filter((i) => i.product.id !== productId));
   182→   182→  }
   183→   183→
   184→   184→  function clearCart() {
   185→   185→    setCart([]);
   186→   186→    setCustomerId("");
   187→   187→    setDiscount(0);
   188→   188→    setNotes("");
   189→   189→  }
   190→   190→
   191→   191→  const subtotal = cart.reduce((s, i) => s + i.product.salePrice * i.qty, 0);
   192→   192→  const discountAmount = Math.min(discount, subtotal);
   193→   193→  const taxable = subtotal - discountAmount;
   194→   194→  const taxAmount = taxEnabled ? taxable * (taxRate / 100) : 0;
   195→   195→  const selectedMethod = paymentMethods.find((m) => m.id === paymentMethodId);
   196→   196→  const surchargeRate = selectedMethod?.surcharge || 0;
   197→   197→  const surchargeAmount = (taxable + taxAmount) * (surchargeRate / 100);
   198→   198→  const total = taxable + taxAmount + surchargeAmount;
   199→   199→
   200→   200→  async function processSale() {
   201→   201→    if (cart.length === 0) {
   202→   202→      toast.error("El carrito está vacío");
   203→   203→      return;
   204→   204→    }
   205→   205→    if (!paymentMethodId) {
   206→   206→      toast.error("Seleccioná un método de pago");
   207→   207→      return;
   208→   208→    }
   209→   209→    setProcessing(true);
   210→   210→    try {
   211→   211→      const res = await fetch("/api/sales", {
   212→   212→        method: "POST",
   213→   213→        headers: { "Content-Type": "application/json" },
   214→   214→        body: JSON.stringify({
   215→   215→          items: cart.map((i) => ({
   216→   216→            productId: i.product.id,
   217→   217→            quantity: i.qty,
   218→   218→          })),
   219→   219→          customerId: customerId || null,
   220→   220→          discount: discountAmount,
   221→   221→          paymentMethodId,
   222→   222→          notes,
   223→   223→          taxRate: taxEnabled ? taxRate : 0,
   224→   224→        }),
   225→   225→      });
   226→   226→      const data = await res.json();
   227→   227→      if (!res.ok) throw new Error(data.error);
   228→   228→      toast.success("Venta registrada!");
   229→   229→      setLastSale({
   230→   230→        ...data,
   231→   231→        items: cart,
   232→   232→        customer: customers.find((c) => c.id === customerId),
   233→   233→        paymentMethod: selectedMethod,
   234→   234→        discount: discountAmount,
   235→   235→        tax: taxAmount,
   236→   236→        surcharge: surchargeAmount,
   237→   237→        total,
   238→   238→        subtotal,
   239→   239→      });
   240→   240→      setCheckoutOpen(false);
   241→   241→      setReceiptOpen(true);
   242→   242→      clearCart();
   243→   243→      // Refrescar stock
   244→   244→      const refreshed = await fetch("/api/products").then((r) => r.json());
   245→   245→      setProducts(refreshed.filter((x: Product) => x.active));
   246→   246→    } catch (e: any) {
   247→   247→      toast.error(e.message);
   248→   248→    } finally {
   249→   249→      setProcessing(false);
   250→   250→    }
   251→   251→  }
   252→   252→
   253→   253→  return (
   254→   254→    <div className="space-y-4 h-full">
   255→   255→      <div>
   256→   256→        <h1 className="text-2xl font-bold tracking-tight">Punto de Venta</h1>
   257→   257→        <p className="text-sm text-muted-foreground">
   258→   258→          Buscá productos, agregalos al carrito y registrá la venta
   259→   259→        </p>
   260→   260→      </div>
   261→   261→
   262→   262→      <div className="grid grid-cols-1 lg:grid-cols-5 gap-4">
   263→   263→        {/* Productos - 3 cols */}
   264→   264→        <div className="lg:col-span-3 space-y-3">
   265→   265→          <Card>
   266→   266→            <CardContent className="p-3 space-y-2">
   267→   267→              <div className="relative">
   268→   268→                <Barcode className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   269→   269→                <Input
   270→   270→                  placeholder="Escanear código o buscar producto..."
   271→   271→                  value={search}
   272→   272→                  onChange={(e) => setSearch(e.target.value)}
   273→   273→                  onKeyDown={handleSearchEnter}
   274→   274→                  className="pl-9 h-11 text-base"
   275→   275→                  autoFocus
   276→   276→                />
   277→   277→              </div>
   278→   278→              <div className="flex gap-2 overflow-x-auto pb-1">
   279→   279→                <Button
   280→   280→                  variant={filterCat === "all" ? "default" : "outline"}
   281→   281→                  size="sm"
   282→   282→                  onClick={() => setFilterCat("all")}
   283→   283→                  className={
   284→   284→                    filterCat === "all" ? "bg-emerald-600 hover:bg-emerald-700" : ""
   285→   285→                  }
   286→   286→                >
   287→   287→                  Todos
   288→   288→                </Button>
   289→   289→                {categories.map((c) => (
   290→   290→                  <Button
   291→   291→                    key={c.id}
   292→   292→                    variant={filterCat === c.id ? "default" : "outline"}
   293→   293→                    size="sm"
   294→   294→                    onClick={() => setFilterCat(c.id)}
   295→   295→                    className={
   296→   296→                      filterCat === c.id ? "bg-emerald-600 hover:bg-emerald-700 whitespace-nowrap" : "whitespace-nowrap"
   297→   297→                    }
   298→   298→                  >
   299→   299→                    {c.name}
   300→   300→                  </Button>
   301→   301→                ))}
   302→   302→              </div>
   303→   303→            </CardContent>
   304→   304→          </Card>
   305→   305→
   306→   306→          {loading ? (
   307→   307→            <div className="p-12 text-center">
   308→   308→              <Loader2 className="w-6 h-6 animate-spin mx-auto text-emerald-600" />
   309→   309→            </div>
   310→   310→          ) : (
   311→   311→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-2 max-h-[60vh] overflow-y-auto pr-1">
   312→   312→              {filtered.map((p) => (
   313→   313→                <button
   314→   314→                  key={p.id}
   315→   315→                  onClick={() => addToCart(p)}
   316→   316→                  disabled={p.stock <= 0}
   317→   317→                  className="group text-left p-3 rounded-lg border border-emerald-100 bg-white hover:border-emerald-400 hover:shadow-md transition-all disabled:opacity-50 disabled:cursor-not-allowed"
   318→   318→                >
   319→   319→                  <p className="text-sm font-medium line-clamp-2 min-h-[2.5rem]">
   320→   320→                    {p.name}
   321→   321→                  </p>
   322→   322→                  <p className="text-sm font-bold text-emerald-700 mt-1">
   323→   323→                    {formatCurrency(p.salePrice, symbol)}
   324→   324→                  </p>
   325→   325→                  <p
   326→   326→                    className={`text-xs mt-0.5 ${
   327→   327→                      p.stock <= 0
   328→   328→                        ? "text-red-600"
   329→   329→                        : p.stock <= 5
   330→   330→                        ? "text-amber-600"
   331→   331→                        : "text-muted-foreground"
   332→   332→                    }`}
   333→   333→                  >
   334→   334→                    {p.stock <= 0
   335→   335→                      ? "Sin stock"
   336→   336→                      : `${p.stock} ${p.unit === "KG" ? "kg" : "u"}`}
   337→   337→                  </p>
   338→   338→                </button>
   339→   339→              ))}
   340→   340→              {filtered.length === 0 && (
   341→   341→                <div className="col-span-full p-8 text-center text-muted-foreground">
   342→   342→                  <Search className="w-8 h-8 mx-auto mb-2 opacity-50" />
   343→   343→                  <p className="text-sm">No se encontraron productos</p>
   344→   344→                </div>
   345→   345→              )}
   346→   346→            </div>
   347→   347→          )}
   348→   348→        </div>
   349→   349→
   350→   350→        {/* Carrito - 2 cols */}
   351→   351→        <div className="lg:col-span-2">
   352→   352→          <Card className="sticky top-20">
   353→   353→            <CardHeader className="pb-3">
   354→   354→              <div className="flex items-center justify-between">
   355→   355→                <CardTitle className="text-base flex items-center gap-2">
   356→   356→                  <ShoppingCart className="w-4 h-4" />
   357→   357→                  Carrito ({cart.length})
   358→   358→                </CardTitle>
   359→   359→                {cart.length > 0 && (
   360→   360→                  <Button variant="ghost" size="sm" onClick={clearCart}>
   361→   361→                    Vaciar
   362→   362→                  </Button>
   363→   363→                )}
   364→   364→              </div>
   365→   365→            </CardHeader>
   366→   366→            <CardContent className="space-y-3">
   367→   367→              {cart.length === 0 ? (
   368→   368→                <div className="py-12 text-center text-muted-foreground">
   369→   369→                  <ShoppingCart className="w-10 h-10 mx-auto opacity-30 mb-2" />
   370→   370→                  <p className="text-sm">Agregá productos al carrito</p>
   371→   371→                </div>
   372→   372→              ) : (
   373→   373→                <div className="space-y-2 max-h-[40vh] overflow-y-auto pr-1">
   374→   374→                  {cart.map((i) => (
   375→   375→                    <div
   376→   376→                      key={i.product.id}
   377→   377→                      className="flex items-center gap-2 p-2 rounded-md bg-muted/50"
   378→   378→                    >
   379→   379→                      <div className="flex-1 min-w-0">
   380→   380→                        <p className="text-sm font-medium truncate">{i.product.name}</p>
   381→   381→                        <p className="text-xs text-muted-foreground">
   382→   382→                          {formatCurrency(i.product.salePrice, symbol)} ·{" "}
   383→   383→                          {i.product.unit === "KG" ? "kg" : "u"}
   384→   384→                        </p>
   385→   385→                      </div>
   386→   386→                      <div className="flex items-center gap-1">
   387→   387→                        <Button
   388→   388→                          variant="outline"
   389→   389→                          size="icon"
   390→   390→                          className="h-7 w-7"
   391→   391→                          onClick={() => updateQty(i.product.id, -1)}
   392→   392→                        >
   393→   393→                          <Minus className="w-3 h-3" />
   394→   394→                        </Button>
   395→   395→                        <Input
   396→   396→                          type="number"
   397→   397→                          value={i.qty}
   398→   398→                          onChange={(e) =>
   399→   399→                            setQty(i.product.id, Number(e.target.value))
   400→   400→                          }
   401→   401→                          className="h-7 w-12 text-center px-1"
   402→   402→                        />
   403→   403→                        <Button
   404→   404→                          variant="outline"
   405→   405→                          size="icon"
   406→   406→                          className="h-7 w-7"
   407→   407→                          onClick={() => updateQty(i.product.id, 1)}
   408→   408→                        >
   409→   409→                          <Plus className="w-3 h-3" />
   410→   410→                        </Button>
   411→   411→                        <Button
   412→   412→                          variant="ghost"
   413→   413→                          size="icon"
   414→   414→                          className="h-7 w-7 text-red-600"
   415→   415→                          onClick={() => removeItem(i.product.id)}
   416→   416→                        >
   417→   417→                          <X className="w-3 h-3" />
   418→   418→                        </Button>
   419→   419→                      </div>
   420→   420→                      <p className="text-sm font-semibold w-20 text-right">
   421→   421→                        {formatCurrency(i.product.salePrice * i.qty, symbol)}
   422→   422→                      </p>
   423→   423→                    </div>
   424→   424→                  ))}
   425→   425→                </div>
   426→   426→              )}
   427→   427→
   428→   428→              {cart.length > 0 && (
   429→   429→                <div className="space-y-2 pt-3 border-t">
   430→   430→                  <div className="flex justify-between text-sm">
   431→   431→                    <span className="text-muted-foreground">Subtotal</span>
   432→   432→                    <span>{formatCurrency(subtotal, symbol)}</span>
   433→   433→                  </div>
   434→   434→                  <div className="flex justify-between text-sm items-center">
   435→   435→                    <span className="text-muted-foreground">Descuento</span>
   436→   436→                    <Input
   437→   437→                      type="number"
   438→   438→                      value={discount || ""}
   439→   439→                      placeholder="0"
   440→   440→                      onChange={(e) => setDiscount(Number(e.target.value))}
   441→   441→                      className="h-7 w-24 text-right"
   442→   442→                    />
   443→   443→                  </div>
   444→   444→                  {taxEnabled && taxRate > 0 && (
   445→   445→                    <div className="flex justify-between text-sm">
   446→   446→                      <span className="text-muted-foreground">
   447→   447→                        Impuesto ({taxRate}%)
   448→   448→                      </span>
   449→   449→                      <span>{formatCurrency(taxAmount, symbol)}</span>
   450→   450→                    </div>
   451→   451→                  )}
   452→   452→                  {surchargeAmount > 0 && (
   453→   453→                    <div className="flex justify-between text-sm text-amber-700">
   454→   454→                      <span>
   455→   455→                        Recargo {selectedMethod?.name} ({surchargeRate}%)
   456→   456→                      </span>
   457→   457→                      <span>+{formatCurrency(surchargeAmount, symbol)}</span>
   458→   458→                    </div>
   459→   459→                  )}
   460→   460→                  <div className="flex justify-between text-lg font-bold pt-2 border-t">
   461→   461→                    <span>Total</span>
   462→   462→                    <span className="text-emerald-700">
   463→   463→                      {formatCurrency(total, symbol)}
   464→   464→                    </span>
   465→   465→                  </div>
   466→   466→
   467→   467→                  <Button
   468→   468→                    className="w-full bg-emerald-600 hover:bg-emerald-700 h-11 text-base"
   469→   469→                    onClick={() => setCheckoutOpen(true)}
   470→   470→                  >
   471→   471→                    Cobrar {formatCurrency(total, symbol)}
   472→   472→                  </Button>
   473→   473→                </div>
   474→   474→              )}
   475→   475→            </CardContent>
   476→   476→          </Card>
   477→   477→        </div>
   478→   478→      </div>
