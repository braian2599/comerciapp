    59→    59→
    60→    60→export function ReportsView() {
    61→    61→  const { store } = useAppStore();
    62→    62→  const symbol = store?.currencySymbol || "$";
    63→    63→  const [tab, setTab] = useState<TabKey>("sales");
    64→    64→  const [from, setFrom] = useState("");
    65→    65→  const [to, setTo] = useState("");
    66→    66→  const [loading, setLoading] = useState(false);
    67→    67→  const [data, setData] = useState<any>(null);
    68→    68→
    69→    69→  // Defaults: últimos 30 días
    70→    70→  useEffect(() => {
    71→    71→    if (!from && !to) {
    72→    72→      const end = new Date();
    73→    73→      const start = new Date();
    74→    74→      start.setDate(start.getDate() - 30);
    75→    75→      setFrom(start.toISOString().slice(0, 10));
    76→    76→      setTo(end.toISOString().slice(0, 10));
    77→    77→    }
    78→    78→  }, []);
    79→    79→
    80→    80→  async function loadReport() {
    81→    81→    if (!from || !to) return;
    82→    82→    setLoading(true);
    83→    83→    setData(null); // reset para que no se renderice data de otro tab
    84→    84→    try {
    85→    85→      const params = new URLSearchParams({ from, to });
    86→    86→      const res = await fetch(`/api/reports/${tab}?${params}`);
    87→    87→      const json = await res.json();
    88→    88→      if (!res.ok) {
    89→    89→        toast.error(json.error || "Error cargando reporte");
    90→    90→        return;
    91→    91→      }
    92→    92→      setData(json);
    93→    93→    } catch (err) {
    94→    94→      toast.error("Error de conexión");
    95→    95→    } finally {
    96→    96→      setLoading(false);
    97→    97→    }
    98→    98→  }
    99→    99→
   100→   100→  useEffect(() => {
   101→   101→    if (from && to) loadReport();
   102→   102→  }, [tab, from, to]);
   103→   103→
   104→   104→  function setPeriod(days: number) {
   105→   105→    const end = new Date();
   106→   106→    const start = new Date();
   107→   107→    start.setDate(start.getDate() - days);
   108→   108→    setFrom(start.toISOString().slice(0, 10));
   109→   109→    setTo(end.toISOString().slice(0, 10));
   110→   110→  }
   111→   111→
   112→   112→  function exportarCSV() {
   113→   113→    if (!data) return;
   114→   114→    let rows: string[][] = [];
   115→   115→    let filename: string = tab;
   116→   116→
   117→   117→    if (tab === "sales" && data.series) {
   118→   118→      rows = [["Período", "Cantidad", "Total", "Descuentos", "Recargos"]];
   119→   119→      data.series.forEach((s: any) => {
   120→   120→        rows.push([s.period, s.count, s.total, s.discount, s.surcharge]);
   121→   121→      });
   122→   122→      filename = "ventas";
   123→   123→    } else if (tab === "profits" && data.series) {
   124→   124→      rows = [["Fecha", "Ventas", "Costo", "Gastos", "Ganancia Bruta", "Ganancia Neta"]];
   125→   125→      data.series.forEach((s: any) => {
   126→   126→        rows.push([s.date, s.ventas, s.costo, s.gastos, s.gananciaBruta, s.gananciaNeta]);
   127→   127→      });
   128→   128→      filename = "ganancias";
   129→   129→    } else if (tab === "products" && data.ranking) {
   130→   130→      rows = [["Producto", "Cantidad", "Ingresos", "Costo", "Ganancia", "Margen %", "Stock Actual"]];
   131→   131→      data.ranking.forEach((p: any) => {
   132→   132→        rows.push([p.productName, p.quantitySold, p.revenue, p.cost, p.profit, p.margin, p.currentStock]);
   133→   133→      });
   134→   134→      filename = "productos";
   135→   135→    } else if (tab === "customers" && data.topClientes) {
   136→   136→      rows = [["Cliente", "Compras", "Cantidad ventas", "Fiado", "Pagado"]];
   137→   137→      data.topClientes.forEach((c: any) => {
   138→   138→        rows.push([c.customerName, c.totalCompras, c.cantidadVentas, c.totalFiado, c.totalPagado]);
   139→   139→      });
   140→   140→      filename = "clientes";
   141→   141→    } else if (tab === "taxes" && data.byTipo) {
   142→   142→      rows = [["Tipo", "Cantidad", "Neto", "IVA", "Total"]];
   143→   143→      data.byTipo.forEach((t: any) => {
   144→   144→        rows.push([t.tipo, t.count, t.netoGravado, t.iva, t.total]);
   145→   145→      });
   146→   146→      filename = "fiscal";
   147→   147→    } else if (tab === "cashflow" && data.series) {
   148→   148→      rows = [["Fecha", "Ventas Efectivo", "Cobros Cuenta", "Ing. Manuales", "Gastos", "Egresos Manuales", "Flujo Neto"]];
   149→   149→      data.series.forEach((s: any) => {
   150→   150→        rows.push([s.date, s.ventasEfectivo, s.cobrosCuenta, s.ingresosManuales, s.gastosEfectivo, s.egresosManuales, s.flujoNeto]);
   151→   151→      });
   152→   152→      filename = "flujo-caja";
   153→   153→    }
   154→   154→
   155→   155→    const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n");
   156→   156→    const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
   157→   157→    const url = URL.createObjectURL(blob);
   158→   158→    const a = document.createElement("a");
   159→   159→    a.href = url;
   160→   160→    a.download = `reporte-${filename}-${new Date().toISOString().slice(0, 10)}.csv`;
   161→   161→    a.click();
   162→   162→    URL.revokeObjectURL(url);
   163→   163→  }
   164→   164→
   165→   165→  const TABS: { key: TabKey; label: string; icon: any }[] = [
   166→   166→    { key: "sales", label: "Ventas", icon: Receipt },
   167→   167→    { key: "profits", label: "Ganancias", icon: TrendingUp },
   168→   168→    { key: "taxes", label: "Fiscal", icon: BarChart3 },
   169→   169→    { key: "products", label: "Productos", icon: Package },
   170→   170→    { key: "customers", label: "Clientes", icon: Users },
   171→   171→    { key: "cashflow", label: "Flujo Caja", icon: Wallet },
   172→   172→  ];
   173→   173→
   174→   174→  return (
   175→   175→    <div className="p-4 space-y-4">
   176→   176→      <div className="flex flex-wrap items-center justify-between gap-3">
   177→   177→        <div>
   178→   178→          <h1 className="text-2xl font-bold flex items-center gap-2">
   179→   179→            <BarChart3 className="w-6 h-6 text-emerald-600" />
   180→   180→            Reportes
   181→   181→          </h1>
   182→   182→          <p className="text-sm text-muted-foreground">
   183→   183→            Análisis de ventas, ganancias, fiscal, productos y clientes
   184→   184→          </p>
   185→   185→        </div>
   186→   186→        <Button variant="outline" onClick={exportarCSV} disabled={!data}>
   187→   187→          <Download className="w-4 h-4 mr-2" />
   188→   188→          Exportar CSV
   189→   189→        </Button>
   190→   190→      </div>
   191→   191→
   192→   192→      {/* Tabs */}
   193→   193→      <div className="flex flex-wrap gap-2">
   194→   194→        {TABS.map((t) => {
   195→   195→          const Icon = t.icon;
   196→   196→          return (
   197→   197→            <Button
   198→   198→              key={t.key}
   199→   199→              variant={tab === t.key ? "default" : "outline"}
   200→   200→              size="sm"
   201→   201→              onClick={() => setTab(t.key)}
   202→   202→              className={tab === t.key ? "bg-emerald-600 hover:bg-emerald-700" : ""}
   203→   203→            >
   204→   204→              <Icon className="w-4 h-4 mr-1" />
   205→   205→              {t.label}
   206→   206→            </Button>
   207→   207→          );
   208→   208→        })}
   209→   209→      </div>
   210→   210→
   211→   211→      {/* Filtros */}
   212→   212→      <Card>
   213→   213→        <CardContent className="pt-4 flex flex-wrap items-end gap-3">
   214→   214→          <div>
   215→   215→            <Label className="text-xs">Desde</Label>
   216→   216→            <Input type="date" value={from} onChange={(e) => setFrom(e.target.value)} className="w-[160px]" />
   217→   217→          </div>
   218→   218→          <div>
   219→   219→            <Label className="text-xs">Hasta</Label>
   220→   220→            <Input type="date" value={to} onChange={(e) => setTo(e.target.value)} className="w-[160px]" />
   221→   221→          </div>
   222→   222→          <div className="flex gap-1">
   223→   223→            <Button size="sm" variant="outline" onClick={() => setPeriod(7)}>7 días</Button>
   224→   224→            <Button size="sm" variant="outline" onClick={() => setPeriod(30)}>30 días</Button>
   225→   225→            <Button size="sm" variant="outline" onClick={() => setPeriod(90)}>90 días</Button>
   226→   226→            <Button size="sm" variant="outline" onClick={() => setPeriod(365)}>1 año</Button>
   227→   227→          </div>
   228→   228→        </CardContent>
   229→   229→      </Card>
   230→   230→
   231→   231→      {/* Contenido */}
   232→   232→      {loading ? (
   233→   233→        <div className="flex items-center justify-center py-12">
   234→   234→          <Loader2 className="w-8 h-8 animate-spin text-emerald-600" />
   235→   235→        </div>
   236→   236→      ) : !data ? (
   237→   237→        <Card>
   238→   238→          <CardContent className="text-center py-12 text-muted-foreground">
   239→   239→            <Calendar className="w-12 h-12 mx-auto mb-2 opacity-30" />
   240→   240→            <p>Selecciona un rango de fechas</p>
   241→   241→          </CardContent>
   242→   242→        </Card>
   243→   243→      ) : (
   244→   244→        <>
   245→   245→          {/* VENTAS */}
   246→   246→          {tab === "sales" && <SalesReport data={data} symbol={symbol} />}
   247→   247→          {/* GANANCIAS */}
   248→   248→          {tab === "profits" && <ProfitsReport data={data} symbol={symbol} />}
   249→   249→          {/* FISCAL */}
   250→   250→          {tab === "taxes" && <TaxesReport data={data} symbol={symbol} />}
   251→   251→          {/* PRODUCTOS */}
   252→   252→          {tab === "products" && <ProductsReport data={data} symbol={symbol} />}
   253→   253→          {/* CLIENTES */}
   254→   254→          {tab === "customers" && <CustomersReport data={data} symbol={symbol} />}
   255→   255→          {/* FLUJO CAJA */}
   256→   256→          {tab === "cashflow" && <CashFlowReport data={data} symbol={symbol} />}
   257→   257→        </>
   258→   258→      )}
   259→   259→    </div>
   260→   260→  );
   261→   261→}
